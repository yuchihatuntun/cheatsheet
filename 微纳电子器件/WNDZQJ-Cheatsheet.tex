\section{第二章\ SOI\ MOS器件}

\subsection{SOI 技术的演变}

\textbf{\hlblue{SOI 核心定义}}

SOI 即 \textbf{Silicon-on-insulator}（绝缘体上硅）。\textbf{\textcolor{myred}{核心电学特征：沟道与衬底之间引入绝缘层进行电学隔离}}。

\imgleft[0.3]{images/image-2026-06-11-14-28-12.png}{
\textbf{\hlblue{三层物理结构（由上至下）}}

\concepttable{
    \conrow{表面硅薄层（顶层硅）}{用于制作器件的源（S）、漏（D）和沟道。}
    \conrow{埋层氧化物（BOX）}{通常为 $\text{SiO}_2$，起到关键的电学绝缘隔离作用。}
    \conrow{硅衬底（Si substrate）}{位于最底层，主要起机械支撑作用。}
}

\entry{技术演变}{SOI 并非一步到位，其前身是早期的 \textbf{SOS}（Silicon-on-Sapphire）技术。}
}

\subsubsection{SOS 技术——SOI 前身}

\entry{定义}{Silicon on Sapphire（蓝宝石上硅），诞生于1963年，是 SOI 的早期异质外延形态。}

\concepttable{
    \conrow{工艺原理}{在高纯度人工蓝宝石（$\text{Al}_2\text{O}_3$）晶体衬底上，外延生长单晶硅层。}
    \conrow{结构差异}{SOS \textbf{\textcolor{myred}{底部为整块蓝宝石绝缘衬底}}；现代 SOI 底部为硅衬底，中间嵌入 $\text{SiO}_2$（BOX）。}
}

\textbf{\hlblue{SOS / SOI 隔离器件的核心电学优势}}

\textbf{\textcolor{myred}{简答题：}}相比于传统体硅（Bulk Si）器件，SOS/SOI 等隔离技术具备哪些核心优势？

\concepttable{
    \conrow{强抗辐射}{绝缘衬底\textbf{\textcolor{myred}{阻断高能粒子在衬底中的电荷收集路径}}，有效抑制单粒子效应（SEE）。}
    \conrow{低寄生电容}{消除源/漏与衬底间 pn 结面积，\textbf{\textcolor{myred}{寄生电容大幅降低}}，提升频率、降低功耗。}
    \conrow{提高集成电路性能}{寄生效应减小直接带来\textbf{\textcolor{myred}{芯片速度的整体提高}}。}
    \conrow{消除闩锁效应}{体硅中寄生晶闸管导致闩锁；SOS/SOI 介质完全隔离各器件，切断寄生路径，彻底消除。}
}

\textbf{\hlblue{从 SOS 向现代 SOI 的转变}}

\concepttable{
    \conrow{SOS 的局限性}{20世纪80年代，SOS \textbf{\textcolor{myred}{价格昂贵}}、蓝宝石衬底成本极高，且与标准单晶硅工艺兼容性差，不适合普及和民用。}
    \conrow{现代 SOI 替代方案}{放弃蓝宝石，改用在常规硅衬底和表面硅薄层之间嵌入 $\text{SiO}_2$（BOX）绝缘层，降低成本并兼容主流硅工艺。}
    \conrow{标准 SOI 结构}{表面硅薄层$\rightarrow$ $\text{SiO}_2$ 绝缘层（BOX）$\rightarrow$ 硅衬底。}
}

\subsubsection{现代 SOI 晶圆制造}

业界制造SOI晶圆的方法都是利用嵌入或者键和的方法形成\textbf{\textcolor{myred}{埋层氧化物}}隔离顶层硅薄膜层和硅衬底

\concepttable{
    \conrow{工艺难点}{传统硅衬底/SOS 顶层单晶可直接在同质或异质单晶上外延生长；但在\textbf{\textcolor{myred}{非晶态 $\text{SiO}_2$ 表面无法直接外延生长单晶硅}}。}
    \conrow{解决思路}{不能直接生长，只能利用\textbf{\textcolor{myred}{嵌入或键合}}方法，在结构内部形成埋层氧化物（BOX），以此强行隔离顶层单晶硅薄膜和硅衬底。}
}

\textbf{\hlblue{主流 SOI 制造技术}}

\concepttable{
    \conrow{SIMOX（注入氧分离）}{\textbf{\textcolor{myred}{高剂量氧离子注入硅中}}，高温退火在内部形成 $\text{SiO}_2$ 层。}
    \conrow{BESOI（键合回蚀）}{\textbf{\textcolor{myred}{两片氧化后的硅片表面键合}}，再从背面回蚀减薄其中一片至所需薄膜厚度。}
    \conrow{Smart-Cut（智能剪切）}{利用氢离子注入在硅片内部形成损伤带，键合后通过热/机械剥离精准切断，\textbf{\textcolor{myred}{目前商业化最主流}}的高质量制造技术。}
}

\textbf{\hlblue{SIMOX技术}}

\imgleft[0.3]{images/image-2026-06-11-14-42-44.png}{

\concepttable{
    \conrow{定义}{即 Separation by Implanted Oxygen（注入氧分离技术），是最早出现的 SOI 晶圆制备技术之一。}
    \conrow{物理原理}{利用离子注入技术，将高剂量的氧离子（$\text{O}^+$）强行注入到单晶硅晶圆内部，\textbf{\textcolor{myred}{在特定深度通过化学反应形成 $\text{SiO}_2$ 埋层}}，以此将顶层单晶硅薄膜与底部硅衬底在电学上彻底隔离。}
}

\textbf{工艺优势：}

\concepttable{
    \conrow{工艺简单}{属于单片加工流程，不需要复杂的键合与减薄工序。}
    \conrow{硅膜质量好}{顶层单晶硅直接继承原硅片的晶格结构。}
    \conrow{产能高}{适合批量离子注入生产。}
}
}

\textbf{工艺流程：}

\imgleft[0.3]{images/image-2026-06-11-14-48-42.png}{
    \concepttable{
    \conrow{晶圆衬底裸片}{选择高纯度的标准单晶硅作为起始材料。}
    \conrow{氧离子注入}{使用高能量（120\textasciitilde200 keV）将高剂量氧离子注入硅晶圆。\textbf{\textcolor{myred}{高能量具有特定射程分布}}，氧离子集中分布在表面下方特定深度（峰值区）。}
    \conrow{高温退火}{注入完成后，进行 3\textasciitilde6 小时高温退火，氧离子与硅原子反应\textbf{\textcolor{myred}{生成厚度约 400 nm 的 $\text{SiO}_2$ 绝缘层}}（BOX）。}
}
}

\textbf{关键工艺参数}

\concepttable{
    \conrow{注入剂量}{\textbf{\textcolor{myred}{必须 $> 1.8 \times 10^{18}\text{/cm}^2$}}。
    
    \textbf{物理意义}：满足 $\text{Si:O} = 1:2$ ，剂量不足则无法形成连续氧化层，仅生成互不相连的 $\text{SiO}_2$ 沉淀物，隔离失败。}
    \conrow{注入温度}{\textbf{\textcolor{myred}{控制在 600°C\textasciitilde700°C}}。
    
    \textbf{物理意义}：注入过程中发生自退火效应，及时修复高能离子撞击造成的晶格损伤，保持顶层硅单晶状态。}
    \conrow{高温退火}{\textbf{\textcolor{myred}{温度必须 $> 1100\text{°C}$}}。
    
    \textbf{物理意义}：促使不连续 $\text{SiO}_2$ 沉淀物分解、合并为连续氧化层；彻底恢复顶层硅晶格结构，消除注入损伤。}
}

\textbf{技术局限性}

\concepttable{
    \conrow{结论}{\textbf{\textcolor{myred}{不能制备超薄}}SOI硅膜。}
    \conrow{原因}{氧离子注入深度和分布范围\textbf{\textcolor{myred}{受控于高能注入的统计学规律}}，顶层单晶硅薄膜厚度由注入能量和后续退火反应共同决定，无法在保证单晶质量的前提下无限减薄顶层硅。}
}

\textbf{\hlblue{BESOI技术}}

\entry{定义}{BESOI 即 Bond and Etch-back SOI（键合回蚀技术）。}

\textbf{基本原理与缺点}

\concepttable{
    \conrow{物理原理}{通过键合技术把两片标准硅晶圆紧密键合在一起，\textbf{\textcolor{myred}{晶圆之间预先通过热氧化形成的 $\text{SiO}_2$ 层作为埋层氧化物（BOX）}}。键合完成后，再利用回蚀技术把其中一侧晶圆的厚度削薄到目标厚度，形成SOI晶圆。}
    \conrow{核心缺点}{\textbf{\textcolor{myred}{均匀性差，薄膜难制备}}。通过机械或化学减薄（回蚀）大面积晶圆很难精准控制残留厚度，导致表面顶层硅的厚度均匀性极差。}
}

\textbf{四步标准工艺流程}

\concepttable{
    \conrow{准备两片裸片}{准备晶圆衬底裸片A和裸片B，均为标准硅单晶裸片。}
    \conrow{热氧化处理}{对晶圆B热氧化，表面生成 $\text{SiO}_2$ 绝缘层。\textbf{\textcolor{myred}{须控制氧化环境温度}}，降低氧化层与硅界面的缺陷和杂质。}
    \conrow{低温键合}{利用\textbf{\textcolor{myred}{硅熔融键合技术}}，将未氧化的裸晶圆A直接键合到晶圆B的 $\text{SiO}_2$ 层表面，形成"硅-氧化物-硅"三层夹心结构。}
    \conrow{回蚀减薄}{通过回蚀技术去除晶圆A多余厚度，使其变薄。最后利用退火和CMP（化学机械抛光）消除表面损伤，形成平滑清洁的晶圆表面。}
}

\entry{技术演进}{BESOI \textbf{\textcolor{myred}{最后一步通过外部机械/化学回蚀减薄无法做到高精度的均匀性}}，不能用于 sub-3nm 等先进节点所需的超薄顶层硅。业界为克服这一缺陷，由 BESOI 衍生出了目前最主流的 Smart-Cut 技术。}

\textbf{\hlblue{Smart-Cut技术}}

\textbf{基本概念}

\entry{定义}{Smart-Cut（智能剪切）技术是从 BESOI 衍生而来的\textbf{\textcolor{myred}{超薄 SOI 衬底主要制备方法}}。}

\concepttable{
    \conrow{起源}{从 BESOI 技术衍生而来，用氢离子注入替代机械/化学回蚀。}
    \conrow{核心机理}{准备两片硅晶圆（A和B）。对晶圆B热氧化生成 $\text{SiO}_2$ 绝缘层后，\textbf{\textcolor{myred}{再通过离子注入将 $\text{H}^+$ 注入晶圆B}}；晶圆A不做特殊处理。}
    \conrow{核心优势}{克服 SIMOX 和 BESOI 无法做到超薄、高均匀性顶层硅的缺陷，\textbf{\textcolor{myred}{是当前商业化最主流的 SOI 制造技术}}。}
}

\textbf{四步工艺流程}

\imgleft[0.25]{images/image-2026-06-11-14-54-43.png}{
    \concepttable{
    \conrow{热氧化 + $\text{H}^+$ 注入}{先对晶圆B表面热氧化生成 $\text{SiO}_2$，再注入高剂量 $\text{H}^+$。\textbf{\textcolor{myred}{离子注入具有射程确定性}}，$\text{H}^+$ 在晶圆B内部特定深度聚集，形成一层结构损伤带。}
    \conrow{低温键合}{将未处理的裸片A与晶圆B的 $\text{SiO}_2$ 表面进行紧密键合，形成"硅-氧化物-硅"三层结构。}
    \conrow{\textbf{\textcolor{myred}{热反应剥离}}}{将键合后的整体加热，损伤带处氢气膨胀，晶圆B沿损伤带\textbf{\textcolor{myred}{自动断裂/剥离}}。留在A侧的顶层硅薄膜厚度由 $\text{H}^+$ 注入射程精准决定，均匀性极高。随后高温退火加固键合，CMP处理平整表面。晶圆B剩余部分可回收重复使用。}
    }
}

\subsubsection{SOI 技术共性优缺点}

\textbf{\hlblue{优点}}

\concepttable{
    \conrow{工艺与成本}{与体硅为同平台技术，设计方案与制造工艺\textbf{\textcolor{myred}{完全兼容}}，节约制造成本。}
    \conrow{可靠性与速度}{见上文 SOI 技术的电学优势}
    \conrow{功耗}{结漏电小，适于低功耗电路：源漏区与衬底隔离后，反向源漏结泄漏电流显著减小，\textbf{\textcolor{myred}{静态功耗大幅降低}}。}
}

\textbf{\hlblue{缺点}}

\concepttable{
    \conrow{\textbf{\textcolor{myred}{浮体效应（FBE）}}}{\textbf{机理}：顶层硅沟道（Body）被栅氧化层、源漏结及底部 BOX 完全包围，\textbf{\textcolor{myred}{无零电位参考面}}，体区电位悬空/浮空。
    
    \textbf{危害}：体区电位随前一时刻源漏电压操作历史变化（历史效应），导致 AC 特性不确定，电路仿真需极精确模型，否则时序失配。}
    \conrow{\textbf{\textcolor{myred}{自热效应（SHE）}}}{\textbf{机理}：埋层 $\text{SiO}_2$ 导热系数远低于单晶硅，\textbf{\textcolor{myred}{沟道焦耳热无法通过衬底有效耗散}}，热量积聚使器件温度升高。
    
    \textbf{危害}：温度升高 $\rightarrow$ 载流子迁移率显著降低 $\rightarrow$ 饱和电流退化。}
    \conrow{Scaling 需极薄硅膜}{为抑制短沟道效应，SOI 按比例缩小时必须同步减薄顶层硅至纳米级。极薄硅膜厚度控制极难，原子层级厚度偏差即可导致严重器件性能波动（如 $V_T$ 波动）。}
}

    \conrow{耗尽区分布}{栅氧化层下方\textbf{正界面}与埋氧化层上方\textbf{背界面}各存在一个耗尽区，\textbf{\textcolor{myred}{两耗尽区互不重合}}。}

\textbf{\hlblue{固有缺陷——浮体效应（FBE）}}

\concepttable{
    \conrow{成因}{PD-SOI 器件中，夹在中间的\textbf{\textcolor{myred}{中性体区}}（阱区）\textbf{\textcolor{myred}{无连接固定电压电极}}。由于它四周被栅氧化层、源漏结和底部 BOX 完全包围隔离，该中性体区在电学上处于\textbf{悬空状态}（Floating Body）。}
}

\textbf{（1）翘曲效应（Kink-effect）}

\concepttable{
    \conrow{产生机理}{PD-SOI NMOS 高 $V_D$ 下，沟道电子在漏端耗尽区高电场加速获得高能$\rightarrow$\textbf{碰撞电离} 产生电子-空穴对。电子迅速入漏极，空穴滞留浮体区$\rightarrow$空穴积聚$\rightarrow$浮体电位升高$\rightarrow$体-源正偏$\rightarrow$体区势垒降低$\rightarrow$开启电压减小$\rightarrow$$I_D$ 不再饱和而迅速增加，在 $I_D$-$V_D$ 曲线上出现\textbf{台阶式突起}。}
    \conrow{极性差异}{\textbf{NMOS 显著，PMOS 不明显}。原因：空穴电离率远低于电子，PMOS 漏端碰撞电离概率低。}
    \conrow{数字电路影响}{有一定好处：饱和漏电流突增$\rightarrow$最大电流增大$\rightarrow$提高数字开关电路翻转速度。}
    \conrow{模拟电路影响}{有严重危害：跨导 $g_m$ 突然增加$\rightarrow$恶化输出阻抗 $r_o$ 和增益 $A_v$$\rightarrow$电路工作不稳定。}
    \conrow{抑制方法}{\textbf{体接触}技术：将阱体区引出接固定电位，导走碰撞电离多余空穴，控制体电势$\rightarrow$稳定源漏电路。}
}

\textbf{（2）寄生双极晶体管效应（Parasitic Bipolar Transistor Effect）}

也是因碰撞电离而生的。

\concepttable{
    \conrow{结构基础}{PD-SOI 器件的源区（$n^+$）-中性体区（$p$）-漏区（$n^+$）天然构成寄生 \textbf{$n$-$p$-$n$ 双极型晶体管}。}
    \conrow{触发条件}{碰撞电离$\rightarrow$体区电势上升$\rightarrow$源-体 $pn$ 结正向偏置$\rightarrow$寄生 BJT 导通。}
    \conrow{正反馈机制}{沟道电流 $I_c$ 在漏端碰撞电离产生空穴流入浮体区$\rightarrow$作为寄生 BJT 基极电流 $I_b$，经 $\beta$ 放大为 $\beta I_b$$\rightarrow$总漏端电流 $I_d = M(I_c + \beta I_b)$（$M$ 为碰撞电离倍增因子）$\rightarrow$漏电流增大$\rightarrow$更多空穴反馈回基极$\rightarrow$形成\textbf{\textcolor{myred}{内部正反馈}}。当 $V_D$ 足够大时，正反馈导致器件发生\textbf{击穿（Snapback）}。}
    \conrow{抑制手段}{\textbf{体接触}技术：将体区（寄生BJT基极）引出接固定电位。}
    \conrow{物理原因}{体接触电极导出体区多子（NMOS为空穴），削弱空穴堆积$\rightarrow$直接控制基区电势$\rightarrow$\textbf{\textcolor{myred}{阻断源-体 $pn$ 结正偏}}$\rightarrow$切断正反馈路径。}
    \conrow{IV曲线特征}{无体接触时低电压下即因正反馈提前击穿（曲线陡峭上冲）；有体接触后击穿电压显著提高。}
}

\textbf{（3）自热效应（Self-heating Effect）}

\concepttable{
    \conrow{成因}{埋层氧化物（BOX，$\text{SiO}_2$）提供电学隔离的同时也造成\textbf{热隔离}。单晶硅（Si）导热系数 $G_{Th}=148\text{ W/(m}\cdot\text{K)}$，$\text{SiO}_2$ 仅 $1.4\text{ W/(m}\cdot\text{K)}$，\textbf{\textcolor{myred}{相差两个数量级}}。}
    \conrow{物理过程}{沟道大电流产生焦耳热$\rightarrow$热量无法经底部 BOX 层有效耗散$\rightarrow$\textbf{\textcolor{myred}{顶层硅薄膜内热量堆积}}。}
    \conrow{电学危害}{温度剧烈升高$\rightarrow$晶格散射加强$\rightarrow$载流子迁移率下降$\rightarrow$饱和电流退化。}
    \conrow{曲线特征}{$I_D$-$V_D$ 输出特性曲线饱和区出现\textbf{负微分电阻}：漏电压增大时，漏电流反而降低（曲线向下倾斜）。}
}

}

\concepttable{
    \conrow{浮体效应抑制}{FD-SOI 饱和区工作时顶层硅薄膜\textbf{全耗尽}。碰撞电离空穴在源区附近即被复合，\textbf{\textcolor{myred}{体内不发生电荷积累}}，从根本上消除 PD-SOI 的浮体效应。}
    \conrow{短沟道效应抑制}{FD-SOI 结深极薄$\rightarrow$源漏耗尽层横向扩散宽度大幅减小$\rightarrow$\textbf{\textcolor{myred}{栅极对沟道电场控制力增强}}。}
    \conrow{背面偏置能力}{超薄 BOX 存在，可\textbf{\textcolor{myred}{对衬底施加 Back-bias}}，动态调节阈值电压。}
    \conrow{极低电源电压}{最小工作电压可达 $0.4\text{ V}$，是超低功耗的核心支撑。}
}

\textbf{\hlblue{FD-SOI 的工艺流程：前栅HKMG和应变硅技术}}

\textbf{1. 前栅HKMG工艺}

\entry{工艺类型}{FD-SOI采用\textbf{前栅（Gate-First）}工艺：在高温退火前完成完整栅极堆叠（High-k介质 + 金属栅极)}

\entry{栅极堆叠}{高k介质（HfO₂等）直接沉积在超薄硅膜（$<\text{10 nm}$）上，替代传统SiO₂栅氧层，\textbf{\textcolor{myred}{解决栅极漏电流问题}}。}

\textbf{工艺流程}

\concepttable{
    \conrow{界面层生长}{在超薄SOI硅膜上生长界面层（IL，通常为SiO$_\text{x}$或SiON）。}
    \conrow{High-k沉积}{沉积High-k介质层（HfO$_2$等）。}
    \conrow{金属栅沉积}{沉积功函数金属（TiN、TaN等）+ 栅极填充金属。}
    \conrow{图形化与刻蚀}{进行栅极图形化和刻蚀。}
    \conrow{高温退火}{激活源漏掺杂时栅极堆叠已存在$\rightarrow$需选用热稳定性好的材料组合。}
}

\textbf{FD-SOI特有优势}

\concepttable{
    \conrow{栅控增强}{全耗尽$\rightarrow$栅极对沟道控制力更强$\rightarrow$HKMG配合实现低漏电、高驱动电流。}
    \conrow{界面态可控}{超薄硅膜$\rightarrow$栅极电场更均匀$\rightarrow$High-k界面态问题相对容易控制。}
}

\textbf{2. 应变硅技术}

\entry{应变引入机制}{FD-SOI中通常采用全局应变或局部应变技术，实现方式与体硅不同。}

\textbf{三种工艺}

\concepttable{
    \conrow{\textbf{SMT}（应力记忆技术）}{源漏区沉积高应力SiN盖帽层，退火使应力"记忆"在硅膜中，去除盖帽层后应变保留在沟道。}
    \conrow{\textbf{epi-S/D}（源漏外延生长）}{源漏凹槽中外延生长SiGe（pMOS）或SiC（nMOS）等晶格失配材料；BOX隔离使外延层对超薄硅膜产生单轴压应变（pMOS）或张应变（nMOS）；BOX层允许更大晶格失配容忍度。}
    \conrow{\textbf{CESL}（接触刻蚀阻挡层）}{沉积高应力SiN作为接触刻蚀阻挡层，直接对沟道施加张应力（nMOS）或压应力（pMOS）。}
}

\textbf{\hlblue{FD-SOI的应变优势}}

\concepttable{
    \conrow{BOX机械隔离}{硅膜下方为BOX而非体硅，应变更高效保留在沟道，减少应力向衬底泄漏。}
    \conrow{超薄硅膜高灵敏度}{硅膜越薄，相同应力下应变效果越显著，载流子迁移率提升更明显。}
    \conrow{双轴→单轴自然转换}{FD-SOI平面结构结合局部应力技术，更易实现方向性应变，优化电子和空穴迁移率。}
}

\imgleft[0.3]{images/image-2026-06-12-05-08-33.png}{
\concepttable{
    \conrow{电学耦合状态}{FD SOI 硅膜厚度 $T_{si}$ 小于正栅和背栅最大耗尽层宽度之和$\rightarrow$正栅与背栅\textbf{\textcolor{myred}{相互耦合}}$\rightarrow$\textbf{\textcolor{myred}{两者 $V_T$ 相互影响}}。}
    \conrow{理想简化}{忽略源/漏短沟道效应，仅考虑\textbf{\textcolor{myred}{一维（1D）SOI MOS 结构}}，沿深度方向（$x$ 轴）做一维静电学分析。}
    \conrow{核心方程}{通过求解一维泊松方程，得到阈值电压 $V_T$。}
}
}

\textbf{积分求解结果：}

在正表面（$x = 0$ 处），电势为正表面电势 $\phi_{sf}$；

在背表面（$x = T_{si}$ 处），电势为背表面电势 $\phi_{sb}$

\concepttable{
    \conrow{电势分布}{$\phi(x) = \frac{qN_Ax^2}{2\varepsilon_s} + \left( \frac{\phi_{sb} - \phi_{sf}}{T_{si}} - \frac{qN_AT_{si}}{2\varepsilon_s} \right)x + \phi_{sf}$}
    \conrow{电场分布}{对电势求负导数：$E(x) = -\frac{qN_Ax}{\varepsilon_s} + \frac{\phi_{sf} - \phi_{sb}}{T_{si}} + \frac{qN_AT_{si}}{2\varepsilon_s}$}
}

\concepttable{
    \conrow{正表面电场}{$E_{sf} = \dfrac{\phi_{sf} - \phi_{sb}}{T_{si}} + \dfrac{qN_AT_{si}}{2\varepsilon_s}$}
    \conrow{背表面电场}{$E_{sb} = -\dfrac{qN_AT_{si}}{\varepsilon_s} + E_{sf}$}
}

\textbf{氧化层电压降公式：}

设正、背栅氧化层电容分别为 $C_{oxf}$ 和 $C_{oxb}$。在忽略界面电荷和氧化层内部电荷的前提下，根据高斯定律连续性条件：

\concepttable{
    \conrow{正栅氧化层电压降}{$V_{oxf} = E_{oxf}T_{oxf} = \dfrac{\varepsilon_s E_{sf}}{\varepsilon_{ox}}T_{oxf} = \dfrac{\varepsilon_s E_{sf}}{C_{oxf}}$}
    \conrow{背栅氧化层电压降}{$V_{oxb} = E_{oxb}T_{oxb} = -\dfrac{\varepsilon_s E_{sb}}{\varepsilon_{ox}}T_{oxb} = -\dfrac{\varepsilon_s E_{sb}}{C_{oxb}}$}
}

\textbf{正/背栅电压最终表达式与电荷耦合:}

\concepttable{
    \conrow{正栅电压}{$$V_{Gf} = V_{FBf} + \left(1 + \frac{C_{Si}}{C_{oxf}}\right)\phi_{sf} - \frac{C_{Si}}{C_{oxf}}\phi_{sb} + \frac{qN_AT_{si}}{2C_{oxf}}$$}
    \conrow{背栅电压}{$$V_{Gb} = V_{FBb} - \frac{C_{Si}}{C_{oxb}}\phi_{sf} + \left(1 + \frac{C_{Si}}{C_{oxb}}\right)\phi_{sb} + \frac{qN_AT_{si}}{2C_{oxb}}$$}
    \conrow{硅膜电容}{$C_{Si} = \dfrac{\varepsilon_s}{T_{si}}$}
}

\textbf{电荷耦合：}

\concepttable{
    \conrow{耦合关系}{正栅电压 $V_{Gf}$ 直接受背表面电势 $\phi_{sb}$ 控制；背栅电压 $V_{Gb}$ 直接受正表面电势 $\phi_{sf}$ 控制。}
    \conrow{物理结论}{两方程定量反映正/背栅间电荷耦合，数学上证实 FD SOI 器件阈值电压相互影响。}
}

\textbf{\textcolor{myred}{阈值电压：}}

\textbf{情况 1 —— 背面积累时的阈值电压模型}

\concepttable{
    \conrow{边界条件}{正表面：$\phi_{sf} = 2\phi_{Fp}$（强反型导通）；背界面：$\phi_{sb} \sim 0$（积累态，电势几近为零）}
    \conrow{阈值电压 $V_{T,ba}$}{将边界条件代入正栅压通用公式，消除 $\phi_{sb}$ 项：$$V_{T,ba} = V_{FBf} + \left(1 + \frac{C_{Si}}{C_{oxf}}\right)2\phi_{Fp} + \frac{qN_AT_{si}}{2C_{oxf}}$$}
    \conrow{曲线特征}{背面多子积累，形成了静电屏蔽，使得背栅电压的变化无法穿透到正面，因此 $V_T$ 不随 $V_{Gb}$ 改变。}
}

\textbf{情况 2 —— 背面反型时的阈值电压模型}

\concepttable{
    \conrow{物理边界条件}{正背界面电势\textbf{\textcolor{myred}{均达强反型}}：$\phi_{sf} = \phi_{sb} = 2\phi_{Fp}$。}
    \conrow{阈值电压 $V_{T,bi}$}{代入正栅压通用公式，$\phi_{sf}$ 与 $\phi_{sb}$ 项部分抵消，化简得：$$V_{T,bi} = V_{FBf} + 2\phi_{Fp} + \frac{qN_AT_{si}}{2C_{oxf}}$$}
    \conrow{曲线特征}{背面由于强反型产生了高浓度的反型电荷层，再次起到静电屏蔽作用，阻断了正背栅之间的进一步耦合，使 $V_T$ 达到饱和最小值不再发生改变。}
}

\textbf{情况 3 —— 背面耗尽时的阈值电压模型}

\imgleft[0.3]{images/image-2026-06-12-10-06-09.png}{
    \concepttable{
    \conrow{物理边界条件}{正面电势 $\phi_{sf} = 2\phi_{Fp}$（强反型）。}
    \conrow{背表面电势特征}{从积累态 $0$ 到反型态 $2\phi_{Fp}$。}
    \conrow{阈值电压 $V_{T,bd}$}{在积累型与反型态之间呈线性函数关系。}
}

由$V_{Gb,a} = V_{FBb} - \frac{C_{Si}}{C_{oxb}}2\phi_{Fp} + \frac{qN_AT_{si}}{2C_{oxb}}$与$V_{Gb,i} = V_{FBb} + 2\phi_{Fp} + \frac{qN_AT_{si}}{2C_{oxb}}$

所以$V_{T,bd} = V_{T,ba} - \frac{C_{Si}C_{oxb}}{C_{oxf}(C_{Si}+C_{oxb})}(V_{Gb} - V_{Gb,a}) = V_{T,bi} - \frac{C_{Si}C_{oxb}}{C_{oxf}(C_{Si}+C_{oxb})}(V_{Gb} - V_{Gb,i})$

}

\subsubsection{电流电压特性}

\textbf{自变量依赖关系}

由于全耗尽（FD）器件中背栅与正栅存在电荷耦合，所以$I_D$必须表示为三自变量电压的函数：
$$I_D = I_D(V_{Gf}, V_{Gb}, V_D)$$

\textbf{五大理论假设}

推导基本电流公式时，\textbf{\textcolor{myred}{完全沿用了体硅器件的物理假设}}：

\concepttable{
    \conrow{恒定迁移率}{$\mu$ 保持不变}
    \conrow{缓变沟道近似（GCA）}{横向电场远小于纵向电场}
    \conrow{均匀掺杂}{沟道掺杂浓度处处一致}
    \conrow{忽略扩散电流}{仅考虑漂移电流}
    \conrow{仅正面沟道导电}{背面只负责电学耦合调控，不形成漏极电流导通沟道}
}

\textbf{基础公式结构}

在上述假设下，电流公式的基本框架依然符合标准的体硅平面 MOSFET 线性区电流方程：

$$I_D = \beta \left[ (V_{GS} - V_T)V_{DS} - \frac{1}{2}V_{DS}^2 \right]$$

\textbf{\blue{背面耗尽态电流公式与定性规律}}

\concepttable{
    \conrow{电流公式}{$I_{D,d} = \frac{Z}{L}\mu_{nf}C_{oxf}\left[ (V_{Gf} - V_{T,bd})V_D - \left( 1 + \frac{C_{Si}C_{oxb}}{(C_{Si}+C_{oxb})C_{oxf}} \right) \frac{V_D^2}{2} \right]$}
    \conrow{$\frac{V_D^2}{2}$ 修正系数}{$\frac{C_{Si}C_{oxb}}{(C_{Si}+C_{oxb})C_{oxf}}$ 恰为 $V_T - V_{Gb}$ 关系曲线在耗尽区直线的绝对斜率值}
}

\textbf{背栅压对电流的定性调控}

\concepttable{
    \conrow{结论}{背面耗尽时，$I_D$ 随 $V_{Gb}$ 增大而增大}
    \conrow{因果链}{$V_{Gb} \uparrow \;\rightarrow\; V_{T,bd}$ 线性下降（正背栅耦合）$\;\rightarrow\; (V_{Gf}-V_{T,bd}) \uparrow \;\rightarrow\;$ 反型层电荷密度 $\uparrow \;\rightarrow\; I_D \uparrow$}
}

\subsubsection{亚阈值效应}

\textbf{\hlblue{亚阈值斜率}}

\concepttable{
    \conrow{定义}{$$S \equiv \frac{dV_{GS}}{d(\lg I_{Dsub})} = \ln(10)\frac{nkT}{q} = \ln(10)\left[1 + \frac{C_D(\phi_s)}{C_{ox}}\right]\frac{kT}{q}$$}
    \conrow{物理意义}{表征栅电压对亚阈值漏电流的控制能力：$S$ 越小，栅控能力越强，关态向开态转换越快。}
}

\textbf{\hlblue{背面耗尽态下的 $S_{bd}$}}

\entry{背景}{当背面处于耗尽时，背表面电势 $\phi_{sb}$ 从积累时的 $0$ 变化到反型时的 $2\phi_{Fp}$。}

\concepttable{
    \conrow{$S_{bd}$ 表达式}{$$S_{bd} = (\ln 10)\frac{kT}{q}\left[1 + \frac{C_{Si}C_{oxb}}{C_{oxf}(C_{Si} + C_{oxb})}\right]$$}
}

一般情况下，埋氧化层（BOX）厚度远大于顶层硅层厚度和正栅氧化层厚度。电容大小与介质厚度成反比，因此：

$$C_{oxb} \ll C_{Si} \quad \text{和} \quad C_{oxb} \ll C_{oxf}$$

\concepttable{
    \conrow{结论}{将上述近似关系代入 $S_{bd}$ 公式，$C_{oxb}$ 极小使 $\frac{C_{Si}C_{oxb}}{C_{oxf}(C_{Si} + C_{oxb})} \to 0$，背面耗尽时亚阈值斜率近似理想值：$$S \approx (\ln 10)\frac{kT}{q}$$}
    \conrow{电学优势}{背面耗尽器件亚阈值斜率小，关态泄漏电流小，有利于大幅降低芯片静态功耗。}
}

\concepttable{
    \conrow{杂质数目涨落}{沟道内杂质原子总数的统计起伏。}
    \conrow{杂质位置涨落}{杂质原子在耗尽层内的具体坐标随机变化。}
}

RDD通过掺杂浓度 $N_A$ 的起伏\textbf{\textcolor{myred}{映射为阈值电压 $V_T$ 波动}}：
$$V_T = \phi_{MS} - \frac{Q_{ox}}{C_{ox}} + \frac{K(2\phi_{FP})^{\frac{1}{2}}}{C_{ox}} + 2\phi_{FP}$$
其中系数 $K \propto \sqrt{N_A}$，掺杂浓度的微观起伏直接传导至 $V_T$。

\textbf{\hlblue{RDD 定量影响}}

\entry{掺杂浓度相对涨落}{$\frac{\sigma_{N_A}}{N_A} \approx 2.4\%$}
\entry{沟道杂质原子数相对起伏}{$\frac{\sigma_{n_a}}{n_a} \approx 2.3\%$}

\imgleft[0.25]{images/image-2026-06-09-20-54-07.png}{
$\sigma_{V_T}$（阈值电压标准差）随 $L_{eff}$ （有效栅长）缩小的演进：

\concepttable{
    \conrow{$L_{eff} = 0.5\,\mu\text{m}$}{$\sigma_{V_T} = 6.9\text{ mV}\ (\approx 1.4\%)$}
    \conrow{$L_{eff} = 0.3\,\mu\text{m}$}{$\sigma_{V_T} = 10.6\text{ mV}\ (\approx 2.2\%)$}
}

\textbf{\textcolor{myred}{随 $L_{eff}$ 缩小，$\sigma_{V_T}$ 显著增大，分布（近高斯）变宽变矮}}，器件间性能离散度增加。影响 $\sigma_{V_T}$ 的设计变量还包括栅氧化层厚度、栅宽及掺杂浓度本身。
}

\textbf{\hlblue{$\sigma_{V_T}$ 对结构与工艺参数的依赖关系}}

\concepttable{
    \conrow{栅氧化层厚度 $t_{ox}$}{$t_{ox}\uparrow\ \Rightarrow\ \sigma_{V_T}\uparrow$。机理：$t_{ox}$ 增大 $\rightarrow$ $C_{ox}$ 减小 $\rightarrow$ 栅对沟道电荷控制减弱，微观电荷起伏对 $V_T$ 的调制放大。}
    \conrow{器件面积 $W\times L$}{$WL\downarrow$（即 $1/\sqrt{LW}\uparrow$）$\Rightarrow$ $\sigma_{V_T}\uparrow$。机理：符合面积平方根倒数定律，\textbf{\textcolor{myred}{沟道内杂质原子绝对数越少，统计起伏越剧烈}}。}
    \conrow{掺杂浓度 $N$}{$N\uparrow\ \Rightarrow\ \sigma_{V_T}\uparrow$（$\sigma_{V_T}\propto N^{1/4}$）。机理：耗尽区电荷总量随 $N$ 增大而增大，\textbf{\textcolor{myred}{绝对电荷涨落总量同步上升}}。}
}

\textbf{\hlblue{RDD 第二维度：杂质空间位置随机性}}

\concepttable{
    \conrow{控制变量}{固定沟道耗尽层内杂质原子数 $\equiv 170$}
    \conrow{现象}{$V_T$ 仍从 $\sim 0.5\text{ V}$ 至 $\sim 0.8\text{ V}$ 宽幅离散}

\subsubsection{线条边沿粗糙（LER）}

\entry{线条边沿粗糙（LER）}{\textbf{\textcolor{myred}{栅极线条边沿的原子级不规则凹凸不平}}，导致沿栅宽 $W$ 方向局域实际栅长随机变异。}

\imgleft[0.2]{images/image-2026-06-09-21-11-03.png}{

\concepttable{
    \conrow{理想版图}{栅极为规则直线矩形，具有均匀栅长 $L$ 与栅宽 $W$。}
    \conrow{实际形貌}{光刻光子散粒效应与刻蚀离子轰击不均匀 $\rightarrow$ 边沿非陡直，\textbf{\textcolor{myred}{SEM 下呈现微观粗糙}}。}
    \conrow{核心问题}{随器件尺寸缩小，LER 如何演变？$\rightarrow$ 工艺极限下的缩放规律。}
}

}

\entry{工艺极限下 LER 的非比例缩放}{经厂商实验证明：\textbf{\textcolor{myred}{各工艺节点下 LER 绝对值基本保持在 $\sim 5\text{ nm}$ 不变}}。\textbf{\textcolor{myred}{栅 LER 不能按比例缩小}}。随 $L$ 持续微缩，LER 相对栅长占比急剧飙升，$5\text{ nm}$ 级粗糙度对沟道几何产生主导性破坏，导致器件参数剧烈涨落。}

\textbf{\hlblue{LER 数学表征}}

方均根值 $\Delta$（RMS）：

$\Delta = \left( \frac{1}{n} \sum_{i=1}^n \delta_i^2 \right)^{\frac{1}{2}}$

$\delta_i$：实际栅边沿与理想边沿的几何位置偏差；$n$：沿线条方向采样点数目。\textbf{\textcolor{myred}{工业界常用 $3\Delta$（3 倍 RMS）表征 LER}}，覆盖约 99.73$\%$ 统计波动。

\imgleft[0.35]{images/image-2026-06-09-21-17-11.png}{

\entry{从 LER 到 LWR（线宽粗糙度）}{MOSFET 沟道由\textbf{双侧}栅边缘共同定义，故引入线条宽度粗糙度（LWR）：$\Delta_{LWR} = \left[ \frac{1}{n} \sum_{i=1}^n (L_{gi} - L_g)^2 \right]^{\frac{1}{2}}$

其中：$L_{gi}$：局域实际栅线条宽度；$L_g$：平均栅长（设计标称值）。}

\textbf{双侧相关系数}：

\concepttable{
    \conrow{$\rho = 0$（最常见）}{两侧粗糙不相关，独立叠加：$\sigma_{LWR}^2 = 2\sigma_{LER}^2$。}
    \conrow{$\rho = -1$（完全负相关）}{互补波浪状，线宽变异最大：$\sigma_{LWR}^2 = 4\sigma_{LER}^2$。}
    \conrow{$\rho = 1$（完全正相关）}{整体弯曲但宽度恒定，线宽无起伏：$\sigma_{LWR}^2 = 0$。}
}

}

\textbf{\hlblue{LER/LWR 对器件电学性能的影响}}

\entry{尺寸依赖性}{参数标准差随 $W$ 减小而显著增大。}

\entry{电学不对称响应}{\textbf{\textcolor{myred}{LER 对 $I_{on}$ 影响小，对 $I_{off}$ 影响显著}}。机理：$I_{on}$ 取决于沟道平均电阻（并联沟道的线性平均效应），局部栅长起伏被均值平抑；$I_{off}$ 与栅长呈指数反比（DIBL 在短栅长处加剧），LWR 导致局域栅长极度缩短处漏电流指数暴增，拉高整体 $I_{off}$。}

\textbf{\hlblue{LER 对电学参数的量化影响}}

\entry{阈值电压 $V_T$ 起伏}{$\sigma_{V_T}$ 随方均根涨落 $\Delta$ 单调上升；相同 $\Delta$ 下，$L=30\text{ nm}$ 的 $\sigma_{V_T}$ 斜率远高于 $L=50\text{ nm}$，表明\textbf{\textcolor{myred}{器件缩小剧烈放大 LER 对 $V_T$ 的扰动}}，并伴随 $V_T$ 降低效应。}

\entry{关态电流 $I_{off}$ 起伏}{$\sigma_{\log_{10}I_{off}}$ 随 $\Delta$ 增大而上升，$V_D = 1.0\text{ V}$ 时波动远大于 $V_D = 0.1\text{ V}$。机理：高漏压下 DIBL 加剧，LER 致局域栅长极度缩短处势垒严重下陷，\textbf{\textcolor{myred}{局域漏电指数暴增}}。}

\entry{开态电流 $I_{on}$ 对比}{$\sigma_{I_{on}}$ 虽随 $\Delta$ 上升，但斜率与幅度远小于 $I_{off}$，\textbf{\textcolor{myred}{印证 LER 对 $I_{off}$ 影响远大于 $I_{on}$}}。}

\textbf{\hlblue{工艺解决方案：侧墙光刻技术}}

\entry{原理}{以 ALD 薄膜厚度与各向异性刻蚀的物理边界定义图形，\textbf{\textcolor{myred}{避开传统光刻光子散射造成的边界畸变}}。}

\concepttable{
    \conrow{常规光刻}{光刻胶分子随机溶解度变异与化学放大胶随机扩散 → 栅边缘锯齿状畸变 → 不同断面 $L_g$ 空间变异剧烈，易诱发局域漏电。}
    \conrow{侧墙光刻}{边缘由均匀薄膜沉积外延侧壁定义 → 沿栅宽方向 $L_g$ 高度一致，消除几何畸变引入的局域短沟道效应。}
}

\subsubsection{栅材料的功函数变化}

\entry{材料依赖性（栅面积 $(16\text{ nm})^2$）}{$\sigma_{V_{th}}$ 由低到高：$\text{TiN}\,(98\text{ mV}) < \text{TaN}\,(120\text{ mV}) < \text{WN}\,(157\text{ mV}) < \text{MoN}\,(295\text{ mV})$。\textbf{\textcolor{myred}{晶粒尺寸与晶向各向异性决定涨落敏感度}}。}

\subsection{存储器概述}

\textbf{\textcolor{myred}{"动态"物理根源}}：DRAM存"1"时pn结反偏$\to$反向泄漏电流$I_{leak}$致电容电荷流失、电平衰退，须周期性刷新。

\concepttable{

    \conrow{保持时间}{$t_h = \dfrac{0.2 V_{S1} C_s}{I_{leak}}$，\textbf{\textcolor{myred}{$C_s$越大、$I_{leak}$越小则保持越久}}；系数0.2为允许电压衰减裕度。}
}

\textbf{\hlblue{读操作与电荷共享}}

\noindent\textbf{流程}：BL预充至$V_R$ $\to$ WL拉高 $\to$ $C_B$与$C_s$连通 $\to$ 电荷共享，电平趋于一致。

\imgleft[0.4]{images/image-2026-06-08-20-48-14.png}{

\concepttable{
    \conrow{读"0"终态}{$V_{B0} = \dfrac{C_B V_R + C_s V_{S0}}{C_B + C_s}$，检测量 $\Delta V_0 = V_R - V_{B0}$。}
    \conrow{读"1"终态}{$V_{B1} = \dfrac{C_B V_R + C_s V_{S1}}{C_B + C_s}$，检测量 $\Delta V_1 = V_{B1} - V_R$。}
}

\noindent\textbf{两大致命问题}：
\begin{itemize}
    \item \textbf{读出信号微弱}：$C_B \gg C_s$ $\Rightarrow$ \textbf{\textcolor{myred}{$\Delta V$极小，后续电路难以识别}}。
    \item \textbf{破坏性读出}：读后$C_s$电压变为中间电平$V_{B0}$/$V_{B1}$，原存储数据丢失。
\end{itemize}

}

\textbf{\hlblue{灵敏再生放大器（S/R）:}}

\noindent 解决读出信号微弱与破坏性读出两大问题。\textbf{结构}：交叉耦合反相器（两对PMOS+NMOS）构成\textbf{\textcolor{myred}{双稳态触发器}}。

\concepttable{
    \conrow{放大}{将微弱$\Delta V$放大至满幅逻辑电平（$V_{DD}$或GND）。}
    \conrow{再生（回写）}{放大信号立即写回$C_s$，恢复原存储数据，解决破坏性读出。}
}

\noindent\textbf{传输特性}：横轴$\Delta V_{in}$，纵轴$\Delta V_{out}$。即使$\Delta V_{in}$极小，放大器亦能迅速拉伸至$\sim V_{DD}$，\textbf{\textcolor{myred}{增益极高、再生能力强}}。

\noindent\textbf{阵列差分工作（差分输入放大器）}：

\imgleft[0.4]{images/image-2026-06-08-20-55-48.png}{

\concepttable{
    \conrow{差分结构}{BL与$\overline{BL}$配对；读前均预充至$V_R = \frac{V_{S1} + V_{S0}}{2}$（公正比较基准）。}
    \conrow{信号差}{选通$WL_i$：$C_s$与BL电荷共享，BL变，$\overline{BL}$保持$V_R$。
    $\Delta V_1 = V_{B1} - V_R = \frac{C_s}{C_s + C_B}(V_{S1} - V_R),\quad
      \Delta V_0 = V_R - V_{B0} = \frac{C_s}{C_s + C_B}(V_R - V_{S0})$

    \conrow{判决与再生}{S/R感知$\Delta V_1$或$\Delta V_0$后，正反馈迅速拉伸至电源轨，完成逻辑判决并回写$V_{S1}$/$V_{S0}$。}
}

}

\textbf{\hlblue{设计要求与核心矛盾}}

\concepttable{

    \conrow{单元面积（密度优值）}{$A_{cell} = NF^2$（$F$为特征尺寸/最小布线间距之半，$N$为常数）；提升容量须不断缩小$N$与$F$。}

}

\textbf{\hlblue{DRAM 的刷新}}

\concepttable{
    \conrow{“动态”之源}{$C_s$无绝对绝缘，泄漏电流$I_{leak}$使存储电荷持续流失，需周期刷新维持数据。}
    \conrow{失效阈值}{高电平下降\textbf{\textcolor{myred}{20\%}}即判定信息丢失（对应$\Delta V = 0.2V_{S1}$）。}
    \conrow{刷新操作}{失效前由S/R将衰退电平重新放大并写回$C_s$，恢复标准逻辑电平。}
}

\noindent\textbf{\hlblue{泄漏电流来源（6类）}}

\concepttable{
    \conrow{亚阈值电流}{WL低电平时，表面势引起源漏间微弱载流子扩散。}
    \conrow{穿通电流}{短沟下源漏耗尽区展宽接触，势垒降低，栅控失效。}
    \conrow{pn结反向电流}{$n^+$节点与p衬底反偏，产生势垒区产生-复合电流及少子扩散电流。}
    \conrow{氧化层隧穿电流}{栅氧极薄时电子量子隧穿穿透绝缘层。}
    \conrow{热电子效应}{强场下载流子碰撞电离，少子被存储节点收集，改变存储电荷。}
    \conrow{场区寄生沟道泄漏}{场氧隔离不良，下方硅表面形成寄生反型层，导致相邻单元电荷串扰。}
}

\textbf{\hlblue{刷新效率 $\eta$（系统性能指标）}}

\noindent 刷新期间DRAM无法响应CPU读写请求（死区时间）。$\eta$越小则有效读写时间占比越大。

$$\eta = \frac{m \times t_{RC}}{t_h} \times 100\%$$

\concepttable{
    \conrow{$t_h$}{保持时间，电荷衰减达20\%阈值的物理时间极限，为考查时间窗口。}
    \conrow{$m$}{刷新周期数（阵列需刷新的总行数）。}
    \conrow{$t_{RC}$}{单行刷新周期耗时。}
    \conrow{$m \times t_{RC}$}{$t_h$窗口内所有刷新操作的总物理耗时。}
}

\noindent\textbf{实例}（256Kb DRAM）：$m=256$，$t_h=4\text{ms}$，$t_{RC}=400\text{ns}$。
分子：$256 \times 400 \times 10^{-9} = 102.4\,\mu\text{s}$；分母：$4 \times 10^{-3} = 4000\,\mu\text{s}$。

$$\eta = \frac{102.4}{4000} \times 100\% = 2.56\%$$

即4ms窗口内2.56\%时间被刷新锁死，97.44\%可供正常存取。

\imgleft[0.4]{images/image-2026-06-08-21-46-30.png}{

\concepttable{

    \conrow{核心参数}{$\displaystyle\beta_Q=\frac{K_p}{K_a}=\frac{K_3}{K_5}=\frac{K_4}{K_6}$；$K_p$为负载管（PMOS）导电因子，$K_a$为门管（NMOS）导电因子。}
    \conrow{物理本质}{$\beta_Q$越小写入越易：门管导电能力须强于负载管，方能将N1电位拉低越过反相器翻转阈值。故设计上需\textbf{\textcolor{myred}{减小PMOS尺寸、增大传输门NMOS尺寸}}。}

\textbf{\hlblue{写操作动态机制}}

写翻转的本质是\textbf{\textcolor{myred}{强行拉低原本为"1"的节点}}。因读稳定性约束，外部难以通过PG将"0"节点直接拉高，突破口在"1"节点N2。

\concepttable{
    \conrow{第1阶段}{初始 N1=0,N2=1。外部$\overline{BL}$低电平经PG2强行拉低$V_{N2}$。}
    \conrow{第2阶段}{$V_{N2}$低于反相器开关阈值时正反馈启动：PU1导通$\to$N1充电($0\!\to\!1$)；PU2截止$\to$N2彻底放电至0，写入完成。}
}

\noindent\textbf{\hlblue{等效分压与$\gamma$ ratio}}

写入瞬间N2节点经历PU2（拉高）与PG2（拉低）的电荷角力。

$$\gamma=\frac{(W/L)_{PU}}{(W/L)_{PG}}\;\text{等效于}\;\beta_Q=\frac{K_p}{K_a}$$

PG导电能力须战胜PU——$I_{\text{WRITE}}$主要流向GND而非维持N2电位。WL开启后$\overline{BL}$拉低，$V_{N2}$从$V_{DD}$开始下降，\textbf{\textcolor{myred}{PG强度必须克服PU强度方能率先拉低"1"}}。

\textbf{\hlblue{读稳定性与$\beta_R$（Cell Ratio）}}

读稳定性是写稳定性的对立面。读操作前BL与$\overline{BL}$预充至$V_{DD}$。以读"1"（N1=1, N2=0）为例：WL开启，M6导通，$\overline{BL}$高电平经M6向N2注入电荷，形成$\overline{BL}$-M6-N2-M2-GND通路，N2产生瞬态电压扰动$\Delta V$。

$$\beta_R = \frac{K_d}{K_a} = \frac{K_1}{K_5} = \frac{K_2}{K_6}$$

\concepttable{
    \conrow{$K_d$}{驱动管（Pull-Down NMOS M1/M2）导电因子。}
    \conrow{$K_a$}{门管（Pass-Gate NMOS M5/M6）导电因子。}
    \conrow{物理本质}{M2须远强于M6，将$\overline{BL}$涌入电荷迅速泄放至地，\textbf{\textcolor{myred}{钉住N2于低电位}}，防止$\Delta V$超反相器阈值致误翻转。}
    \conrow{设计原则}{$\beta_R \uparrow\ \Rightarrow$ 读稳定性$\uparrow$；驱动管NMOS尺寸须远大于门管NMOS。}
}

\subsubsection{传统 MOSFET 的亚阈值摆幅极限}

\entry{物理含义}{$S$（或称 $SS$）表示漏极电流改变一个数量级（10倍）所需的栅电压变化量。$S$ 越小，器件关断越快。}

\entry{公式}{$$S \equiv \frac{\text{d}V_{\text{GS}}}{\text{d}(\lg I_{\text{Dsub}})} = \ln(10)\frac{nkT}{q} = \ln(10)\left[1 + \frac{C_{\text{D}}(\phi_{\text{S}})}{C_{\text{ox}}}\right]\frac{kT}{q}$$}


\concepttable{

\concepttable{
    \conrow{传统器件}{亚阈值摆幅因子项为 $1 + \frac{C_{\text{s}}}{C_{\text{ox}}} \ge 1$，故 $SS \ge 60\text{ mV/dec}$。}
    \conrow{NCFET}{引入负电容 $C_{\text{FE}}$ 后，等效公式变为：$$1 + C_{\text{s}}\left(\frac{1}{C_{\text{ox}}} + \frac{1}{C_{\text{FE}}}\right) < 1$$}
    \conrow{核心结论}{因为 $C_{\text{FE}}$ 是负数，使得括号内相减，\textbf{\textcolor{myred}{整个系数小于 1}}，从而推导出 $SS < 60\text{ mV/dec}$。}
}

\subsubsection{电容匹配}

\entry{电容匹配}{负电容虽能带来超低 $SS$，但须满足严格的电容匹配条件。下面对比三种匹配状态（重点考核：量化条件、曲线形态、后果）。}

\imgleft[0.3]{images/image-2026-06-14-16-26-55.png}{

\concepttable{
    \conrow{电容失配（曲线 a）}{\textbf{条件：}$-C_{\text{FE}} < C_{\text{MOS}}$。局部表现出陡峭亚阈值摆幅，但\textbf{\textcolor{myred}{负电容效应不稳定。存在回滞现象}}，正反向扫描曲线不重合。}
    \conrow{电容完美匹配（曲线 b）$\star$}{\textbf{条件：}$C_{\text{MOS}} < -C_{\text{FE}} < C_{\text{ox}}$。负电容效应稳定。\textbf{\textcolor{myred}{$-C_{\text{FE}}$ 越接近 $C_{\text{MOS}}$，电容匹配程度越高。无回滞现象。}}}
    \conrow{电容匹配但增益低（曲线 c）}{\textbf{条件：}$-C_{\text{FE}} > C_{\text{ox}}$。负电容效应稳定，无回滞；但因 $-C_{\text{FE}}$ 绝对值过大，倒数对整体反馈贡献减小。性能相对传统 MOSFET 无太大提升。}
}
}

\subsubsection{铁电材料发展}

\textbf{\hlblue{传统钙钛矿材料}}

\concepttable{
    \conrow{PZT}{$\text{Pb(Zr,Tis)O}_3$（钛酸铅锆）}
    \conrow{BFO}{$\text{BiFeO}_3$（铁酸铋）}
    \conrow{SBT}{$\text{SrBi}_2\text{Ta}_2\text{O}_9$（钽酸铋锶）}
}

\concepttable{
    \conrow{优点}{能够实现较小的亚阈值摆幅（SS）；居里温度高，热稳定性高}
    \conrow{缺点}{高矫顽场导致回滞较大；开启电压（$V_{\text{on}}$）过大；\textbf{\textcolor{myred}{与 CMOS 工艺不兼容}}，无法直接用于大规模集成电路制造}
}

\textbf{实际器件数据}

\concepttable{
    \conrow{PZT}{$\text{SS}_{\text{min}} \sim 13\text{ mV/dec}$（SS 小），Hysteresis $\sim 12\text{V}$（证实回滞极大）}
    \conrow{BFO}{$\text{SS}_{\text{min}} = 8.5\text{ mV/dec}$，Hysteresis $\sim 4\text{V}$，$V_{\text{on}} = 4\text{V}$（证实开启电压过大，无法满足低功耗要求）}
}

\textbf{\hlblue{有机聚合物材料}}

由于钙钛矿材料回滞大且不兼容 CMOS，研究转向有机聚合物。典型材料：$\text{P(VDF-TrFE)}$（聚偏氟乙烯-三氟乙烯）。

\concepttable{
    \conrow{优点}{可以实现较小的亚阈值摆幅（SS）}
    \conrow{缺点}{在空气中器件性能会发生退化（\textbf{\textcolor{myred}{化学稳定性差}}）；依然存在工艺不兼容的问题}
}

\textbf{实际器件数据}

\concepttable{
    \conrow{串联铁电 NCFET}{原 MOSFET $\text{SS} = 92\text{ mV/dec} \rightarrow$ 串联 PVDF 后 $\text{SS}$ 降低至 $25\text{ mV/dec}$}
    \conrow{旋涂铁电 NCFET}{原 MOSFET $\text{SS} > 100\text{ mV/dec} \rightarrow$ 旋涂 PVDF 后 $\text{SS}$ 降低至 $18\text{ mV/dec}$}
}

\textbf{\hlblue{HfO2掺杂材料}}

为解决前两类材料共同的致命痛点——工艺不兼容，引入了当前主流的 $\text{HfO}_2$（二氧化铪）掺杂体系。常见掺杂元素：$\text{Zr}$、$\text{Al}$、$\text{Si}$ 等。

\concepttable{
    \conrow{优点}{具有极优的工艺兼容性（因为 $\text{HfO}_2$ 本身就是高 k 栅介质工艺的标准材料）；回滞小}
    \conrow{缺点}{亚阈值摆幅（SS）的\textbf{\textcolor{myred}{优化效果不够明显}}；工艺\textbf{\textcolor{myred}{涨落较大}}（一致性控制难）}
}

\textbf{工艺步骤：}

\concepttable{
    \conrow{Step 1}{在衬底（Substrate）上通过 ALD（原子层沉积）沉积 $\text{HfO}_2$ 并进行循环掺杂（cycles doping）}
    \conrow{Step 2}{通过 CVD（化学气相沉积）沉积金属电极（$\text{TiN}$ 或 $\text{TaN}$）}
    \conrow{Step 3}{形成最终的栅叠层结构}
}

\entry{前提}{$\text{HfO}_2$ \textbf{\textcolor{myred}{本身不具备铁电性}}，必须通过掺杂、引入应力以及退火（三者缺一不可）来实现。}

\entry{相变过程}{上述条件促使材料发生再结晶，晶格结构由不具铁电性的正方晶系（Tetragonal）转变为具备自发极化 $P$ 的斜方晶系（Orthorhombic），从而在 $\text{HfO}_2$ 中成功诱导出铁电性。}

\textbf{$\text{HfO}_2$ 掺杂的 SS 限制与 3D FinFET 架构}

\entry{结论}{虽然 $\text{HfO}_2$ 解决了兼容性，但其亚阈值摆幅优化不明显。\textbf{\textcolor{myred}{该器件最低 $\text{SS}_{\text{min}} = 55\text{ mV/dec}$}}}

\imgleft[0.3]{images/image-2026-06-14-17-12-44.png}{
\textbf{SS 随 $\bm{V_{\text{G}}}$ 变化曲线对比}

\concepttable{
    \conrow{黑色曲线（Gate 外栅极）}{未有效利用内部负电容效应时，基体 FinFET 最低 $\text{SS}_{\text{min}} = 87\text{ mV/dec}$。}
    \conrow{红色曲线（Internal gate 内栅极）}{引入负电容效应后，下探至 $\text{SS}_{\text{min}} = 55\text{ mV/dec}$。}
}
}

\subsection{隧穿场效应晶体管（TFET）}

\subsubsection{基本结构}

\entry{核心结构}{栅控反偏 P-I-N 结（源区-沟道区-漏区分别构成 P-I-N 结构，中间的 i 为本征或低掺杂沟道）。}

\img{images/image-2026-06-14-18-51-12.png}

\concepttable{
    \conrow{N型TFET}{源端为 P+型 掺杂，漏端为 N+型 掺杂。（工作条件：$V_{\text{GS}} > 0, V_{\text{DS}} > 0$）}
    \conrow{P型TFET}{源端为 N+型 掺杂，漏端为 P+型 掺杂。（工作条件：$V_{\text{GS}} < 0, V_{\text{DS}} < 0$）}
}

\entry{注}{TFET 与 MOSFET 在器件结构方面主要的区别是：\textbf{\textcolor{myred}{源漏区掺杂类型相反}}。传统 MOSFET 的源漏掺杂类型相同（如同为 N+），而 TFET 的源漏掺杂类型必然相反。}

\subsubsection{开关机制}

\entry{TFET 导通机制 —— 关态下的超低漏电}{
    当 $V_{\text{GS}} = 0\text{V}$ 时，器件处于关态。源端与沟道之间的势垒极宽，电子无法发生隧穿。$I_{\text{OFF}}$ 本质是反偏 P-I-N 结的\textbf{\textcolor{myred}{反向漏电流}}。

    \concepttable{
        \conrow{示例数据}{硅基 TFET 关态电流约为 $10^{-14}\text{ A/}\mu\text{m}$。}
        \conrow{结论}{\textbf{\textcolor{myred}{TFET 的关态漏电流远小于传统 MOSFET}}，对应控制关态泄漏电流的目标。}
    }
}

\imgleft[0.3]{images/image-2026-06-14-19-11-14.png}{
\entry{导通机制 —— 开态下的带带隧穿（BTBT）}{
    开态物理过程：

    \concepttable{
        \conrow{Step 1}{栅压 $V_{\text{GS}}$ 增大，电场使沟道表面能带向下移动。}
        \conrow{Step 2}{源区和沟道表面之间形成较小的势垒宽度。}
        \conrow{Step 3}{当 $E_{\text{C}}$（沟道导带底）低于 $E_{\text{V}}$（源端价带顶）时，开启能带窗口。}
        \conrow{Step 4}{源端价带电子通过带带隧穿（BTBT）进入沟道导带，形成隧穿电流。}
    }
}
}

\subsubsection{SS 优化原理}

\entry{原理}{传统 MOSFET 通过栅压控制势垒\underline{高度}；TFET 通过栅压控制隧穿势垒\underline{宽度}。}

\entry{WKB 近似与 SS 极限突破}{

    \concepttable{
        \conrow{隧穿概率与电流（WKB 近似）}{$$J_{\text{d}} \propto T_{\text{WKB}} \approx \exp\left( -\frac{4\sqrt{2m_n^*}E_{\text{g}}^{3/2}}{3q\hbar\xi} \right)$$}
        \conrow{决定电流的关键物理量}{$m_n^*$（电子有效质量）、$E_{\text{g}}$（半导体禁带宽度）、$\xi$（隧穿结电场）}
    }

    \textbf{\textcolor{myred}{结论}}：亚阈值斜率 SS 表达式：
    $$SS = \ln 10 \cdot \left( \frac{1}{V_{\text{eff}}} \cdot \frac{\partial V_{\text{eff}}}{\partial V_{\text{GS}}} + \frac{\xi + b}{\xi^2} \cdot \frac{\partial \xi}{\partial V_{\text{GS}}} \right)^{-1}$$
    
    从公式可见，\textbf{\textcolor{myred}{SS 与温度项 $kT/q$ 无关}}。因此 TFET 理论上可突破 $60\text{ mV/dec}$ 极限，实现超陡亚阈值摆幅，从而允许 $V_{\text{DD}}$ 进一步降低。
}

\subsubsection{TFET 缺陷}

\entry{概述}{TFET 虽具有超低关态漏电并能突破 $60\text{ mV/dec}$ 极限，但以下两大核心问题制约其量产。}

\concepttable{
    \conrow{导通电流（$I_{\text{ON}}$）低}{对于硅基 TFET，由于单质硅（Si）的禁带宽度（$E_{\text{g}}$）较大，导致载流子的带带隧穿几率较低。硅基 TFET 的开启电流低于硅基 MOSFET 2\textasciitilde{}3 个数量级，\textbf{\textcolor{myred}{器件驱动能力严重受限}}。}
    \conrow{工艺难度高}{TFET \textbf{\textcolor{myred}{对源区和漏区的掺杂浓度梯度非常敏感}}。在实际热工艺中，源漏区的重掺杂会向沟道区扩散，从而破坏了陡峭的 PN 结特性，最终影响隧穿处的势垒宽度。使掺杂的随机波动性会导致 TFET 的开关态电流出现 2 个数量级的剧烈波动。}
}

\textbf{\hlblue{性能改进：新材料设计}}

\entry{准则}{选择\textbf{\textcolor{myred}{小禁带宽度（small $E_{\text{g}}$）}}和\textbf{\textcolor{myred}{小有效质量（small $m^*$）}}的材料。}

\entry{物理依据}{根据 WKB 公式，减小 $m^*$ 和 $E_{\text{g}}$ 使分式值减小，隧穿电流呈指数级放大。}

\textbf{材料参数对比}

\compTable[L]{4}{
    \compheader{\textbf{材料} & \textbf{$E_{\text{g}}$ (eV)} & \textbf{$m^*$ ($m_0$)} & \textbf{特点}}
    \comprow{$\text{Si}$ (传统) & 1.12 & 0.19 & 隧穿几率低，$I_{\text{ON}}$ 受限}
    \comprow{$\text{Ge}$ & 0.66 & 0.082 & $E_{\text{g}}$ 和 $m^*$ 均显著降低}
    \comprow{$\text{InAs}$ & 0.36 & 0.026 & III-V 族，禁带宽度骤降}
    \comprow{$\text{InSb}$ & 0.14 & 0.0135 & $m^*$ 减小一个数量级以上}
}

\textbf{P 型/N 型 TFET 的最佳材料体系：}

\concepttable{
    \conrow{p-TFETs 改进材料}{选用 $\text{Ge/GeSn}$（锗/锗锡）体系。提供小的 $E_{\text{g}}$ 和 $m^*$，等效减小隧穿势垒宽度。}
    \conrow{n-TFETs 改进材料}{选用 III-V 族化合物材料（典型代表：$\text{InGaAs}$、$\text{InAs}$、$\text{GaSb}$）。提供小的 $E_{\text{g}}$ 和 $m^*$，实现高效直接带带隧穿（direct BTBT）。}
}

\textbf{\hlblue{性能改进：新结构设计（提高隧穿面积）}}

\entry{概述}{除更换材料外，改变器件的几何结构也能提升电流}

\img{images/image-2026-06-14-19-43-44.png}

\concepttable{
    \conrow{\textbf{\textcolor{myred}{“点”隧穿}}}{电子隧穿方向（横向）与栅极电场方向（纵向）\textbf{\textcolor{myred}{相互垂直}}。隧穿仅发生在源区与沟道交界面的局部区域，面积很小，因而限制了 $I_{\text{ON}}$。}
    \conrow{\textbf{\textcolor{myred}{“线”隧穿}}}{通过结构设计，使栅极直接覆盖在部分源区之上。电子隧穿方向（纵向）与栅极电场方向\textbf{\textcolor{myred}{平行}}。}
}

\entry{结论}{“线”隧穿通过大幅提高隧穿面积，在相同电压下释放更多隧穿载流子，\textbf{\textcolor{myred}{有效解决硅基 TFET 导通电流低的问题}}。}

\textbf{\hlblue{性能改进——新结构设计（增强栅控能力）}}

\entry{概述}{结构改进的第二条路径：利用先进的多栅/环绕栅架构强化对隧穿结电场的控制。}

\concepttable{
    \conrow{结构}{双栅（DG，Double Gate）、纳米线（NWs，GAA 全环绕栅）、包裹栅（Wrapped gate）}
}
