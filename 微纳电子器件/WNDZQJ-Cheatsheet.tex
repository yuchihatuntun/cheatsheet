\subsubsection{平面 MOSFET 的特征尺寸}

\textbf{\hlblue{器件层面}}

\entry{特征尺寸}{电学引出的栅电极中心距的一半：$\frac{1}{2}CPP$。}

\entry{几何限制}{$$CPP > l_g + l_{sp} + l_{cnt}$$$l_g$：栅长；$l_{sp}$：侧墙宽度；$l_{cnt}$：源漏接触孔宽度。}

\textbf{\hlblue{工艺与互连层面}}

\entry{特征尺寸}{第一层金属半节距：$\frac{1}{2}\times$相邻导线中心距。传统 ITRS 以此命名技术节点（如 90nm、65nm）。}
\entry{via0}{第0层通孔（接触层与 metal1 间互连），集成度特征尺寸还受层间对准和通孔工艺制约。}

\subsubsection{MOSFET 工作原理}

\concepttable{
    \conrow{$V_{GS} < V_T$（截止）}{源漏之间被衬底隔离，无导电沟道，$I_D = 0$。}
    \conrow{$V_{GS} > V_T$（导通）}{沟道表面形成强反型，形成沟道，$V_{DS}$ 驱动电子产生 $I_D$。}
}

\imgleft[0.3]{images/image-2026-06-30-18-58-54.png}{
    \entry{阈值电压 $V_T$}{使栅下硅表面开始强反型时的栅电压。\textbf{\textcolor{myred}{表面少子浓度 $\ge$ 体内平衡多子浓度时}}，称为表面发生了强反型。}

    \entry{平带电压 $V_{FB}$}{使半导体体内\textbf{\textcolor{myred}{无能带弯曲}}的栅压（表面势 $\phi_S = 0$）。}

    \entry{控制规律}{恒定 $V_{DS}$ 下，$V_{GS} \uparrow \rightarrow$ 沟道电子 $\uparrow \rightarrow$ 沟道电阻 $\downarrow \rightarrow I_D \uparrow$。}

    \concepttable{
        \conrow{N 沟道增强型}{$V_T>0$，常关，$V_{GS}>V_T$ 开启}
        \conrow{N 沟道耗尽型}{$V_T<0$，常开，$V_{GS}<V_T$ 截止}
        \conrow{P 沟道增强型}{$V_T<0$，常关，$V_{GS}<V_T$ 开启}
        \conrow{P 沟道耗尽型}{$V_T>0$，常开，$V_{GS}>V_T$ 截止}
    }
}

\subsubsection{MOS结构阈值电压}

以下推导以 P 衬底 MOSFET 为例（公式记忆也以 NMOS 为主）。

\imgleft[0.3]{images/image-2026-06-21-16-09-56.png}{
\textbf{\hlblue{理想 MOS 能带结构}}

\concepttable{
    \conrow{理想条件}{$\phi_{MS}=0$，$Q_{ox}=0$}
    \conrow{$V_G=0$ 时}{全系统能带水平无弯曲（默认为平带状态）}
}

\entry{$\phi_{FP}$（P型衬底费米势）}{\textbf{\textcolor{myred}{$\phi_{FP} = \frac{1}{q}(E_i - E_F) = \frac{kT}{q}\ln\frac{N_A}{n_i}$}}

\textbf{\textcolor{myred}{}}意义：$E_F$ 偏离 $E_i$ 的电势差。}
}

\imgleft[0.3]{images/image-2026-06-21-16-12-04.png}{
\textbf{\hlblue{实际 MOS 能带结构（$V_G=0$）}}

\concepttable{
    \conrow{实际参数}{$\phi_{MS}<0$，$Q_{ox}>0$（材料差异/工艺缺陷所致）}
    \conrow{能带特征}{半导体表面能带向下弯曲，$\phi_s>0$}
}

\entry{表面势 $\phi_s$}{硅表面到体内平衡处的电势差，\textbf{\textcolor{myred}{$q\phi_s={}$能带弯曲量}}。能带向下弯曲 $\rightarrow \phi_s>0$（表面电势高于体内）。}

\entry{平带电压}{实际 MOS 在 $V_G=0$ 时能带已弯曲，需外加电压恢复平带，由此定义平带电压。}
}

\concepttable{
    \conrow{$V_{FB}$ 定义式}{$V_{FB} = \phi_{MS} - \frac{Q_{ox}}{C_{ox}}$}
}

\textbf{\hlblue{\textbf{\textcolor{myred}{MOS 结构}}阈值电压公式}}

\textbf{对于 NMOS：}

$$\boxed{V_T = V_{FB} + V_{ox} + 2\phi_{FP} = \phi_{MS} - \frac{Q_{ox}}{C_{ox}} - \frac{Q_S}{C_{ox}} + 2\phi_{FP}}$$

MOS 结构的阈值电压为

$$V_{T} = \phi_{MS} - \frac{Q_{OX}}{C_{OX}} + K(2\phi_{FP})^{\frac{1}{2}} + 2\phi_{FP} = \phi_{MS} - \frac{Q_{OX}}{C_{OX}} + K(\phi_{S})^{\frac{1}{2}} + 2\phi_{FP}$$

式中，$ K = \frac{(2q N_A \varepsilon_s)^{\frac{1}{2}}}{C_{OX}} $，称为\textbf{\textcolor{myred}{体因子}}，仅由工艺参数（衬底掺杂浓度 $N_A$、氧化层电容 $C_{ox}$）决定，与外加电势无关。

\subsubsection{MOSFET 阈值电压}

\textbf{MOSFET（四端）与 MOS 结构（三端）的边界条件有区别：}

\imgleft[0.3]{images/image-2026-06-22-12-29-32.png}{
\entry{栅极有效驱动电压修正}{栅-衬电压由 $V_G$ 变为 $V_G-V_B$。}

\entry{\textbf{\textcolor{myred}{强反型临界 $\phi_s$ 修正}}}{源/漏及反型层与衬底间 PN 结承受反向电压 $V_S-V_B$，使强反型更难。$\phi_s$ 阈值由 $2\phi_{FP}$ 增大为：
$$\phi_s = 2\phi_{FP} + V_S - V_B$$}
}

将 $\phi_s = 2\phi_{FP} + V_S - V_B$ 代入传统氧化层压降公式中，得到：

$$-\frac{Q_S}{C_{ox}} = \frac{(2q N_A \varepsilon_s)^{\frac{1}{2}}}{C_{ox}} (2\phi_{FP} + V_S - V_B)^{\frac{1}{2}} = K(2\phi_{FP} + V_S - V_B)^{\frac{1}{2}}$$

\subsubsection{阈值电压影响因素}

\textbf{\hlblue{栅氧化层厚度 $T_{ox}$}}

\textbf{工艺参数化统一公式：}

$$V_T = \phi_{MS} - \frac{T_{ox}}{\varepsilon_{ox}}Q_{ox} - \frac{T_{ox}}{\varepsilon_{ox}}Q_{AD} + 2\phi_{FB}$$

从公式可见，\textbf{\textcolor{myred}{$T_{ox}$ 越厚，电荷引起的阈值电压漂移量就越大}}。

\textbf{$T_{ox}$ 的工艺演进数值：}

\concepttable{
    \conrow{早期 MOSFET}{约 150 nm}
    \conrow{目前高性能 MOSFET}{可达 10 nm 以下}
}

\textbf{\hlblue{衬底费米势 $\phi_{FB}$}}

\entry{NP 沟道电学符号区分}{
    $$\phi_{FB} = \begin{cases} \phi_{FP} = \frac{kT}{q}\ln\frac{N_A}{n_i} > 0 & \text{(N 沟道)} \\ \phi_{FN} = -\frac{kT}{q}\ln\frac{N_D}{n_i} < 0 & \text{(P 沟道)} \end{cases}$$
}

$\phi_{FB}$ 与衬底掺杂浓度（$N_A$ 或 $N_D$）呈对数（$\ln$）关系，因此\textbf{\textcolor{myred}{掺杂浓度对其影响不大}}。

\textbf{\hlblue{功函数差 $\phi_{MS}$}}

\entry{物理决定因素}{功函数差 $\phi_{MS}$ 仅与\textbf{\textcolor{myred}{金属种类、半导体种类以及掺杂浓度}}相关。}

\textbf{Al-Si 材料},当 $N = 10^{15}\,\text{cm}^{-3}$ 时，

$$\phi_{\text{MS}} = \begin{cases} -0.9\,\text{V} & \text{(N 沟)} \\ -0.3\,\text{V} & \text{(P 沟)} \end{cases}$$

\textbf{\hlblue{耗尽区空间电荷 $Q_{AD}$}}

\entry{N 沟道（NMOS）}{$$Q_A = -q N_A x_d = -(4q N_A \varepsilon_s \phi_{FP})^{\frac{1}{2}} < 0$$}

$$Q_{AD} \propto N_{AD}^{\frac{1}{2}}$$

\textbf{\hlblue{氧化层电荷 $Q_{ox}$}}

$Q_{ox}$ 与\textbf{\textcolor{myred}{制造工艺及晶向}}密切相关。为了减小界面态和氧化层缺陷电荷，MOSFET 一般采用 (100) 晶面。

\textbf{\hlblue{调控阈值电压主要手段}}

\textbf{\textcolor{myred}{1. 改变衬底或沟道掺杂浓度 $N$}}
\ \ \ \ \ 
\textbf{\textcolor{myred}{2. 改变栅氧化层厚度 $T_{ox}$}}

\textbf{\hlblue{衬底偏置效应（体效应）}}

\entry{体效应定义}{阈值电压 $V_T$ 随衬源电压 $V_{BS}$ 的变化而变化的现象，即前文所述 MOSFET 的阈值电压公式。}

\textbf{体效应漂移方向}

当 $|V_{BS}|$ 增大时：N阈值电压向正方向变化（$V_T$ 变大，器件更难开启）。

\entry{体因子 $K$}{$$K = \frac{(2q\varepsilon_s N)^{\frac{1}{2}}}{C_{ox}}$$ 反映体效应的强弱。}

\concepttable{
    \conrow{氧化层厚度 $T_{ox}$}{$T_{ox}$ 越厚 $\rightarrow$ $C_{ox}$ 越小 $\rightarrow$ $K$ 越大。}
    \conrow{衬底掺杂浓度 $N$}{$N$ 越高 $\rightarrow$ $K$ 越大。}
}

\entry{结论}{\textbf{\textcolor{myred}{$T_{ox}$ 越厚、掺杂浓度 $N$ 越高，衬底偏置效应（体效应）就越严重}}，阈值电压受衬底电压的影响漂移得越剧烈。}

\textbf{\hlblue{离子注入}}（非重点，了解即可）

\entry{基本概念}{通过离子注入改变沟道掺杂浓度，从而调整阈值电压 $V_T$。工艺上常将杂质分布简化为\textbf{\textcolor{myred}{阶梯形分布}}。}

\entry{关键参数}{
    $N_I$：新增杂质浓度，$N'_A = N_A + N_I$；\\
    $Q_I = -q N_I R$：注入区新增电离杂质电荷面密度。
}

\entry{调整效果}{注入同类型杂质（$N_I>0$）$\rightarrow$ $V_T$ 增大；注入反类型杂质（$N_I<0$）$\rightarrow$ $V_T$ 减小。}

    \concepttable{
        \conrow{饱和漏压}{$V_{SD(sat)} = V_{SG} - |V_{TP}|$}
        \conrow{线性区}{$\frac{W \mu_p C_{ox}}{2L} [2(V_{SG}-|V_{T}|)V_{SD}]$}
        \conrow{饱和区}{$I_D = \frac{W \mu_p C_{ox}}{2L} (V_{SG} - |V_{TP}|)^2$}
        \conrow{线性跨导}{$g_m = \frac{W \mu_p C_{ox}}{2L} V_{SD}$}
        \conrow{饱和跨导}{$g_m = \frac{W \mu_p C_{ox}}{L} (V_{SG} - |V_{TP}|)$}
    }
\end{minipage}

\textbf{亚阈值区：}

$$I_{Dsub} = \frac{Z}{L}\mu_p C_D(\phi_S)\left(\frac{kT}{q}\right)^2
\exp\left[\frac{q}{kT}\left(\frac{V_{GS}-V_T}{n}\right)\right]
\cdot\left[1-\exp\left(-\frac{qV_{DS}}{kT}\right)\right]$$

\textbf{\hlblue{MOSFET 的亚阈区特性}}

\entry{亚阈区摆幅 $S$ 的定义}{亚阈区转移特性斜率的倒数：

$$S \equiv \frac{\text{d}V_{GS}}{\text{d}(\lg I_{\text{Dsub}})} = \ln(10)\frac{nkT}{q} = \ln(10)\left[1 + \frac{C_D(\phi_s)}{C_{ox}}\right]\frac{kT}{q}$$

其中 $C_D(\phi_s)$：耗尽层电容；$C_{ox}$：栅氧化层电容；$n = 1 + \frac{C_D}{C_{ox}}$：理想因子。}

\entry{物理意义}{\textbf{\textcolor{myred}{使 $I_{\text{Dsub}}$ 扩大 10 倍（一个数量级）所需的 $V_{GS}$ 增量}}。}

\entry{开关性能要求与工艺减小措施}{要求 $S$ 尽量小（切换快、关断漏电小）：
\begin{itemize}[nosep,leftmargin=1em]
    \item 降低 $C_D(\phi_s)$：降低衬底掺杂浓度 $N_A$（$N_A \downarrow \Rightarrow C_D \downarrow$）
    \item 增大 $C_{ox}$：减小栅氧化层厚度 $T_{ox}$（$T_{ox} \downarrow \Rightarrow C_{ox} \uparrow$）
\end{itemize}}

\subsubsection{MOSFET 阈值电压测量}

\entry{\textbf{\hlblue{方法 1 —— 联立方程法}}}{

\textbf{1. 饱和区电流基本方程}

方法 1 建立在晶体管饱和区漏极电流方程的基础之上：
$$I_{\text{Dsat}} = \frac{\beta}{2}(V_{GS} - V_T)^2$$
式中 $\beta$ 为导电系数。

\textbf{2. 测量与求解过程}（核心应试理解）

测试动作：在饱和区内，人为改变栅压，分别测量两组不同的数据对：$(V_{GS1}, I_{\text{Dsat1}})$ 和 $(V_{GS2}, I_{\text{Dsat2}})$。联立方程组：
$$\begin{cases} I_{\text{Dsat1}} = \frac{\beta}{2}(V_{GS1} - V_T)^2 \\ I_{\text{Dsat2}} = \frac{\beta}{2}(V_{GS2} - V_T)^2 \end{cases}$$
求解结果：将测得的电压和电流作为已知数代入上式，通过代数消元，即可唯一解出工艺参数 $\beta$ 和阈值电压 $V_T$。}

\entry{\textbf{\hlblue{方法 2 —— $1\ \mu\text{A}$ 法}}}{

\imgleft[0.3]{images/image-2026-06-22-16-34-44.png}{

\entry{定义}{将漏极电流达到某一规定值 $I_{DT}$ 时的栅源电压 $V_{GS}$ 直接定义为阈值电压 $V_T$。}

\entry{标准取值}{在实际工程和早期工艺中，通常将该规定电流 $I_{DT}$ 定为 $1\ \mu\text{A}$。}

\textbf{2. 优缺点对比}

\begin{itemize}[nosep,leftmargin=1em]
    \item \textbf{优点}：此方法不需要进行复杂的方程求解或作图拟合，简单易行，早期较多采用。
    \item \textbf{缺点}（核心漏洞）：误差较大。
\end{itemize}
}

\textbf{原因分析：}直线的斜率满足：$\text{斜率} = \sqrt{\dfrac{\beta}{2}} = \sqrt{\dfrac{Z\mu_n C_{ox}}{2L}}$（式中 $Z$ 为沟道宽度，$L$ 为沟道长度，$\frac{Z}{L}$ 为宽长比）。

从图可见，\textbf{\textcolor{myred}{在电流极小的弱反型区到强反型区过渡段，电流偏离了完美的直线}}。对于两个除宽长比 $\frac{Z}{L}$ 不同、其余特性完全相同的 MOSFET，由于其饱和区电流基数不同，当强行指定 $I_{DT} = 1\ \mu\text{A}$ 时，在横轴上读出的 $V_T$ 值会产生人为的偏差，无法真实反映相同的物理阈值。}

\subsubsection{MOSFET 小信号参数}

\textbf{\hlblue{跨导 $g_m$}}

\entry{数学定义}{$$g_m \equiv \left. \frac{\partial I_D}{\partial V_{GS}} \right|_{V_{DS}}$$}

\entry{\textbf{\textbf{\textcolor{myred}{非饱和区跨导}}}}{
原始电流方程：$I_D = \beta \left[ (V_{GS} - V_T)V_{DS} - \frac{1}{2}V_{DS}^2 \right]$

对 $V_{GS}$ 求偏导：$$\boxed{g_m = \beta V_{DS}}$$
}

\entry{\textbf{\textbf{\textcolor{myred}{饱和区跨导}}}}{
原始电流方程：$I_{\text{Dsat}} = \frac{1}{2}\beta(V_{GS} - V_T)^2$

对 $V_{GS}$ 求偏导：$$\boxed{g_{ms} = \beta(V_{GS} - V_T) = \beta V_{\text{Dsat}}}$$
}

\entry{\textbf{提高饱和区跨导 $g_{ms}$ 的途径}}{

\begin{itemize}[nosep,leftmargin=1em]
    \item \textbf{器件工艺角度}（提高 $\beta$）：增大宽长比 $\frac{Z}{L}$；提高迁移率 $\mu$（改进沟道材料）；减小 $T_{ox}$（增大 $C_{ox}$）。
    \item \textbf{电路设计角度}：提高 $V_{GS}$（增大过驱动电压）。
\end{itemize}

}

\textbf{\hlblue{频率特性}}

\textbf{截止频率} $f_T$：器件电流增益为1时的频率

$$f_T = f\left| \frac{I_d}{I_i} \right|_{=1} = \frac{g_m}{2\pi(C_{gs_T} + C_M)} = \frac{g_m}{2\pi C_G}$$

\textbf{饱和区截止频率：}

$$\boxed{f_T = \frac{\mu_n(V_{GS} - V_T)}{2\pi L^2}}$$

\textbf{\hlyellow{作业习题整理}}

\prob{1}{}{某铝栅N沟道MOSFET的衬底掺杂浓度为$N_A=1\times10^{15}\,\text{cm}^{-3}$，栅SiO$_2$氧化层厚度为$120\,\text{nm}$，栅氧化层中有效电荷面密度为$3\times10^{11}\,\text{cm}^{-2}$，试确定该器件为耗尽型还是增强型，并计算$V_T$}

\textbf{解题思路：} 根据 MOSFET 阈值电压公式，依次计算功函数差 $\phi_{MS}$、氧化层固定电荷项 $Q_{OX}/C_{OX}$、体效应系数 $K$ 及衬底费米势 $\phi_{FP}$，代入求得 $V_T$ 并依据其正负判断器件类型。

\vspace{0.5em}
\textbf{解题过程：}

阈值电压公式为：
$$V_{T} = \phi_{MS} - \frac{Q_{OX}}{C_{OX}} + K\sqrt{2\phi_{FP}} + 2\phi_{FP}, \quad K = \frac{\sqrt{2q N_A \varepsilon_s}}{C_{OX}}$$

\begin{enumerate}
    \item \textbf{功函数差 $\phi_{MS}$}
    
    Al-Si 材料，当 $N_A = 10^{15}\,\text{cm}^{-3}$（N 沟）时：
    $$\phi_{MS} = -0.9\,\text{V}$$

    \item \textbf{氧化层参数}
    
    $$Q_{ox} = 3 \times 10^{15} \times 1.6 \times 10^{-19} = 4.8 \times 10^{-4}\,C/m^2$$
    $$C_{ox} = \frac{\varepsilon_{ox}}{t_{ox}} = \frac{3.9 \times 8.85 \times 10^{-12}}{120 \times 10^{-9}} = 2.88 \times 10^{-4}\,F/m^2$$

    \item \textbf{衬底费米势 $\phi_{FP}$}
    
    $$\phi_{FP} = \frac{kT}{q} \ln \frac{N_A}{n_i} = 0.026 \times \ln \frac{10^{21}}{1.5 \times 10^{16}} \approx 0.288\,\text{V}$$

    \item \textbf{体效应系数 $K$}
    
    $$K = \frac{\sqrt{2 \times 1.6 \times 10^{-19} \times 10^{21} \times 11.5 \times 8.85 \times 10^{-12}}}{2.88 \times 10^{-4}} \approx 0.828\,V^{1/2}$$

    \item \textbf{计算 $V_T$ 并判断类型}
    
    $$V_T = -0.9 - \frac{4.8 \times 10^{-4}}{2.88 \times 10^{-4}} + 0.828 \times \sqrt{2 \times 0.288} + 2 \times 0.288$$
    $$V_T = -0.9 - 1.67 + 0.476 + 0.576 \approx -1.518\,V$$
\end{enumerate}

由于 $V_T < 0$，该器件为\textbf{耗尽型}。

\prob{2}{}{某N沟道MOSFET的$V_T=1V$，$\mu_n=500cm^2V^{-1}s^{-1}$，$SiO_2$氧化层厚度为$80nm$，$W/L=100$，求当$V_{GS}=6V$时，$V_{DS}$分别为2、4、6、8、10V时的漏极电流。}

\textbf{解题思路}

首先计算氧化层电容 $C_{ox}$ 与增强因子 $\beta$，进而得到过驱动电压 $V_{GS}-V_T$。根据 $V_{DS}$ 与 $V_{GS}-V_T$ 的大小关系判断工作区：当 $V_{DS} \le V_{GS}-V_T$ 时器件处于线性区，使用线性区电流公式计算 $I_D$；当 $V_{DS} > V_{GS}-V_T$ 时器件进入饱和区，漏极电流达到饱和值 $I_{Dsat}$，与 $V_{DS}$ 无关。

\textbf{解题过程}

\begin{enumerate}
    \item \textbf{计算 $C_{ox}$ 与 $\beta$}
    
    $$ C_{ox} = \frac{\varepsilon_{ox}}{t_{ox}} = \frac{3.9 \times 8.85 \times 10^{-12}}{80 \times 10^{-9}} = 4.31 \times 10^{-4}\,F/m^2 $$
    $$ \beta = \frac{W}{L} \mu C_{ox} = 100 \times 500 \times 10^{-4} \times 4.31 \times 10^{-4} = 2.16 \times 10^{-3} $$
    
    \item \textbf{过驱动电压}
    
    $$ V_{GS} - V_T = 6\,\text{V} - 1\,\text{V} = 5\,\text{V} $$
    
    \item \textbf{线性区电流 ($V_{DS} = 2\,\text{V}, 4\,\text{V}$)}
    
    线性区公式：
    $$ I_D = \beta \left[ (V_{GS} - V_T)V_{DS} - \frac{1}{2}V_{DS}^2 \right] $$
    
    当 $V_{DS} = 2\,\text{V}$：
    $$ I_D = 2.16 \times 10^{-3} \times [5 \times 2 - 0.5 \times 4] = 17.28\,\text{mA} $$
    
    当 $V_{DS} = 4\,\text{V}$：
    $$ I_D = 2.16 \times 10^{-3} \times [5 \times 4 - 0.5 \times 16] = 25.92\,\text{mA} $$
    
    \item \textbf{饱和区电流 ($V_{DS} = 6\,\text{V}, 8\,\text{V}, 10\,\text{V}$)}
    
    饱和区公式：
    $$ I_{Dsat} = \frac{1}{2}\beta (V_{GS} - V_T)^2 $$
    
    当 $V_{DS} > 5\,\text{V}$ 时，器件进入饱和区，电流恒定为：
    $$ I_D = 0.5 \times 2.16 \times 10^{-3} \times 25 = 27\,\text{mA} $$
\end{enumerate}

\prob{3}{}{某N沟道MOSFET 的$N_A=1\times 10^{15} cm^{-3}$，栅$SiO_2$氧化层厚度为$150nm$，$\mu_n=500cm^2V^{-1}s^{-1}$，$L=4\mu m$，$W=100\mu m$，$V_T=0.5V$，计算$V_{GS}=4V$时的跨导$g_m$和截止频率$f_T$}

\textbf{解题思路}：基于饱和区模型，先计算工艺跨导参数 $\beta$，再代入跨导公式 $g_{ms} = \beta(V_{GS} - V_T)$ 求得 $g_m$；截止频率 $f_T$ 由饱和区特征频率公式 $f_T = \mu_n(V_{GS} - V_T) / (2\pi L^2)$ 直接计算。

\begin{enumerate}
    \item \textbf{计算工艺跨导参数 $\beta$}
    \[
        \beta = \frac{W}{L} \mu_n C_{ox}
             = 25 \times 500 \times 10^{-4}
               \times \frac{3.9 \times 8.85 \times 10^{-12}}{150 \times 10^{-9}}
             = 2.87 \times 10^{-4} \; \text{A/V}^2
    \]

    \item \textbf{计算饱和区跨导 $g_m$}
    \[
        g_{ms} = \beta (V_{GS} - V_T)
              = 2.87 \times 10^{-4} \times 3.5
              \approx 1.0 \times 10^{-3} \; \text{S}
    \]

    \item \textbf{计算截止频率 $f_T$}
    \[
        f_T = \frac{\mu_n (V_{GS} - V_T)}{2\pi L^2}
            = \frac{500 \times 10^{-4} \times 3.5}
                   {2\pi \times (4 \times 10^{-6})^2}
            = 1.74 \times 10^9 \; \text{Hz}
    \]
\end{enumerate}

\prob{4}{}{试计算题3中器件在工作电压为$5V$时，分别按CV和CE规则缩小$0.7$倍时($1/\alpha=0.7$)，对应的$V_T$、$I_D$、$t_d$、动态功耗的值}

\textbf{解题思路：}根据恒定电场（CE）与恒定电压（CV）两种缩放规则，分别对阈值电压 $V_T$、漏极电流 $I_D$、延迟时间 $t_d$ 及动态功耗 $P_d$ 进行缩放计算。CE 规则下，电压与尺寸同比例缩小 $1/\alpha$，电流与功耗按 $1/\alpha$ 缩放；CV 规则下，电压保持不变，电流与功耗按 $\alpha$ 放大，延迟按 $1/\alpha^2$ 缩小。

\vspace{0.5em}
\textbf{CV 缩放规则（$\alpha = 1.43$）}

\begin{align*}
    V_T &= 0.5 \, \text{V} \\
    I_D &= 1.76 \, \text{mA} \times 1.43 = 2.52 \, \text{mA} \\
    t_d &= 2.61 \times 10^{-10} \, \text{s} \times 0.7 \times 0.7
         = 1.28 \times 10^{-10} \, \text{s} \\
    P_d &= 8.81 \times 10^{-3} \, \text{W} \times 1.43
         = 1.26 \times 10^{-2} \, \text{W}
\end{align*}

\vspace{0.5em}
\textbf{CE 缩放规则（$\alpha = 1.43$）}

\begin{align*}
    V_T &= 0.5 \, \text{V} \times 0.7 = 0.35 \, \text{V} \\
    I_D &= 1.76 \, \text{mA} \times 0.7 = 1.23 \, \text{mA} \\
    t_d &= 2.61 \times 10^{-10} \, \text{s} \times 0.7
         = 1.83 \times 10^{-10} \, \text{s} \\
    P_d &= 8.81 \times 10^{-3} \, \text{W} \times 0.7 \times 0.7
         = 4.32 \times 10^{-3} \, \text{W}
\end{align*}

\prob{5}{}{已知某技术代的等效$EOT$为$1.7nm$，试计算分别采用$HfO_2$和$Al_2O_3$作为栅介质材料的厚度}

\textbf{解题思路：}利用等效氧化层厚度（EOT）定义式，将目标介质的物理厚度表示为 EOT 与相对介电常数之比的函数，代入已知参数求解。

已知 $SiO_2$ 相对介电常数 $\varepsilon_{SiO_2}=3.9$，$HfO_2$ 相对介电常数 $\varepsilon_{HfO_2}=25$，$Al_2O_3$ 相对介电常数 $\varepsilon_{Al_2O_3}=9$，目标 $EOT=1.7\,\text{nm}$。

由 EOT 定义：
\[
EOT = \frac{\varepsilon_{SiO_2}}{\varepsilon_{ox}} t_{ox}
\quad\Longrightarrow\quad
t_{ox} = \frac{\varepsilon_{ox}}{\varepsilon_{SiO_2}} EOT
\]

分别计算两种介质厚度：
\begin{align*}
t_{HfO_2} &= \frac{\varepsilon_{HfO_2}}{\varepsilon_{SiO_2}} EOT
          = \frac{25 \times 1.7}{3.9}
          \approx 10.9\,\text{nm} \\[4pt]
t_{Al_2O_3} &= \frac{\varepsilon_{Al_2O_3}}{\varepsilon_{SiO_2}} EOT
            = \frac{9 \times 1.7}{3.9}
            \approx 3.9\,\text{nm}
\end{align*}

\section{第二章\ SOI\ MOS器件}

\subsection{SOI 技术的演变}

\subsubsection{概述}

\textbf{\hlblue{SOI 核心定义}}

SOI 即 \textbf{Silicon-on-insulator}（绝缘体上硅）。即\textbf{\textcolor{myred}{在沟道与衬底之间引入绝缘层进行电学隔离}}。

\imgleft[0.2]{images/image-2026-06-11-14-28-12.png}{
\textbf{\hlblue{三层物理结构}}

\concepttable{
    \conrow{表面硅薄层（顶层硅）}{用于制作器件的源（S）、漏（D）和沟道。}
    \conrow{\textbf{\textcolor{myred}{埋层氧化物（BOX）}}}{通常为 $\text{SiO}_2$，起到关键的电学绝缘隔离作用。}
    \conrow{硅衬底}{位于最底层，主要起\textbf{机械支撑}作用。}
}

\entry{技术演变}{SOI 并非一步到位，其前身是早期的 \textbf{SOS}（蓝宝石上硅）技术。}
}

\textbf{\hlblue{SOI器件的电学优势}}

\concepttable{
    \conrow{成本低}{与体硅为同平台技术，可以节约制造成本。}
    \conrow{强抗辐射}{绝缘衬底\textbf{\textcolor{myred}{阻断高能粒子在衬底中的电荷收集路径}}，有效抑制单粒子效应。}
    \conrow{低寄生电容}{消除源/漏与衬底间 pn 结面积，\textbf{\textcolor{myred}{寄生电容大幅降低}}，提升频率、降低功耗。}
    \conrow{提高电路性能}{寄生效应减小直接带来\textbf{\textcolor{myred}{芯片速度的整体提高}}。}
    \conrow{消除闩锁效应}{体硅中寄生晶闸管导致闩锁；SOI 介质完全隔离各器件，切断寄生路径，彻底消除。}
}

\textbf{\hlblue{SOI器件的缺陷}}

\concepttable{
    \conrow{浮体效应}{没有零电位参考面，对AC特性影响较为复杂，需要精确模型。}
    \conrow{背界面缺陷}{衬底技术引入背界面缺陷，导致载流子散射、俘获、泄漏。}
    \conrow{Scaling 挑战}{需要极薄的硅膜厚度，会引起相当的器件波动。}
    \conrow{自热效应}{绝缘层不利于热耗散，造成器件电流退化。}
}

\subsubsection{现代 SOI 晶圆制造}

业界方法是利用\textbf{\textcolor{myred}{嵌入或者键和}}的方法形成\textbf{\textcolor{myred}{埋层氧化物隔离}}顶层硅薄膜层和硅衬底

\concepttable{
    \conrow{难点}{\textbf{\textcolor{myred}{非晶态 $\text{SiO}_2$ 表面无法直接外延生长单晶硅}}。}
    \conrow{解决思路}{不直接生长，只能利用嵌入或键合方法，\textbf{\textcolor{myred}{在结构内部形成埋层氧化物（BOX）}}，以此强行隔离顶层单晶硅薄膜和硅衬底。}
}

\textbf{\hlblue{主流 SOI 制造技术}}

\concepttable{
    \conrow{SIMOX（注入氧分离）}{\textbf{\textcolor{myred}{高剂量氧离子注入硅中}}，高温退火在内部形成 $\text{SiO}_2$ 层。}
    \conrow{BESOI（键合回蚀）}{\textbf{\textcolor{myred}{两片氧化后的硅片表面键合}}，再从\textbf{\textcolor{myred}{背面回蚀}}减薄其中一片至所需薄膜厚度。}
    \conrow{Smart-Cut（智能剪切）}{利用氢离子注入在硅片内部形成损伤带，键合后通过热/机械剥离精准切断。}
}

\textbf{\hlblue{SIMOX技术}}

\imgleft[0.24]{images/image-2026-06-11-14-42-44.png}{

\concepttable{
    \conrow{物理原理}{利用离子注入技术，将高剂量的氧离子（$\text{O}^+$）强行注入到单晶硅晶圆内部，\textbf{\textcolor{myred}{在特定深度通过化学反应形成 $\text{SiO}_2$ 埋层}}，以此将顶层单晶硅薄膜与底部硅衬底在电学上彻底隔离。}
}

\textbf{工艺优势：}

\concepttable{
    \conrow{工艺简单}{属于单片加工流程，不需要复杂的键合与减薄工序。}
    \conrow{硅膜质量好}{顶层单晶硅直接继承原硅片的晶格结构。}
    \conrow{产能高}{适合批量离子注入生产。}
}

\textbf{技术局限性：}不能制备超薄SOI硅膜。
}

\textbf{\hlblue{BESOI技术}}

\textbf{基本原理与缺点}

\concepttable{
    \conrow{物理原理}{通过键合技术把两片标准硅晶圆紧密键合在一起，\textbf{\textcolor{myred}{晶圆之间预先通过热氧化形成的 $\text{SiO}_2$ 层作为埋层氧化物（BOX）}}。键合完成后，再利用回蚀技术把其中一侧晶圆的厚度削薄到目标厚度，形成SOI晶圆。}
}

\entry{技术局限性}{\textbf{\textcolor{myred}{均匀性差，薄膜难制备}}。通过机械或化学减薄（回蚀）大面积晶圆很难精准控制残留厚度，导致表面顶层硅的厚度均匀性极差。}

\entry{技术演进}{BESOI 不适合制取超薄顶层硅。为克服这一缺陷，由 BESOI 衍生出了目前最主流的 Smart-Cut 技术。}

\textbf{\hlblue{Smart-Cut技术}}

\concepttable{
    \conrow{核心优势}{克服 SIMOX 和 BESOI 无法做到超薄、高均匀性顶层硅的缺陷，\textbf{\textcolor{myred}{是当前商业化最主流的 SOI 制造技术}}。}
}

\subsection{SOI 器件特性}

\subsubsection{阈值电压}

\textbf{\hlblue{PD SOI}}

对于 PD SOI 器件，由于其中性区的隔离作用，正栅和背栅之间没有电学耦合，器件在电学上等效为由正栅和背栅\textbf{\textcolor{myred}{独立控制的两个体硅器件并联}}，故在此不做讨论。

\textbf{\hlblue{FD SOI}}

\imgleft[0.3]{images/image-2026-06-12-05-08-33.png}{
\concepttable{
    \conrow{电学耦合状态}{正栅与背栅\textbf{\textcolor{myred}{相互耦合}}$\rightarrow$\textbf{\textcolor{myred}{两者 $V_T$ 相互影响}}。}
    \conrow{简化模型}{忽略短沟道效应，仅考虑\textbf{\textcolor{myred}{一维（1D）SOI MOS 结构}}，沿深度方向做一维静电学分析（和半导体器件物理类似）}
}
}

\concepttable{
    \conrow{电势分布}{$\phi(x) = \frac{qN_Ax^2}{2\varepsilon_s} + \left( \frac{\phi_{sb} - \phi_{sf}}{T_{si}} - \frac{qN_AT_{si}}{2\varepsilon_s} \right)x + \phi_{sf}$}
    \conrow{电场分布}{$E(x) = -\frac{qN_Ax}{\varepsilon_s} + \frac{\phi_{sf} - \phi_{sb}}{T_{si}} + \frac{qN_AT_{si}}{2\varepsilon_s}$}
}

\concepttable{
    \conrow{正表面电场}{$E_{sf} = \dfrac{\phi_{sf} - \phi_{sb}}{T_{si}} + \dfrac{qN_AT_{si}}{2\varepsilon_s}$}
    \conrow{背表面电场}{$E_{sb} = -\dfrac{qN_AT_{si}}{\varepsilon_s} + E_{sf}$}
}

\textbf{氧化层电压降：}

设正、背栅氧化层电容分别为 $C_{oxf}$ 和 $C_{oxb}$：

\concepttable{
    \conrow{正栅氧化层电压降}{$V_{oxf} = E_{oxf}T_{oxf} = \dfrac{\varepsilon_s E_{sf}}{\varepsilon_{ox}}T_{oxf} = \dfrac{\varepsilon_s E_{sf}}{C_{oxf}}$}
    \conrow{背栅氧化层电压降}{$V_{oxb} = E_{oxb}T_{oxb} = -\dfrac{\varepsilon_s E_{sb}}{\varepsilon_{ox}}T_{oxb} = -\dfrac{\varepsilon_s E_{sb}}{C_{oxb}}$}
}

\textbf{正/背栅电压最终表达式与电荷耦合:}（其中硅膜电容：$C_{Si} = \dfrac{\varepsilon_s}{T_{si}}$）

\concepttable{
    \conrow{正栅电压}{$$V_{Gf} = V_{FBf} + \left(1 + \frac{C_{Si}}{C_{oxf}}\right)\phi_{sf} - \frac{C_{Si}}{C_{oxf}}\phi_{sb} + \frac{qN_AT_{si}}{2C_{oxf}}$$}
    \conrow{背栅电压}{$$V_{Gb} = V_{FBb} - \frac{C_{Si}}{C_{oxb}}\phi_{sf} + \left(1 + \frac{C_{Si}}{C_{oxb}}\right)\phi_{sb} + \frac{qN_AT_{si}}{2C_{oxb}}$$}
}

\entry{电荷耦合关系}{正反栅电压 $V_{Gf/Gb}$ 分别受背正表面电势 $\phi_{sb/sf}$ 控制}


\textbf{\textcolor{myred}{阈值电压：}}

\textbf{1. 背栅为积累型}

\concepttable{
    \conrow{边界条件}{$\phi_{sf} = 2\phi_{Fp}$（强反型导通）；$\phi_{sb} \sim 0$（积累态，电势几近为零）}
    \conrow{阈值电压 $V_{T,ba}$}{将边界条件代入正栅压通用公式，消除 $\phi_{sb}$ 项：$$V_{T,ba} = V_{FBf} + \left(1 + \frac{C_{Si}}{C_{oxf}}\right)2\phi_{Fp} + \frac{qN_AT_{si}}{2C_{oxf}}$$}
    \conrow{变化特征}{背面多子积累，形成了静电屏蔽，使得 $V_T$ 不随 $V_{Gb}$ 改变。}
}

\textbf{2. 背栅为反型}

\concepttable{
    \conrow{边界条件}{$\phi_{sf} = \phi_{sb} = 2\phi_{Fp}$。}
    \conrow{阈值电压 $V_{T,bi}$}{代入正栅压通用公式，$\phi_{sf}$ 与 $\phi_{sb}$ 项部分抵消，化简得：$$V_{T,bi} = V_{FBf} + 2\phi_{Fp} + \frac{qN_AT_{si}}{2C_{oxf}}$$}
    \conrow{变化特征}{背面由于强反型起到静电屏蔽作用，阻断了正背栅耦合，使 $V_T$ 达到饱和最小值不再发生改变。}
}

\textbf{3. 背栅为耗尽型（主要工作区间）}

\imgleft[0.25]{images/image-2026-06-12-10-06-09.png}{
    \concepttable{
    \conrow{边界条件}{正面电势 $\phi_{sf} = 2\phi_{Fp}$（强反型）。}
    \conrow{背表面电势特征}{从积累态 $0$ 到反型态 $2\phi_{Fp}$。}
    \conrow{阈值电压}{在积累型与反型态之间呈线性函数关系。}
}

\textbf{表达式：}

$$V_{T,bd} = V_{T,bi} - \frac{C_{Si}C_{oxb}}{C_{oxf}(C_{Si}+C_{oxb})}(V_{Gb} - V_{Gb,i})$$

}

\textbf{\textcolor{myred}{调控规律：}}背栅压越大，正栅阈值电压越小。

\subsubsection{电流电压特性}

由于FD SOI中正背栅电荷耦合，所以$I_D$表示为三自变量电压的函数：

$$I_D = I_D(V_{Gf}, V_{Gb}, V_D)$$

依然假设电流-电压方程满足 MOSFET 的一般框架：

$$I_D = \beta \left[ (V_{GS} - V_T)V_{DS} - \frac{1}{2}V_{DS}^2 \right]$$

\textbf{\blue{背栅为耗尽型}}（依然研究主要工作区间）

\entry{电流公式}{
        
    $$\boxed{I_{D,d} = \frac{Z}{L}\mu_{nf}C_{oxf}\left[ (V_{Gf} - V_{T,bd})V_D - \left( 1 + \frac{C_{Si}C_{oxb}}{(C_{Si}+C_{oxb})C_{oxf}} \right) \frac{V_D^2}{2} \right]}$$
}

\textbf{\textcolor{myred}{调控规律：}}背面耗尽时，$I_D$ 随 $V_{Gb}$ 增大而增大

\subsubsection{亚阈值效应}

\textbf{\hlblue{亚阈值斜率 $S$}}

\entry{物理意义}{表征\textbf{\textcolor{myred}{栅电压对亚阈值漏电流的控制能力}}：$S$ 越小，栅控能力越强，关态向开态转换越快。}

\textbf{\hlblue{背面耗尽态下的 $S_{bd}$}}

\entry{表达式}{$$S_{bd} = (\ln 10)\frac{kT}{q}\left[1 + \frac{C_{Si}C_{oxb}}{C_{oxf}(C_{Si} + C_{oxb})}\right]$$}

一般情况下，埋氧化层（BOX）厚度远大于顶层硅层厚度和正栅氧化层厚度。电容大小与介质厚度成反比，因此：

$$C_{oxb} \ll C_{Si} \quad \text{和} \quad C_{oxb} \ll C_{oxf}$$

代入 $S_{bd}$ 公式，$C_{oxb}$ 极小使 $\frac{C_{Si}C_{oxb}}{C_{oxf}(C_{Si} + C_{oxb})} \to 0$，背面耗尽时亚阈值斜率近似理想值：

$$S \approx (\ln 10)\frac{kT}{q}$$

\entry{电学优势}{\textbf{\textcolor{myred}{背面耗尽器件亚阈值斜率小}}，关态泄漏电流小，有利于大幅降低芯片静态功耗。}


\subsection{SOI 器件增强技术}

\subsubsection{短沟道效应}

\entry{短沟道特性判据}{$L/\lambda_1$，即沟道长度与特征长度之比。\textbf{\textcolor{myred}{比值越大，抗短沟道效应特性越好。}}}

\imgleft[0.3]{images/image-2026-06-24-05-32-42.png}{

\entry{特征长度$\lambda_1$}{$$\lambda_1 = \sqrt{\frac{\varepsilon_s}{\varepsilon_{ox}} T_{Si} T_{oxf}}$$}

由特征长度表达式可知，增强抗短沟道性可以通过减小 $T_{Si}$ 或 $T_{oxf}$ 来实现。

于是 SOI 器件的发展顺序：\textbf{\textcolor{myred}{PD-SOI $\rightarrow$ FD-SOI $\rightarrow$ UTB-SOI}}
}

\subsection{器件的涨落}

\subsubsection{杂质随机分布(RDD)}

\textbf{物理机理}：\textbf{\textcolor{myred}{离子注入与扩散在原子尺度离散随机}}。纳米尺度下（$L_g < 100\text{ nm}$）沟道耗尽层体积 $W \cdot L \cdot X_{dep}$ 极小，杂质原子总数 $N$ 降至数十至数百个。

\concepttable{
    \conrow{杂质数目涨落}{沟道内杂质原子总数的统计起伏。}
    \conrow{杂质位置涨落}{杂质原子在耗尽层内的具体坐标随机变化。}
}

\entry{杂质数目 N}{服从泊松分布，$\sigma_N = \sqrt{N}$，相对涨落 $\frac{\sigma_N}{N} = \frac{1}{\sqrt{N}}$。\textbf{\textcolor{myred}{$N$ 越小，涨落越大，相邻器件杂质数目出现显著微观差异}}。}

RDD通过掺杂浓度 $N_A$ 的起伏\textbf{\textcolor{myred}{映射为阈值电压 $V_T$ 波动}}，其中主要影响系数 $K \propto \sqrt{N_A}$，掺杂浓度的微观起伏直接传导至 $V_T$。

\textbf{\hlblue{杂质数目涨落影响}}

\entry{掺杂浓度相对涨落}{$\frac{\sigma_{N_A}}{N_A} \approx 2.4\%$}
\entry{沟道杂质原子数相对起伏}{$\frac{\sigma_{n_a}}{n_a} \approx 2.3\%$}

\imgleft[0.25]{images/image-2026-06-09-20-54-07.png}{
$\sigma_{V_T}$（阈值电压标准差）随 $L_{eff}$ （有效栅长）缩小的演进：

\concepttable{
    \conrow{$0.5\,\mu\text{m}$}{$\sigma_{V_T} = 6.9\text{ mV}\ (\approx 1.4\%)$}
    \conrow{$0.3\,\mu\text{m}$}{$\sigma_{V_T} = 10.6\text{ mV}\ (\approx 2.2\%)$}
}

\textbf{\textcolor{myred}{随 $L_{eff}$ 缩小，$\sigma_{V_T}$ 显著增大，分布（近高斯）变宽变矮}}，器件间性能离散度增加。影响 $\sigma_{V_T}$ 的设计变量还包括栅氧化层厚度、栅宽及掺杂浓度本身。
}

\concepttable{
    \conrow{栅氧化层厚度 $t_{ox}$}{\textbf{\textcolor{myred}{$t_{ox}\uparrow\ \Rightarrow\ \sigma_{V_T}\uparrow$}}。
    
    \textbf{机理：}$t_{ox}$ 增大 $\rightarrow$ $C_{ox}$ 减小 $\rightarrow$ 栅对沟道电荷控制减弱，微观电荷起伏对 $V_T$ 的调制放大。}
    \conrow{器件面积 $W\times L$}{\textbf{\textcolor{myred}{$WL\downarrow$（即 $1/\sqrt{LW}\uparrow$）$\Rightarrow$ $\sigma_{V_T}\uparrow$}}。
    
    \textbf{机理：}符合面积平方根倒数定律，沟道内杂质原子绝对数越少，统计起伏越剧烈。}
    \conrow{掺杂浓度 $N$}{\textbf{\textcolor{myred}{$N\uparrow\ \Rightarrow\ \sigma_{V_T}\uparrow$}}。
    
    \textbf{机理：}耗尽区电荷总量随 $N$ 增大而增大，绝对电荷涨落总量同步上升。}
}

\subsubsection{线条边沿粗糙（LER）}

\entry{定义}{\textbf{\textcolor{myred}{栅极线条边沿的原子级不规则凹凸不平}}，导致沿栅宽 $W$ 方向局域实际栅长随机变异。}

\imgleft[0.2]{images/image-2026-06-09-21-11-03.png}{

\concepttable{
    \conrow{理想版图}{栅极为规则直线矩形，具有均匀栅长 $L$ 与栅宽 $W$。}
    \conrow{实际形貌}{光刻光子散粒效应与刻蚀离子轰击不均匀 $\rightarrow$ 边沿非陡直，\textbf{\textcolor{myred}{SEM 下呈现微观粗糙}}。}
    \conrow{核心问题}{随器件尺寸缩小，LER 如何演变？$\rightarrow$ 工艺极限下的缩放规律。}
}

}

\entry{工艺极限下 LER 的非比例缩放}{经厂商实验证明：\textbf{\textcolor{myred}{各工艺节点下 LER 绝对值基本保持在 $\sim 5\text{ nm}$ 不变}}。\textbf{\textcolor{myred}{栅 LER 不能按比例缩小}}。随 $L$ 持续微缩，LER 相对栅长占比急剧飙升，$5\text{ nm}$ 级粗糙度对沟道几何产生主导性破坏，导致器件参数剧烈涨落。}

\textbf{\hlblue{LER 数学表征}}

方均根值 $\Delta$（RMS）：

$$\Delta = \left( \frac{1}{n} \sum_{i=1}^n \delta_i^2 \right)^{\frac{1}{2}}$$

$\delta_i$：实际栅边沿与理想边沿的几何位置偏差；$n$：沿线条方向采样点数目。\textbf{\textcolor{myred}{工业界常用 $3\Delta$（3 倍 RMS）表征 LER}}，覆盖约 99.73$\%$ 统计波动。

\imgleft[0.3]{images/image-2026-06-09-21-17-11.png}{

\entry{LWR（线宽粗糙度）}{MOSFET 沟道由\textbf{双侧}栅边缘共同定义，故引入线条宽度粗糙度（LWR）：$$\Delta_{LWR} = \left[ \frac{1}{n} \sum_{i=1}^n (L_{gi} - L_g)^2 \right]^{\frac{1}{2}}$$

其中：$L_{gi}$：局域实际栅线条宽度；$L_g$：平均栅长。}

}

\textbf{双侧相关系数}：

\concepttable{
    \conrow{$\rho = 0$（最常见）}{两侧粗糙不相关，独立叠加：$\sigma_{LWR}^2 = 2\sigma_{LER}^2$。}
    \conrow{$\rho = -1$（负相关）}{互补波浪状，线宽变异最大：$\sigma_{LWR}^2 = 4\sigma_{LER}^2$。}
    \conrow{$\rho = 1$（正相关）}{整体弯曲但宽度恒定，线宽无起伏：$\sigma_{LWR}^2 = 0$。}
}

\textbf{\hlblue{定性影响}}

\entry{尺寸依赖性}{参数标准差随 $W$ 减小而显著增大。}

\entry{不对称影响}{\textbf{\textcolor{myred}{LER 对 $I_{on}$ 影响小，对 $I_{off}$ 影响显著}}。}
\entry{阈值电压 $V_T$}{$\sigma_{V_T}$ 随方均根涨落 $\Delta$ 单调上升；相同 $\Delta$ 下，$L=30\text{ nm}$ 的 $\sigma_{V_T}$ 斜率远高于 $L=50\text{ nm}$，表明\textbf{\textcolor{myred}{器件缩小剧烈放大 LER 对 $V_T$ 的扰动}}，并伴随 $V_T$ 降低效应。}

\textbf{\hlblue{工艺解决方案：侧墙光刻技术}}

\entry{原理}{以 ALD 薄膜厚度与各向异性刻蚀的物理边界定义图形，\textbf{\textcolor{myred}{避开传统光刻光子散射造成的边界畸变}}。}

\concepttable{
    \conrow{常规光刻}{光刻胶分子随机溶解度变异与化学放大胶随机扩散 → 栅边缘锯齿状畸变 → 不同断面 $L_g$ 空间变异剧烈，易诱发局域漏电。}
    \conrow{侧墙光刻}{边缘由均匀薄膜沉积外延侧壁定义 → 沿栅宽方向 $L_g$ 高度一致，消除几何畸变引入的局域短沟道效应。}
}

\subsubsection{栅材料的功函数变化（WKF）}

\entry{物理机制}{金属栅（HK/MG）由随机取向晶粒构成，各晶向逸出功不同 → 栅平面内功函数 $\phi_M$ 空间不均匀。\textbf{\textcolor{myred}{由 $\phi_{MS} = \phi_M - \phi_S$，$\phi_M$ 局域波动直接传导至 $V_T$，引发 $\sigma_{V_T}$ 退化。}}}

\entry{材料依赖性（栅面积 $(16\text{ nm})^2$）}{$\sigma_{V_{th}}$ 由低到高：$\text{TiN}\,(98\text{ mV}) < \text{TaN}\,(120\text{ mV}) < \text{WN}\,(157\text{ mV}) < \text{MoN}\,(295\text{ mV})$。\textbf{\textcolor{myred}{晶粒尺寸与晶向各向异性决定涨落敏感度}}。}

\subsubsection{NMOS 与 PMOS 涨落源对比}

NMOS 和 PMOS上拉电阻的涨落源非对称性对比：

\concepttable{
    \conrow{NMOS $\sigma_{V_{th}}$ 排序}{RDD（$43\text{ mV}$）$>$ PVE（工艺变化造成关键尺寸变化）（$28\text{ mV}$）$>$ WKF（$24.7\text{ mV}$）。\textbf{\textcolor{myred}{沟道掺杂精度是 NMOS 一致性核心}}。}

}

\textbf{\hlblue{电路级时序涨落对比}}

}

\subsection{器件的可靠性}

\textbf{核心定义}：器件在正常工作电场与温度下，因\textbf{\textcolor{myred}{老化（aging）}}引起微观结构不可逆物理病变，导致宏观参数随时间衰减，\textbf{\textcolor{myred}{关注排除早期制造缺陷后的长期退化规律}}。

\entry{物理根源}{宏观电流衰减直接反映老化应力下 \textbf{\textcolor{myred}{$\Delta V_T > 0$（阈值电压正向漂移）或 $\mu_{eff}$（有效迁移率）严重退化}} 等微观病变。}

\entry{可靠性研究方法}{
    \concepttable{
        \conrow{加速退化实验}{工作条件下退化时间很长（~年）；利用高应力条件（高温、高压）加速退化；从高应力条件外推器件寿命。}
        \conrow{退化表征方法}{电学参数提取（阈值电压漂移、饱和漏电流改变）；缺陷测量（CV、电荷泵等，测氧化层陷阱、界面态）。}
        \conrow{缺陷产生模型、寿命预测模型}{预测准确性依赖于退化随时间演化的模型准确性。}
    }
}

\entry{定义}{沟道载流子在强横向电场下加速获得高动能，与氧化层界面作用释放能量，破坏结构并产生缺陷。}

\subsection{挥发性存储器}

\subsubsection{DRAM}

\textbf{\hlblue{1T1C结构}}

\imgleft[0.25]{images/image-2026-06-06-21-12-01.png}{
    \entry{WL（字线）}{接栅极，控制MOS管导通/截止。}\entry{BL（位线）}{接源/漏区，负责数据输入输出。}\entry{PL（极板线）}{与存储电容$C_s$顶部连接。}$C_s = A_s(C_{ox}+C_j) \approx A_s C_{ox}$，\textbf{\textcolor{myred}{氧化层极薄故$C_{ox}\gg C_j$}}。
    
    \textbf{读写}：WL高$\to$MOS导通$\to$BL与电容连通，电荷转移。
    
    \textbf{保持}：WL低$\to$MOS截止$\to$电荷隔离存储。
}

\textbf{\textcolor{myred}{"动态"物理根源}}：DRAM存"1"时pn结反偏$\to$反向泄漏电流$I_{leak}$致电容电荷流失、电平衰退，须周期性刷新。

\imgleft[0.25]{images/image-2026-06-06-21-17-12.png}{
\concepttable{
    \conrow{存"0"}{自然稳定态：反型层与p型衬底无外部电压，仅存自建电势$V_{bi}$，系统平衡。}
    \conrow{存"1"}{非平衡态：反型层带电荷，pn结反偏。}
    \conrow{保持时间}{$t_h = \dfrac{0.2 V_{S1} C_s}{I_{leak}}$，\textbf{\textcolor{myred}{$C_s$越大、$I_{leak}$越小则保持越久}}；系数0.2为允许电压衰减裕度。}
}
}

\textbf{\hlblue{读操作与电荷共享}}

\noindent\textbf{流程}：BL预充至$V_R$ $\to$ WL拉高 $\to$ $C_B$与$C_s$连通 $\to$ 电荷共享，电平趋于一致。

\imgleft[0.4]{images/image-2026-06-08-20-48-14.png}{

\concepttable{
    \conrow{读"0"终态}{$V_{B0} = \dfrac{C_B V_R + C_s V_{S0}}{C_B + C_s}$，检测量 $\Delta V_0 = V_R - V_{B0}$。}
    \conrow{读"1"终态}{$V_{B1} = \dfrac{C_B V_R + C_s V_{S1}}{C_B + C_s}$，检测量 $\Delta V_1 = V_{B1} - V_R$。}
}

\noindent\textbf{两大致命问题}：
\begin{itemize}
    \item \textbf{读出信号微弱}：$C_B \gg C_s$ $\Rightarrow$ \textbf{\textcolor{myred}{$\Delta V$极小，后续电路难以识别}}。
    \item \textbf{破坏性读出}：读后$C_s$电压变为中间电平$V_{B0}$/$V_{B1}$，原存储数据丢失。
\end{itemize}

}

\textbf{\hlblue{灵敏再生放大器（S/R）:}}

\noindent 解决读出信号微弱与破坏性读出两大问题。\textbf{结构}：交叉耦合反相器（两对PMOS+NMOS）构成\textbf{\textcolor{myred}{双稳态触发器}}。

\concepttable{
    \conrow{放大}{将微弱$\Delta V$放大至满幅逻辑电平（$V_{DD}$或GND）。}
    \conrow{再生（回写）}{放大信号立即写回$C_s$，恢复原存储数据，解决破坏性读出。}
}
\textbf{\hlblue{DRAM 的刷新}}

\concepttable{
    \conrow{动态成因}{$C_s$无绝对绝缘，泄漏电流$I_{leak}$使存储电荷持续流失。}
    \conrow{失效阈值}{高电平下降\textbf{\textcolor{myred}{20\%}}即判定信息丢失（对应$\Delta V = 0.2V_{S1}$）。}
    \conrow{刷新操作}{失效前由S/R将衰退电平重新放大并写回$C_s$，恢复标准逻辑电平。}
}

\textbf{\hlblue{刷新效率 $\eta$（系统性能指标）}}

\noindent 刷新期间DRAM无法响应CPU读写请求（死区时间）。$\eta$越小则有效读写时间占比越大。

$$\eta = \frac{m \times t_{RC}}{t_h} \times 100\%$$

\concepttable{
    \conrow{$t_h$}{保持时间，电荷衰减达20\%阈值的物理时间极限，为考查时间窗口。}
    \conrow{$m$}{刷新周期数（阵列需刷新的总行数）。}
    \conrow{$t_{RC}$}{单行刷新周期耗时。}
    \conrow{$m \times t_{RC}$}{$t_h$窗口内所有刷新操作的总物理耗时。}
}

\noindent\textbf{实例}（256Kb DRAM）：$m=256$，$t_h=4\text{ms}$，$t_{RC}=400\text{ns}$。
分子：$256 \times 400 \times 10^{-9} = 102.4\,\mu\text{s}$；分母：$4 \times 10^{-3} = 4000\,\mu\text{s}$。

$$\eta = \frac{102.4}{4000} \times 100\% = 2.56\%$$

即4ms窗口内2.56\%时间被刷新锁死，97.44\%可供正常存取。

\textbf{\hlblue{核心结构}}

\imgleft[0.4]{images/image-2026-06-08-21-32-02.png}{

SRAM核心为\textbf{\textcolor{myred}{交叉耦合反相器对}}（节点N1/N2），外接双位线（BL / $\overline{BL}$）与单字线（WL）控制两访问管M1/M2。

\concepttable{
    \conrow{写"1"}{WL置高$\to$M1/M2导通；BL拉高、$\overline{BL}$拉低$\to$N1充电、N2放电$\to$强制锁入 N1=1,N2=0 稳态。}

\concepttable{
    \conrow{读操作}{读前BL与$\overline{BL}$\textbf{\textcolor{myred}{预充至高电平}}；WL置高$\to$N2=0时$\overline{BL}$经M2放电（读"1"），N1=0时BL经M1放电（读"0"），在位线对产生差分电压差。}

}

}

    \compheader{类型 & 负载 & 静态电流 & 特征}

\textbf{\textcolor{myred}{对比DRAM}}：静态正反馈锁存，无需刷新，速度极快，广泛用于cache与嵌入式存储。

\textbf{\hlblue{SRAM单元稳定性}}

\concepttable{
    \conrow{不稳定性来源}{\textbf{内部：}工艺偏差致阈值电压/宽长比失配；
    
    \textbf{外部：}$\alpha$粒子辐射、线路串扰、$V_{DD}$波动。干扰超阈值则正反馈破坏$\to$状态翻转。}
    \conrow{写稳定性}{外部驱动强制改变内部状态的难易程度——越易写入则写稳定性越好。}
    \conrow{读稳定性}{读操作时内部节点受位线电压扰动，抵抗干扰维持原态的能力。}
}

\textbf{\hlblue{写稳定性与$\beta_Q$}}

\imgleft[0.4]{images/image-2026-06-08-21-46-30.png}{

\concepttable{
    \conrow{物理过程（写"0"）}{WL拉高$\to$M5导通，BL拉低$\to$M3与M5在N1形成$V_{DD}$-GND分压，门管须强于负载管方可拉低N1翻转。}
    \conrow{核心参数}{$\displaystyle\beta_Q=\frac{K_p}{K_a}=\frac{K_3}{K_5}=\frac{K_4}{K_6}$；$K_p$为负载管（PMOS）导电因子，$K_a$为门管（NMOS）导电因子。}
    \conrow{物理本质}{$\beta_Q$越小写入越易：门管导电能力须强于负载管，方能将N1电位拉低越过反相器翻转阈值。故设计上需\textbf{\textcolor{myred}{减小PMOS尺寸、增大传输门NMOS尺寸}}。}
}

}

\textbf{\hlblue{写操作动态机制}}

写翻转的本质是\textbf{\textcolor{myred}{强行拉低原本为"1"的节点}}。因读稳定性约束，外部难以通过PG将"0"节点直接拉高，突破口在"1"节点N2。

\concepttable{
    \conrow{第1阶段}{初始 N1=0,N2=1。外部$\overline{BL}$低电平经PG2强行拉低$V_{N2}$。}
    \conrow{第2阶段}{$V_{N2}$低于反相器开关阈值时正反馈启动：PU1导通$\to$N1充电($0\!\to\!1$)；PU2截止$\to$N2彻底放电至0，写入完成。}
}

\noindent\textbf{\hlblue{等效分压与$\gamma$ ratio}}

写入瞬间N2节点经历PU2（拉高）与PG2（拉低）的电荷角力。

$$\gamma=\frac{(W/L)_{PU}}{(W/L)_{PG}}\;\text{等效于}\;\beta_Q=\frac{K_p}{K_a}$$

PG导电能力须战胜PU——$I_{\text{WRITE}}$主要流向GND而非维持N2电位。WL开启后$\overline{BL}$拉低，$V_{N2}$从$V_{DD}$开始下降，\textbf{\textcolor{myred}{PG强度必须克服PU强度方能率先拉低"1"}}。

\textbf{\hlblue{读稳定性与$\beta_R$（Cell Ratio）}}

读稳定性是写稳定性的对立面。读操作前BL与$\overline{BL}$预充至$V_{DD}$。以读"1"（N1=1, N2=0）为例：WL开启，M6导通，$\overline{BL}$高电平经M6向N2注入电荷，形成$\overline{BL}$-M6-N2-M2-GND通路，N2产生瞬态电压扰动$\Delta V$。

$$\beta_R = \frac{K_d}{K_a} = \frac{K_1}{K_5} = \frac{K_2}{K_6}$$

\concepttable{
    \conrow{$K_d$}{驱动管（Pull-Down NMOS M1/M2）导电因子。}
    \conrow{$K_a$}{门管（Pass-Gate NMOS M5/M6）导电因子。}
    \conrow{物理本质}{M2须远强于M6，将$\overline{BL}$涌入电荷迅速泄放至地，\textbf{\textcolor{myred}{钉住N2于低电位}}，防止$\Delta V$超反相器阈值致误翻转。}
    \conrow{设计原则}{$\beta_R \uparrow\ \Rightarrow$ 读稳定性$\uparrow$；驱动管NMOS尺寸须远大于门管NMOS。}
}

\subsection{不挥发性存储器}

\subsubsection{Mask ROM}

\entry{基本结构与工作原理}{
    每个存储单元仅由一个MOS晶体管构成。引脚连接：栅极（Gate）接字线（WL），漏极（Drain）接位线（BL），源极（Source）共同接地（GND）。
    
    \textbf{物理本质：}存储的逻辑内容（“0”或“1”）完全由管子本身的导通或截止状态决定。
}

\entry{三种编程方法及物理机制}{
    掩膜式ROM在出厂制造时就通过光刻步骤决定了数据，其具体的物理实现手段有三种：
    \concepttable{
        \conrow{离子注入掩膜版编程}{沟道注入硼离子，$V_{TH}$升高，读取时截止，位线不放电。}
        \conrow{有源区掩膜版编程}{控制有源区与栅极叠置，无叠置则无沟道，永久截止。}
        \conrow{引线孔掩膜版编程}{控制接触孔有无，有孔导通放电存“1”，无孔保持预充存“0”。}
    }
}

\subsubsection{PROM}

\entry{单元电路拓扑}{
    采用多晶硅电阻负载PROM单元。其物理层叠为：位线（BL） $\rightarrow$ 高阻多晶硅电阻 $\rightarrow$ MOS晶体管漏极 $\rightarrow$ 源极接地。字线（WL）同样接MOS晶体管的栅极。
}

\entry{阻态与逻辑状态映射}{
    \concepttable{
        \conrow{高阻态（HR-poly）}{单元只有很小的导通电流，系统判决为存“0”。}
        \conrow{低阻态（$\text{n}^+$-poly）}{单元具有较大的导通电流，系统判决为存“1”。}
    }
}

\subsubsection{FeRAM}(Ferroelectric RAM，铁磁随机访问存储器)

\imgleft[0.3]{images/image-2026-06-26-10-31-12.png}{
    \entry{等效电路与拓扑结构（1T1C）}{
        FeRAM在电路拓扑上\textbf{\textcolor{myred}{完全继承了传统DRAM的1T1C}}（1个晶体管+1个电容）架构：
        \concepttable{
            \conrow{WL（字线）}{连接至选择晶体管的栅极（Word Line），控制开关管的导通与截止。}
            \conrow{BL（位线）}{连接至晶体管的源/漏端（Bit Line），用作数据的输入与输出，图中 $C_B$ 代表位线寄生电容。}
            \conrow{$C_{FE}$（铁电电容）}{晶体管的另一个源/漏端垂直连接至铁电电容的上电极。}
            \conrow{PL（极板线）}{铁电电容的下电极统一连接至板线（Plate Line）。}
        }
        \textbf{与DRAM的本质区别：}传统DRAM电容采用普通介质材料，断电后电荷泄露导致信息丢失；而FeRAM将常规电容介质替换为铁电材料，利用其非挥发性物理特性实现非易失性存储。
    }

\entry{存储原理}{
    利用铁电材料\textbf{\textcolor{myred}{在外部电场作用下极化方向发生翻转}}的特性来实现信息的永久存储。即使断开外部供电，材料内部的极化状态依然保持。
}

\imgleft[0.3]{images/image-2026-06-26-10-33-54.png}{
    \entry{电滞曲线（$P \sim E$ 或 $P \sim V$ 曲线）}{
        铁电材料的核心特征体现在其电滞回线上，纵轴为极化强度（$P$），横轴为电压（$V$）或电场（$E$）：
        \concepttable{
            \conrow{$P_s$（自发极化强度）}{施加外加电场使材料内部铁电畴趋向完全对齐时的最大饱和极化强度（图顶部的虚线横坐标投影）。}
            \conrow{$P_r$（剩余极化强度}{当外部电场/电压彻底移除（$V=0$）时，材料内部依然保持的极化强度（即曲线与纵轴的交点）}
        }
    }
    \entry{存储状态与逻辑映射}{
        铁电材料在零偏置下有两个物理稳态，即正向剩余极化强度 $+P_r$（点D）和负向剩余极化强度 $-P_r$（点G）：
        \concepttable{
            \conrow{状态“0”}{当极化强度处于 $+P_r$（点D）状态时，定义为存储逻辑 “0”。}
            \conrow{状态“1”}{当极化强度处于 $-P_r$（点G）状态时，定义为存储逻辑 “1”。}
        }
    }
}

\subsubsection{MRAM}(Magnetic Random Access Memory，磁性随机访问存储器)

\entry{核心效应}{
    基于磁阻隧道结（MTJ），利用巨磁阻（GMR）效应来实现信息存储。
}

\imgleft[0.3]{images/image-2026-06-26-10-24-38.png}{
    \entry{MTJ的三层物理结构}{
        \concepttable{
            \conrow{自由层}{材料为 NiFe。其磁化方向不固定，可随外加磁场发生翻转。}
            \conrow{绝缘层}{材料为 AlOx。}
            \conrow{固定层}{材料为 NiFe。其磁化方向被固定，不随外部操作磁场翻转。}
        }
    }
    \entry{阻态与逻辑定义}{
        \concepttable{
            \conrow{同向低阻}{当自由层与固定层的磁化方向一致（平行）时，电子隧穿阻碍较小，器件呈现低阻态，对应逻辑 “1”。}
            \conrow{反向高阻}{当两层的磁化方向相反（反平行）时，电子阻碍较大，器件呈现高阻态，对应逻辑 “0”。}
        }
    }
}

\subsubsection{PRAM}(Phase-change RAM，\textbf{\textcolor{myred}{相变}}随机访问存储器)

\imgleft[0.3]{images/image-2026-06-26-10-00-11.png}{
    \entry{核心材料与机制}{
        中间的相变材料采用硫族化合物，具体为 GST 合金（$\text{Ge}_x\text{Sb}_y\text{Te}_z$ / 锗锑碲合金）。利用材料在\textbf{\textcolor{myred}{非晶相和结晶相之间的可逆转变}}来代表逻辑“0”和“1”。
    }
    \concepttable{
        \conrow{非晶态}{对应逻辑 “0”。其内部原子排列无序，物理特征为\textbf{\textcolor{myred}{电阻率高（高阻态）}}。}
        \conrow{结晶态}{对应逻辑 “1”。其内部原子排列规则，物理特征为\textbf{\textcolor{myred}{电阻率低（低阻态）}}。}
    }
    \entry{微观}{
        当阻性电极通电时，产生的焦耳热会加热与其接触的加热头附近的一小块 GST 材料，\textbf{\textcolor{myred}{通过控制加热温度和散热速率，实现两态之间的物理转换}}。
    }
}

\textbf{\hlblue{PRAM的工作原理}}

\imgleft[0.3]{images/image-2026-06-26-10-18-33.png}{
    \entry{擦除（RESET）—— 写“0”}{
        \concepttable{
            \conrow{施加电流脉冲}{脉宽小、幅度大（\textbf{\textcolor{myred}{高窄脉冲}}）。}
            \conrow{材料局部温度 $T$}{必须大于材料熔点 $T_m$（$T > T_m$），\textbf{\textcolor{myred}{使 GST 材料熔化}}。}
            \conrow{\textbf{\textcolor{myred}{散热要求}}}{脉冲的下降时间 $t_f$ 必须小于结晶感应时间 $t_{in}$（$t_f < t_{in}$）。}
            \conrow{结果}{由于施加的是窄脉冲且瞬间切断电流，\textbf{\textcolor{myred}{材料在熔化后经历极快速的降温，导致原子来不及排列结晶，直接锁定为无序的非晶态}}，器件呈现高阻，成功写入“0”。}
        }
    }
}

\entry{编程（SET）—— 写“1”}{
    \concepttable{
        \conrow{施加电流脉冲}{脉宽大、幅度小（低宽脉冲）。}
        \conrow{材料局部温度 $T$}{必须满足 $T_x < T < T_m$（即大于结晶温度 $T_x$，但小于熔点 $T_m$）。}
        \conrow{\textbf{\textcolor{myred}{时间要求}}}{脉冲的宽度 $t_2$ 必须大于结晶感应时间 $t_{in}$（$t_2 > t_{in}$）。}
        \conrow{结果}{材料在此温度区间内保持足够长的时间，\textbf{\textcolor{myred}{原子有充分的时间进行重新排列}}，完成重结晶过程，转变为结晶态，器件呈现低阻，成功写入“1”。}
    }
}

\entry{读操作}{
    \concepttable{
        \conrow{偏置条件}{
            被选中单元的字线（WL）加高电平（使 1T1R 中的 MOS 管完全导通）。位线（BL）加低电压（施加一微小的读取电压 $V_R$）。
        }
        \conrow{状态判决逻辑}{
            \textbf{\textcolor{myred}{保持偏置电压不变，检测电流}}：电流小则为高阻态（非晶态），读“0”；电流大则为低阻态（结晶态），读“1”。
        }
        \conrow{非破坏性读出原理}{
            读操作施加远低于阈值电压 $V_{th}$ 的读取电压 $V_R$（$V_R \ll V_{th}$），使焦耳热温度低于 $T_x$ 和 $T_m$，\textbf{\textcolor{myred}{避免相变}}，保证数据完整无损。
        }
    }
}

\section{第六章\ 新原理器件}

\subsection{传统 MOSFET 器件的局限性}

\subsubsection{功耗问题}

由功耗公式：

$$P = C_{\text{L}}V_{\text{DD}}^2\alpha f + I_{\text{leak}}V_{\text{DD}}$$

\imgleft[0.3]{images/image-2026-06-14-15-57-08.png}{
\concepttable{
    \conrow{动态功耗（第一项）}{$C_{\text{L}}V_{\text{DD}}^2\alpha f$，由\textbf{\textcolor{myred}{负载电容充放电}}引起。}
    \conrow{静态功耗（第二项）}{$I_{\text{leak}}V_{\text{DD}}$，即漏电功耗，由器件关态下的\textbf{\textcolor{myred}{漏电流}}引起。}
}

随技术节点减小（特征尺寸按比例缩小），动态功耗和静态功耗均呈指数级增加，\textbf{\textcolor{myred}{10nm 节点}}以下静态功耗急剧上升，严重程度已与动态功耗相当（如左图所示）。
    
\entry{解决手段}{在持续集成的同时进一步降低芯片总功耗，最直接、最核心的手段是\textbf{\textcolor{myred}{降低工作电压 $V_{\text{DD}}$}}。}
}

\hlblue{\textbf{降低 $V_{\text{DD}}$ 的限制因素}}

\imgleft[0.3]{images/image-2026-06-14-16-00-25.png}{
\entry{驱动电流约束}{为维持开态电流 $I_{\text{ON}}$，降低 $V_{\text{DD}}$ 必须同步降低 $V_{\text{T}}$（$I_{\text{ON}} \propto V_{\text{DD}} - V_{\text{T}}$）。}

\entry{亚阈值区指数关系}{$I_{\text{D}} \propto \exp\!\left(\frac{V_{\text{G}}}{kT/q}\right)$，亚阈值斜率 $S$ 固定。}

\entry{关态漏电爆炸}{$V_{\text{T}} \downarrow\ \Rightarrow\ \lg I_{\text{D}}\text{–}V_{\text{G}}$ 曲线左移 $\Rightarrow\ I_{\text{OFF}}\ (I_{\text{leak}})$ 指数级$\uparrow$。}

\concepttable{
    \conrow{\textbf{核心矛盾}}{降低 $V_{\text{DD}}$ 会导致静态漏电功耗爆炸。因此当前降低功耗的终极目标聚焦于\textbf{\textcolor{myred}{控制关态泄漏电流（$I_{\text{OFF}}$ / $I_{\text{leak}}$）}}。}
}
}

\imgleft[0.4]{images/image-2026-06-14-16-03-33.png}{
\entry{传统 MOSFET 漏电机制}{目标是控制关态漏电流，需准确写出名称及对应位置。}

\concepttable{
    \conrow{亚阈值沟道电流 $I_{\text{sub}}$}{源漏之间通过沟道表面的漏电。}
    \conrow{栅隧穿电流 $I_{\text{G}}$}{栅介质减薄，电子隧穿通过栅极引起的漏电。}
    \conrow{结反向电流 $I_{\text{J}}$}{源/漏极与衬底反偏 p-n 结的漏电流。}
    \conrow{栅致漏漏电 $I_{\text{GIDL}}$}{栅漏交叠区高电场引起的带带隧穿漏电。}
}

\entry{结论}{在制约器件按比例缩小的所有漏电机制中，最主要的挑战是\textbf{\textcolor{myred}{亚阈值沟道电流 $I_{\text{sub}}$}}。}
}

\subsubsection{传统 MOSFET 的亚阈值摆幅极限}

\entry{物理含义}{$S$（或称 $SS$）表示漏极电流改变一个数量级（10倍）所需的栅电压变化量。$S$ 越小，器件关断越快。}

\entry{公式}{$$S \equiv \frac{\text{d}V_{\text{GS}}}{\text{d}(\lg I_{\text{Dsub}})} = \ln(10)\frac{nkT}{q} = \ln(10)\left[1 + \frac{C_{\text{D}}(\phi_{\text{S}})}{C_{\text{ox}}}\right]\frac{kT}{q}$$}


\concepttable{
    \conrow{经典理论极限}{传统器件结构决定 $1 + \frac{C_{\text{d}}}{C_{\text{ox}}} \ge 1$。室温（$T = 300\text{ K}$）下，$\ln(10)\frac{kT}{q} \approx 60\text{ mV/dec}$。故传统 MOSFET 的亚阈值摆幅存在不可逾越的物理极限：\textbf{\textcolor{myred}{$SS \ge 60\text{ mV/dec}$}}。}
}

\subsection{负电容效应晶体管（NCFET）}

\entry{基本概念}{NCFET 是在\textbf{\textcolor{myred}{栅极引入了高介电常数的铁电栅介质}}的晶体管。}

\imgleft[0.3]{images/image-2026-06-14-16-14-41.png}{

\concepttable{
    \conrow{普通 NCFET 结构}{从上到下依次为：金属栅极（Metal）$\rightarrow$ 铁电层（Ferroelectric）$\rightarrow$ 高 k 绝缘层（high-k）$\rightarrow$ 衬底沟道。}
    \conrow{MFMIS 结构}{金属/铁电/金属/高 k 叠层结构：\textbf{\textcolor{myred}{在铁电层与高 k 绝缘层之间，多插入了一层金属层（Metal）}}。}
}

}

\subsubsection{铁电材料定义与自发极化}

\entry{引入目的}{引入铁电体是为了实现\textbf{\textcolor{myred}{负电容}}。}

\concepttable{
    \conrow{普通电介质}{只有在外加电场 $\text{E}$ 作用下，电子云被拉长，才会感应出电偶极子。}
    \conrow{铁电体}{即使在无电场情况下，内部也具有\textbf{\textcolor{myred}{自发极化（Spontaneous Polarization）}}的电介质，且极化方向可随外加电场的改变而改变。}
}

\textbf{\hlblue{温度依赖性（居里温度与晶格结构）}}

\entry{居里温度}{当温度超过临界值时，铁电材料的自发极化现象消失。自发极化不是永恒存在的。}

\subsubsection{NCFET 工作机制（负电容效应）与优势}

\entry{概述}{NCFET 利用负电容效应，打破前文所述的 $60\text{ mV/dec}$ 限制。}

\textbf{\hlblue{铁电材料特性}}

\imgleft[0.15]{images/image-2026-06-14-16-23-11.png}{
\concepttable{
    \conrow{电滞曲线特征}{在其电滞曲线中，存在一个\textbf{\textcolor{myred}{极化强度随电场增加而减小的区域}}，在此区域内，铁电电容 $C_{\text{FE}}$ 表现为负值。}
    \conrow{等效电路}{栅极电压 $V_{\text{G}}$ 作用于铁电电容 $C_{\text{FE}}$ 与基体 CMOS 电容 $C_{\text{MOS}}$（由 $C_{\text{ox}}$ 和 $C_{\text{s}}$ 串联组成）构成的串联网络。}
}
}

\textbf{\hlblue{突破极限数学原理}}

\concepttable{
    \conrow{传统器件}{亚阈值摆幅因子项为 $1 + \frac{C_{\text{s}}}{C_{\text{ox}}} \ge 1$，故 $SS \ge 60\text{ mV/dec}$。}
    \conrow{NCFET}{引入负电容 $C_{\text{FE}}$ 后，等效公式变为：$$1 + C_{\text{s}}\left(\frac{1}{C_{\text{ox}}} + \frac{1}{C_{\text{FE}}}\right) < 1$$}
    \conrow{核心结论}{因为 $C_{\text{FE}}$ 是负数，使得括号内相减，\textbf{\textcolor{myred}{整个系数小于 1}}，从而推导出 $SS < 60\text{ mV/dec}$。}
}

\subsubsection{电容匹配}

\entry{电容匹配}{负电容虽能带来超低 $SS$，但须满足严格的电容匹配条件。下面对比三种匹配状态（重点考核：量化条件、曲线形态、后果）。}

\imgleft[0.3]{images/image-2026-06-14-16-26-55.png}{

\concepttable{
    \conrow{电容失配（曲线 a）}{\textbf{条件：}$-C_{\text{FE}} < C_{\text{MOS}}$。局部表现出陡峭亚阈值摆幅，但\textbf{\textcolor{myred}{负电容效应不稳定。存在回滞现象}}，正反向扫描曲线不重合。}
    \conrow{电容完美匹配（曲线 b）$\star$}{\textbf{条件：}$C_{\text{MOS}} < -C_{\text{FE}} < C_{\text{ox}}$。负电容效应稳定。\textbf{\textcolor{myred}{$-C_{\text{FE}}$ 越接近 $C_{\text{MOS}}$，电容匹配程度越高。无回滞现象。}}}
    \conrow{电容匹配但增益低（曲线 c）}{\textbf{条件：}$-C_{\text{FE}} > C_{\text{ox}}$。负电容效应稳定，无回滞；但因 $-C_{\text{FE}}$ 绝对值过大，倒数对整体反馈贡献减小。性能相对传统 MOSFET 无太大提升。}
}
}

\subsubsection{铁电材料发展}

\textbf{\hlblue{传统钙钛矿材料}}

\concepttable{
    \conrow{PZT}{$\text{Pb(Zr,Tis)O}_3$（钛酸铅锆）}
    \conrow{BFO}{$\text{BiFeO}_3$（铁酸铋）}
    \conrow{SBT}{$\text{SrBi}_2\text{Ta}_2\text{O}_9$（钽酸铋锶）}
}

\concepttable{
    \conrow{优点}{能够实现较小的亚阈值摆幅（SS）；居里温度高，热稳定性高}
    \conrow{缺点}{高矫顽场导致回滞较大；开启电压（$V_{\text{on}}$）过大；\textbf{\textcolor{myred}{与 CMOS 工艺不兼容}}，无法直接用于大规模集成电路制造}
}

\textbf{实际器件数据}

\concepttable{
    \conrow{PZT}{$\text{SS}_{\text{min}} \sim 13\text{ mV/dec}$（SS 小），Hysteresis $\sim 12\text{V}$（证实回滞极大）}
    \conrow{BFO}{$\text{SS}_{\text{min}} = 8.5\text{ mV/dec}$，Hysteresis $\sim 4\text{V}$，$V_{\text{on}} = 4\text{V}$（证实开启电压过大，无法满足低功耗要求）}
}

\textbf{\hlblue{有机聚合物材料}}

由于钙钛矿材料回滞大且不兼容 CMOS，研究转向有机聚合物。典型材料：$\text{P(VDF-TrFE)}$（聚偏氟乙烯-三氟乙烯）。

\concepttable{
    \conrow{优点}{可以实现较小的亚阈值摆幅（SS）}
    \conrow{缺点}{在空气中器件性能会发生退化（\textbf{\textcolor{myred}{化学稳定性差}}）；依然存在工艺不兼容的问题}
}

\textbf{实际器件数据}

\concepttable{
    \conrow{串联铁电 NCFET}{原 MOSFET $\text{SS} = 92\text{ mV/dec} \rightarrow$ 串联 PVDF 后 $\text{SS}$ 降低至 $25\text{ mV/dec}$}
    \conrow{旋涂铁电 NCFET}{原 MOSFET $\text{SS} > 100\text{ mV/dec} \rightarrow$ 旋涂 PVDF 后 $\text{SS}$ 降低至 $18\text{ mV/dec}$}
}

\textbf{\hlblue{HfO2掺杂材料}}

为解决前两类材料共同的致命痛点——工艺不兼容，引入了当前主流的 $\text{HfO}_2$（二氧化铪）掺杂体系。常见掺杂元素：$\text{Zr}$、$\text{Al}$、$\text{Si}$ 等。

\concepttable{
    \conrow{优点}{具有极优的工艺兼容性（因为 $\text{HfO}_2$ 本身就是高 k 栅介质工艺的标准材料）；回滞小}
    \conrow{缺点}{亚阈值摆幅（SS）的\textbf{\textcolor{myred}{优化效果不够明显}}；工艺\textbf{\textcolor{myred}{涨落较大}}（一致性控制难）}
}

\textbf{工艺步骤：}

\concepttable{
    \conrow{Step 1}{在衬底（Substrate）上通过 ALD（原子层沉积）沉积 $\text{HfO}_2$ 并进行循环掺杂（cycles doping）}
    \conrow{Step 2}{通过 CVD（化学气相沉积）沉积金属电极（$\text{TiN}$ 或 $\text{TaN}$）}
    \conrow{Step 3}{形成最终的栅叠层结构}
}

\entry{前提}{$\text{HfO}_2$ \textbf{\textcolor{myred}{本身不具备铁电性}}，必须通过掺杂、引入应力以及退火（三者缺一不可）来实现。}

\entry{相变过程}{上述条件促使材料发生再结晶，晶格结构由不具铁电性的正方晶系（Tetragonal）转变为具备自发极化 $P$ 的斜方晶系（Orthorhombic），从而在 $\text{HfO}_2$ 中成功诱导出铁电性。}

\textbf{$\text{HfO}_2$ 掺杂的 SS 限制与 3D FinFET 架构}

\entry{结论}{虽然 $\text{HfO}_2$ 解决了兼容性，但其亚阈值摆幅优化不明显。\textbf{\textcolor{myred}{该器件最低 $\text{SS}_{\text{min}} = 55\text{ mV/dec}$}}}

\imgleft[0.3]{images/image-2026-06-14-17-12-44.png}{
\textbf{SS 随 $\bm{V_{\text{G}}}$ 变化曲线对比}

\concepttable{
    \conrow{黑色曲线（Gate 外栅极）}{未有效利用内部负电容效应时，基体 FinFET 最低 $\text{SS}_{\text{min}} = 87\text{ mV/dec}$。}
    \conrow{红色曲线（Internal gate 内栅极）}{引入负电容效应后，下探至 $\text{SS}_{\text{min}} = 55\text{ mV/dec}$。}
}
}

\subsection{隧穿场效应晶体管（TFET）}

\subsubsection{基本结构}

\entry{核心结构}{栅控反偏 P-I-N 结（源区-沟道区-漏区分别构成 P-I-N 结构，中间的 i 为本征或低掺杂沟道）。}

\img{images/image-2026-06-14-18-51-12.png}

\concepttable{
    \conrow{N型TFET}{源端为 P+型 掺杂，漏端为 N+型 掺杂。（工作条件：$V_{\text{GS}} > 0, V_{\text{DS}} > 0$）}
    \conrow{P型TFET}{源端为 N+型 掺杂，漏端为 P+型 掺杂。（工作条件：$V_{\text{GS}} < 0, V_{\text{DS}} < 0$）}
}

\entry{注}{TFET 与 MOSFET 在器件结构方面主要的区别是：\textbf{\textcolor{myred}{源漏区掺杂类型相反}}。传统 MOSFET 的源漏掺杂类型相同（如同为 N+），而 TFET 的源漏掺杂类型必然相反。}

\subsubsection{开关机制}

\entry{关态下具有超低漏电}{
    当 $V_{\text{GS}} = 0\text{V}$ 时，器件处于关态。源端与沟道之间的势垒极宽，电子无法发生隧穿。$I_{\text{OFF}}$ 本质是反偏 P-I-N 结的\textbf{\textcolor{myred}{反向漏电流}}。

    \concepttable{
        \conrow{示例数据}{硅基 TFET 关态电流约为 $10^{-14}\text{ A/}\mu\text{m}$。}
        \conrow{结论}{\textbf{\textcolor{myred}{TFET 的关态漏电流远小于传统 MOSFET}}，对应控制关态泄漏电流的目标。}
    }
}

\imgleft[0.3]{images/image-2026-06-14-19-11-14.png}{
\entry{导通机制 —— 开态下的带带隧穿（BTBT）}{
    开态物理过程：

    \concepttable{
        \conrow{Step 1}{栅压 $V_{\text{GS}}$ 增大，电场使沟道表面能带向下移动。}
        \conrow{Step 2}{源区和沟道表面之间形成较小的势垒宽度。}
        \conrow{Step 3}{当 $E_{\text{C}}$（沟道导带底）低于 $E_{\text{V}}$（源端价带顶）时，开启能带窗口。}
        \conrow{Step 4}{源端价带电子通过带带隧穿（BTBT）进入沟道导带，形成隧穿电流。}
    }
}
}

\subsubsection{SS 优化原理}

\entry{原理}{传统 MOSFET 通过栅压控制势垒\underline{高度}；TFET 通过栅压控制隧穿势垒\underline{宽度}。}

\entry{WKB 近似与 SS 极限突破}{

    \concepttable{
        \conrow{隧穿概率与电流（WKB 近似）}{$$J_{\text{d}} \propto T_{\text{WKB}} \approx \exp\left( -\frac{4\sqrt{2m_n^*}E_{\text{g}}^{3/2}}{3q\hbar\xi} \right)$$}
        \conrow{决定电流的关键物理量}{$m_n^*$（电子有效质量）、$E_{\text{g}}$（半导体禁带宽度）、$\xi$（隧穿结电场）}
    }

    \textbf{\textcolor{myred}{结论}}：亚阈值斜率 SS 表达式：
    $$SS = \ln 10 \cdot \left( \frac{1}{V_{\text{eff}}} \cdot \frac{\partial V_{\text{eff}}}{\partial V_{\text{GS}}} + \frac{\xi + b}{\xi^2} \cdot \frac{\partial \xi}{\partial V_{\text{GS}}} \right)^{-1}$$
    
    从公式可见，\textbf{\textcolor{myred}{SS 与温度项 $kT/q$ 无关}}。因此 TFET 理论上可突破 $60\text{ mV/dec}$ 极限，实现超陡亚阈值摆幅，从而允许 $V_{\text{DD}}$ 进一步降低。
}

\subsubsection{TFET 缺陷}

\entry{概述}{TFET 虽具有超低关态漏电并能突破 $60\text{ mV/dec}$ 极限，但以下两大核心问题制约其量产。}

\concepttable{
    \conrow{导通电流（$I_{\text{ON}}$）低}{对于硅基 TFET，由于单质硅（Si）的禁带宽度（$E_{\text{g}}$）较大，导致载流子的带带隧穿几率较低。硅基 TFET 的开启电流低于硅基 MOSFET 2\textasciitilde{}3 个数量级，\textbf{\textcolor{myred}{器件驱动能力严重受限}}。}
    \conrow{工艺难度高}{TFET \textbf{\textcolor{myred}{对源区和漏区的掺杂浓度梯度非常敏感}}。在实际热工艺中，源漏区的重掺杂会向沟道区扩散，从而破坏了陡峭的 PN 结特性，最终影响隧穿处的势垒宽度。使掺杂的随机波动性会导致 TFET 的开关态电流出现 2 个数量级的剧烈波动。}
}

\textbf{\hlblue{性能改进：新材料设计}}

\entry{准则}{选择\textbf{\textcolor{myred}{小禁带宽度（small $E_{\text{g}}$）}}和\textbf{\textcolor{myred}{小有效质量（small $m^*$）}}的材料。}

\entry{物理依据}{根据 WKB 公式，减小 $m^*$ 和 $E_{\text{g}}$ 使分式值减小，隧穿电流呈指数级放大。}

\textbf{材料参数对比}

\compTable[L]{4}{
    \compheader{\textbf{材料} & \textbf{$E_{\text{g}}$ (eV)} & \textbf{$m^*$ ($m_0$)} & \textbf{特点}}
    \comprow{$\text{Si}$ (传统) & 1.12 & 0.19 & 隧穿几率低，$I_{\text{ON}}$ 受限}
    \comprow{$\text{Ge}$ & 0.66 & 0.082 & $E_{\text{g}}$ 和 $m^*$ 均显著降低}
    \comprow{$\text{InAs}$ & 0.36 & 0.026 & III-V 族，禁带宽度骤降}
    \comprow{$\text{InSb}$ & 0.14 & 0.0135 & $m^*$ 减小一个数量级以上}
}

\textbf{P 型/N 型 TFET 的最佳材料体系：}

\concepttable{
    \conrow{p-TFETs 改进材料}{选用 $\text{Ge/GeSn}$（锗/锗锡）体系。提供小的 $E_{\text{g}}$ 和 $m^*$，等效减小隧穿势垒宽度。}
    \conrow{n-TFETs 改进材料}{选用 III-V 族化合物材料（典型代表：$\text{InGaAs}$、$\text{InAs}$、$\text{GaSb}$）。提供小的 $E_{\text{g}}$ 和 $m^*$，实现高效直接带带隧穿（direct BTBT）。}
}

\textbf{\hlblue{性能改进：新结构设计（提高隧穿面积）}}

\entry{概述}{除更换材料外，改变器件的几何结构也能提升电流}

\img{images/image-2026-06-14-19-43-44.png}

\concepttable{
    \conrow{\textbf{\textcolor{myred}{“点”隧穿}}}{电子隧穿方向（横向）与栅极电场方向（纵向）\textbf{\textcolor{myred}{相互垂直}}。隧穿仅发生在源区与沟道交界面的局部区域，面积很小，因而限制了 $I_{\text{ON}}$。}
    \conrow{\textbf{\textcolor{myred}{“线”隧穿}}}{通过结构设计，使栅极直接覆盖在部分源区之上。电子隧穿方向（纵向）与栅极电场方向\textbf{\textcolor{myred}{平行}}。}
}

\entry{结论}{“线”隧穿通过大幅提高隧穿面积，在相同电压下释放更多隧穿载流子，\textbf{\textcolor{myred}{有效解决硅基 TFET 导通电流低的问题}}。}

\textbf{\hlblue{性能改进——新结构设计（增强栅控能力）}}

\entry{概述}{结构改进的第二条路径：利用先进的多栅/环绕栅架构强化对隧穿结电场的控制。}

\concepttable{
    \conrow{结构}{双栅（DG，Double Gate）、纳米线（NWs，GAA 全环绕栅）、包裹栅（Wrapped gate）}
}
