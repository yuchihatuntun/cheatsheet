\subsubsection{时域混叠}

首先假设一个情形：

\concepttable{
    \conrow{连续信号建模}{$x_a(t) = A\cos(2\pi F_0 t + \theta)$，$F_0$ 为模拟频率}
    \conrow{采样离散化}{采样频率 $F_s = 1/T$，采样周期 $T$，$t = nT = \frac{n}{F_s}$}
    \conrow{离散时间信号}{代入采样，$x(n) = A\cos\left(2\pi \frac{F_0}{F_s} n + \theta\right) = A\cos(2\pi f_0 n + \theta)$}
    \conrow{数字频率定义}{$f_0 = \frac{F_0}{F_s}$}
}

% --- 时域混叠配图笔记（精简学术版）---
\concepttable{
    \conrow{混叠信号构造}{$F_k = F_0 + kF_s$，$k \in \mathbb{N}$}
    \conrow{采样表达}{$x'_a(t) = A\cos(2\pi F_k t + \theta)$}
    \conrow{离散化推导}{$x'(n) = A\cos\left(2\pi \frac{F_0 + kF_s}{F_s} n + \theta\right)$}
    \conrow{恒等变换}{$x'(n) = A\cos\left(2\pi \frac{F_0}{F_s} n + \theta\right) = x(n)$}
    \conrow{混叠本质}{\textbf{\textcolor{myred}{不同模拟频率信号采样后可产生完全相同的离散序列}}}
}

\subsubsection{从模拟信号理解采样}

我们已知冲激函数的抽样性：

\concepttable{
    \conrow{采样脉冲序列}{$p_\delta(t)=\sum_{n=-\infty}^{\infty}\delta(t-nT)$（公式1-45）}
    \conrow{采样运算}{$\hat{x}_a(t)=x_a(t)\,p_\delta(t)=\sum_{n=-\infty}^{\infty}x_a(t)\delta(t-nT)$}
    \conrow{抽样筛选性质}{由冲激函数的筛选特性得$\hat{x}_a(t)=\sum_{n=-\infty}^{\infty}x_a(nT)\delta(t-nT)$}
    \conrow{物理含义}{采样将连续信号在$t=nT$处的幅值以脉冲加权表示，形成便于离散域处理的样值序列}
}
