\section{第一章\ 采样定理}

\textbf{\hlyellow{模拟信号数字化处理方法}}
\textbf{\hlgreen{采样与量化}}

信号依据\textbf{时间轴}与\textbf{幅值轴}的连续/离散属性，分为四类：
\vspace{-0.2cm}
\begin{multicols}{2}
\begin{minipage}{\columnwidth}
\concepttable{
    \conrow{模拟信号}{时间连续、取值连续}
    \conrow{离散时间}{时间离散、取值连续}
}
\end{minipage}
\begin{minipage}{\columnwidth}
\concepttable{
    \conrow{量化阶梯}{时间连续、取值离散}
    \conrow{数字信号}{时间离散、取值离散}

\vspace{-0.2cm}
\textbf{\hlblue{信号处理流程}}

$
x_a(t)
\xrightarrow[\text{限最高频率，防频谱混叠}]{\text{预滤波}}
\xrightarrow[\text{采样量化}]{\text{A/D}}
x(n)
\xrightarrow[\text{核心}]{\text{DSP}}
y(n)
\xrightarrow[\text{恢复阶梯模拟信号}]{\text{D/A}}
\xrightarrow[\text{滤除高频}]{\text{平滑滤波}}
y_a(t)
$

\textbf{\hlblue{时域混叠}}
\entry{定义}{频率为$F_0$ 与 $F_0+kF_s$ 的周期信号经同一 $F_s$ 采样后所得序列完全相同}

\textbf{数字角频率 $w$：}$w = \dfrac{2\pi f}{f_s} = 2\pi f_0$ \ \ \textbf{模拟角频率 $\Omega$：}$\Omega = 2\pi f = w f_s$\ \ \textbf{映射：}$\displaystyle w = \Omega T_s,\quad \Omega = \frac{w}{T_s}$

\textbf{\hlblue{理想采样信号}}
$\hat{x}_a(t)$ 是强度为 $x_a(nT)$ 的冲激串，\textbf{\textcolor{myred}{仍是连续时间信号}}

\concepttable{
    \conrow{冲激串函数}{$P_\delta(t) = \sum_{n=-\infty}^{\infty} \delta(t - nT)$}
    \conrow{采样信号}{$\hat{x}_a(t) = x_a(t) \cdot P_\delta(t) = \sum_{n=-\infty}^{\infty} x_a(nT)\,\delta(t - nT)$}
    \conrow{频域形式}{$\hat{X}_a(j\Omega) = \frac{1}{T} \sum_{k=-\infty}^{\infty} X_a(j(\Omega - k\Omega_s))$}
}

\textbf{\textcolor{myred}{采样后频谱是原频谱以 $\Omega_s$ 为周期的周期延拓，幅度缩放 $1/T$}}。$k=0$ 为基带，$k\neq 0$ 为镜像谱

\textbf{\hlblue{奈奎斯特采样定理}}

\concepttable{
    \conrow{过采样（正常情况）}{$\Omega_c < \Omega_s/2$，镜像间有隔离带，无混叠，低通滤波（截止 $\Omega_s/2$）可无失真重构}
    \conrow{欠采样}{$\Omega_c > \Omega_s/2$，\textbf{\textcolor{myred}{高频镜像侵入基带，谱线代数叠加}}，频谱结构不可逆破坏}
}

\textbf{\hlblue{时域-频域对偶}}
\textbf{\textcolor{myred}{连续$\leftrightarrow$非周期，离散$\leftrightarrow$周期。}}

\textbf{\hlyellow{有限字长效应}}
在数字系统中，由于寄存器、运算单元、存储单元的\textbf{位数（字长）有限，无法精确表示无限精度的数值与运算}，从而引入量化误差、导致系统性能下降的现象，统称为有限字长效应。

\section{第二章\ 时域离散信号及系统频域分析}

\textbf{\hlyellow{时域离散信号的傅里叶变换}}
\textbf{\hlgreen{DTFT}}

\entry{定义式}{
    \textbf{\textcolor{myred}{$X(e^{j\omega}) = \text{DTFT}[x(n)] = \sum_{n=-\infty}^{\infty} x(n)e^{-j\omega n}$}}}
\entry{充分条件（绝对可和）}{$\displaystyle\sum_{n=-\infty}^{\infty} |x(n)| < \infty$，和 Z 变换收敛条件对应}

\textbf{\hlgreen{IDTFT}}
\textbf{定义式：}\textbf{\textcolor{myred}{$x(n) = \text{IDTFT}[X(e^{j\omega})] = \frac{1}{2\pi}\int_{-\pi}^{\pi} X(e^{j\omega})e^{j\omega n}d\omega$}}

\textbf{\hlgreen{DTFT 的性质}}
主要性质和 Z 变换相同。

\textbf{\hlgreen{DFS}}
设 $x(n)$ 为离散周期序列，周期为 $N$。

\entry{离散傅里叶级数系数}{$\tilde{X}(k)=\sum_{n=0}^{N-1} \tilde{x}(n) e^{-j \frac{2 \pi}{N} k n} \quad 0<k<N-1$}

\entry{DFS 形式}{$\tilde{x}(n)=\frac{1}{N} \sum_{k=0}^{N-1} \tilde{X}(k) e^{j \frac{2 \pi}{N} k n} \quad 0<n<N-1$}

\textbf{\hlgreen{广义 DTFT}}

\entry{定义式}{$X(e^{j\omega}) = \mathrm{FT}[\tilde{x}(n)] = \frac{2\pi}{N}\sum_{k=-\infty}^{\infty}\tilde{X}(k)\delta\left(\omega-\frac{2\pi}{N}k\right)$}

\textbf{\hlgreen{常用 DTFT 对}}
$\cos \omega_0 n \leftrightarrow \pi \displaystyle\sum_{l=-\infty}^{\infty} [\delta(\omega - \omega_0 - 2\pi l) + \delta(\omega + \omega_0 - 2\pi l)]$

$\sin \omega_0 n \leftrightarrow -j\pi \displaystyle\sum_{l=-\infty}^{\infty} [\delta(\omega - \omega_0 - 2\pi l) - \delta(\omega + \omega_0 - 2\pi l)]$

$x(n)=1$ $\leftrightarrow$ $2\pi \displaystyle\sum_{k=-\infty}^{\infty} \delta(\omega - 2\pi k)$

$R_N(n)$ $\leftrightarrow$ $\frac{\sin(\omega N/2)}{\sin(\omega/2)} e^{-j\omega(N-1)/2}$

$\delta(n) \leftrightarrow 1$ \ \ \ \  $a^n u(n) \leftrightarrow \frac{1}{1 - a e^{-j\omega}}$

\end{minipage}

\subsubsection{Z 变换}

Z 变换存在的条件是\textbf{级数收敛，要求级数绝对可和}，即$
\sum_{n=-\infty}^{\infty} \left| x(n) z^{-n} \right| < \infty
$。


\textbf{\textcolor{myred}{\textbf{双边 Z 变换}$X(z) \stackrel{\text{def}}{=} \displaystyle\sum_{n=-\infty}^{\infty} x(n) z^{-n}$\ \ 
\textbf{单边 Z 变换}$X(z) \stackrel{\text{def}}{=} \displaystyle\sum_{n=0}^{\infty} x(n) z^{-n}$(适用于因果序列)}}


\textbf{\hlgreen{Z 变换的收敛域}}

\textbf{\textcolor{myred}{收敛域必为以原点为中心的环状区域}}。

\textbf{有限长序列收敛域：}

仅在 $n_1\le n\le n_2$ 内非零，$X(z)=\sum_{n=n_1}^{n_2}x(n)z^{-n}$。有限项求和，每项有界则收敛。

    \comprow{$n_1\ge0$ & 仅负幂次（或常数） & $0<|z|\le\infty$ & $z=0$}
    \comprow{$n_2\le0$ & 仅正幂次（或常数） & $0\le|z|<\infty$ & $z=\infty$}
}

\textbf{右序列收敛域：}

$n \ge n_1$ 非全零，$n < n_1$ 全零。$R_{x-}$ 为最大极点半径。

$R_{x-} < |z| < R_{x+}$，即环状区域，\textbf{\textcolor{myred}{若 $R_{x-} \ge R_{x+}$，两区域无交集，$z$ 变换不存在}}。

\textbf{\hlgreen{逆Z变换（部分分式展开法）}}
\textbf{整体思路}：把大的有理分式展开为多个简单的分式，每个分式通过查表得到对应的逆变换。

\textbf{左序列处理：}如果分离出来的分式形式为$\frac{bz}{z+a}$，且由收敛域判断其为左序列，则\textbf{\textcolor{myred}{其逆变换为$-b(-a)^{n}u(-n-1)$}}，如果遇到不规则的$u(f(n))$，则为\textbf{\textcolor{myred}{$u(-f(n)-1)$的方式处理为左序列}}。

\textbf{\hlyellow{系统函数与频率特性}}
DTFT是z变换在单位圆上的特例，因此\textbf{\textcolor{myred}{系统稳定的条件是 $H(z)$ 的所有极点都在单位圆内}}，此时DTFT存在。

\textbf{LTI 离散系统的三种表示方式}：冲激响应，差分方程，系统函数

\concepttable{
    \conrow{因果性}{$n < 0$ 时 $h(n) = 0$，即 H(z)的收敛域表示为 $R_{x^{-}} < |z| \leq \infty$}
    \conrow{稳定性}{$H(z)$ 收敛域包含单位圆（即 DTFT 的存在条件）}
    \conrow{因果稳定性}{$R_{x^{-}} < |z| \leq \infty, \quad 0 < R_{x^{-}} < 1$，即所有极点均在单位圆内}
}

\subsubsection{零极点分布与频率特性}
$H(z)= K_1 z^{N-M} \frac{\prod_{m=1}^M (z - c_m)}{\prod_{k=1}^N (z - d_k)}$，其中$c_m$为零点，$d_k$为极点。

系统稳定时，可化简为：$H(e^{j\omega}) = K e^{j(N-M)\omega} \frac{\prod_{m=1}^M (e^{j\omega} - c_m)}{\prod_{k=1}^N (e^{j\omega} - d_k)} = |H(e^{j\omega})| e^{j \arg[H(e^{j\omega})]}$

\textbf{幅频函数（$|H(e^{j\omega})|$）}：等于增益模长（K），乘以单位圆当前频点 $e^{j\omega}$ 到所有零点的向量长度之积，再除以该频点到所有极点的向量长度之积。

\textbf{注：}

1. \textbf{\textcolor{myred}{位于原点处的零点或极点}}对系统的幅频特性完全没有影响；但它们会在相频函数中引入 $(N-M)\omega$ 的线性相位移。

2. \textbf{\textcolor{myred}{实数系统的所有零点和极点，要么严格位于实轴上，要么必须成对以共轭复数的形式出现}}

3. 若分子多项式 $P(z)$ 与分母多项式 $Q(z)$ 包含完全相同的根，则该零极点在代数上可以相互抵消，从而降低系统的实际阶数

\section{第三章\ 离散傅里叶变换}

\subsubsection{定义}

\entry{定义}{对于一个长度为 $M$ 的有限长序列 $x(n)$，其 $N$ 点 DFT 定义为：

\textbf{\textcolor{myred}{$X(k) = \text{DFT}[x(n)]_N = \sum_{n=0}^{N-1} x(n) e^{-j\frac{2\pi}{N}kn}, \quad k = 0,1,\cdots,N-1$}}}

\textbf{注意：}

\concepttable{
    \conrow{变换区间长度 $N$}{必须满足 $N \ge M$（即变换点数不能小于序列实际长度）}
    \conrow{简化形式}{引入旋转因子$W_N = e^{-j\frac{2\pi}{N}}$，\textbf{\textcolor{myred}{$X(k) = \sum_{n=0}^{N-1} x(n) W_N^{kn}$}}}
}

\textbf{逆变换：}\textbf{\textcolor{myred}{$x(n) = \text{IDFT}[X(k)]_N = \frac{1}{N} \sum_{k=0}^{N-1} X(k) W_N^{-kn}, \quad n = 0,1,\cdots,N-1$}}

\concepttable{
    \conrow{周期性\&共轭}{$W_N^{kn}=W_N^{kn+mN}$;\quad $W_N^{-kn}=(W_N^{kn})^*=W_N^{N-kn}$}
    \conrow{半周期\&1/4}{\textbf{\textcolor{myred}{$W_N^{N/2}=-1$}}\textbf{\textcolor{myred}{$W_N^{N/4}=-j,W_N^{-N/4}=j$}}}
}

\subsubsection{与其他变换的关系}

\textbf{\hlblue{DTFT：}}在 $N \ge M$ 的前提下，序列的 $N$ 点 DFT 是其 DTFT 在\textbf{\textcolor{myred}{频域区间 $[0, 2\pi)$ 上的 $N$ 点}}等间隔采样，采样间隔（即基频/频率分辨率）为 $\Delta \omega = \frac{2\pi}{N}$，即：$X(k) = \left. X(e^{j\omega}) \right|_{\omega = \frac{2\pi}{N}k}$

\textbf{\hlblue{ZT：}}同时也是其 Z 变换\textbf{\textcolor{myred}{在 $Z$ 平面单位圆上的 $N$ 点}}等间隔采样，即：$X(k) = \left. X(z) \right| z = e^{j\frac{2\pi}{N}k}$。


\textbf{\hlblue{DFS：}}DFT 的处理对象 $x_N(n)$ 是 DFS 的处理对象 $\tilde{x}_N(n)$ 的主值序列。DFT 的计算结果 $X(k)$ 是 DFS 计算结果 $\tilde{X}(k)$ 的主值序列，\textbf{即DFS 是 DFT 的周期性延拓}。

\subsubsection{时域混叠}

时域混叠（$N < M$）： 当设定的变换点数 $N$ 小于信号实际长度 $M$ 时，时域以 $N$ 为周期进行延拓会导致\textbf{\textcolor{myred}{前后周期的波形在时域发生重叠}}（即时域混叠）。

\subsubsection{离散傅里叶变换的性质}

\textbf{线性：}$y(n) = ax_1(n) + bx_2(n) \rightarrow Y(k) = aX_1(k) + bX_2(k)$

\textbf{时域循环移位：}$\text{DFT}[x((n+m))_N R_N(n)] = W_N^{-km} X(k)$

\textbf{频域循环移位：}$\text{DFT}[W_N^{mn} x(n)] = X((k+m))_N R_N(k)$

\textbf{循环卷积计算：}
$y_c(n) = \left[ \sum_{m=0}^{L-1} h(m) x((n-m))_L \right] R_L(n) = h(n) \bigodot x(n)$

$\begin{bmatrix} y_c(0) \\ y_c(1) \\ y_c(2) \\ \vdots \\ y_c(L-1) \end{bmatrix} = \begin{bmatrix} x(0) & x(L-1) & x(L-2) & \cdots & x(1) \\ x(1) & x(0) & x(L-1) & \cdots & x(2) \\ x(2) & x(1) & x(0) & \cdots & x(3) \\ \vdots & \vdots & \vdots & & \vdots \\ x(L-1) & x(L-2) & x(L-3) & \cdots & x(0) \end{bmatrix} \begin{bmatrix} h(0) \\ h(1) \\ h(2) \\ \vdots \\ h(L-1) \end{bmatrix}$

\textbf{时域循环卷积定理：}
$y_c(n) = h(n) \bigodot x(n) \rightarrow Y_c(k) = \text{DFT}[y_c(n)]_L = H(k)X(k)$

\textbf{帕塞瓦尔定理：}
$\sum_{n=0}^{N-1} |x(n)|^2 = \frac{1}{N} \sum_{k=0}^{N-1} |X(k)|^2$

\textbf{\hlyellow{频域采样}}

「1」$x(n)$ 是任意序列，$X(e^{j\omega}) = DTFT[x(n)]$，对 $X(e^{j\omega})$ 等间隔采样得到
$X_N(k) = X(e^{j\omega}) \big|_{\omega=\frac{2\pi}{N}k} \quad k = 0 \cdots N-1$

则
$x_N(n) = IDFT[X_N(k)] = \sum_{n=-\infty}^{\infty} x(n+iN) R_N(n)$

「2」如果 $x(n)$ 的长度为 $M$，只有当频域采样点数 $N \geqslant M$ 时，$x_N(n) = x(n)$，否则
$\tilde{x}_N(n) = \sum_{n=-\infty}^{\infty} x(n+iN)$ 会发生时域混叠。

通过频率域采样得到频域离散序列 $x_N(k)$，再对 $x_N(k)$ 进行 IDFT 得到的序列 $x_N(n)$ 应是原序列 $x(n)$ 以采样点数 $N$ 为周期进行周期化后的主值区序列。

「3」采样点数 $N$ 大于等于序列的长度 $M$时，可以用频率采样得到的离散函数 $X(k)$ 恢复原序列的 Z 变换 $X(z)$，公式为$X(z) = \sum_{k=0}^{N-1} X(k) \varphi_k(z)$（$z$ 域内插公式）

\textbf{内插函数：}$\varphi_k(z) = \frac{1}{N} \frac{1-z^{-N}}{1-W_N^{-k} z^{-1}}$

\section{第五章\ IIR数字滤波器设计}

\textbf{\hlyellow{数字滤波器参数}}：越像方波越好（通、阻带平（阻带幅值接近 0），过渡带陡）

\imgleft[0.2]{images/image-2026-06-19-11-19-34.png}{
通带截止频率 $w_p$，阻带截止频率 $w_s$，3dB截止频率 $w_c$，有 $|H(e^{jw_c})| = \sqrt{2}/2$

\textbf{通带波纹} $\delta_1 = 1 - \sqrt{1 + \varepsilon^2}$，\textbf{阻带波纹} $\delta_2 = 1/A$。

\textbf{通带最大衰减} $\alpha_p = -20\lg|H(e^{jw_p})| = 10\lg(1 + \varepsilon^2) \text{ dB}$

\textbf{阻带最大衰减} $\alpha_s = -20\lg|H(e^{jw_s})| = 20\lg A \text{ dB}$
}

\textbf{\hlyellow{模拟滤波器设计}}

\textbf{总思路：}
$\text{读取四大指标} \rightarrow \text{计算相关参数} \rightarrow \text{计算滤波器阶数 N（注意高低通区别）} \rightarrow \text{求}G(p)\rightarrow \text{根据滤波器类型代入 p 的表达式得到}H_a(s)$

\textbf{\hlgreen{技术指标（归一化系统函数）}}
\textbf{损耗函数：}$\alpha(\Omega) = -20\lg|H_a(j\Omega)| = -10\lg|H_a(j\Omega)|^2 \text{ dB}$

$\varepsilon = \sqrt{10^{\alpha_p/10} - 1}$
\ \ \ $A = 10^{\alpha_s/10}$
\ \ \ \textbf{过渡比：}$k = \Omega_p / \Omega_s$
\ \ \ \textbf{偏离参数：}$k_1 = \frac{\varepsilon}{\sqrt{A^2 - 1}}$

\textbf{\hlgreen{巴特沃斯低通滤波器（幅频响应单调变化，无波纹）}}

\imgleft[0.26]{images/image-2026-06-19-11-37-43.png}{
\textbf{幅度平方函数（单调递减）：}$|H_a(j\Omega)|^2 = \frac{1}{1 + (\Omega/\Omega_c)^{2N}}$

\textbf{衰减函数：}$\alpha(\Omega) = -20\lg|H_a(j\Omega)| = 10\lg\left[1 + \left(\frac{\Omega}{\Omega_c}\right)^{2N}\right]$

\textbf{阶数计算：}$\boxed{N = \frac{lg(k_{sp})}{lg(\lambda_{sp})} = \frac{\lg\left(\sqrt{10^{\alpha_p/10} - 1} \Big/ \sqrt{10^{\alpha_s/10} - 1}\right)}{\lg(\Omega_p/\Omega_s)}}$

}

\textbf{注：}计算出的 $N$ 如果不是整数，必须\textbf{\textcolor{myred}{向上取整}}(因为阶数 $N$ 越大，过渡带越窄，滤波器性能越好)

\textbf{\hlyellow{模拟滤波器$\rightarrow$数字滤波器}}
\textbf{\hlgreen{脉冲响应不变法}}

\textbf{第二步：}先进行部分分式展开，即$H_a(s) = \sum_{i=0}^{N-1} \frac{A_i}{s - s_i}$，再极点映射（$s \to z$）：

\concepttable{
    \conrow{单阶极点}{$\displaystyle\frac{A_i}{s - s_i} \longrightarrow \frac{A_i T}{1 - e^{s_i T} z^{-1}}$（最常用）}
    \conrow{$2$ 重极点}{$\displaystyle\frac{A_i}{(s - s_i)^2} \longrightarrow \frac{A_i T^2 e^{s_i T} z^{-1}}{(1 - e^{s_i T} z^{-1})^2}$}
}

\textbf{\hlgreen{双线性变换法}}

\textbf{第一步：}我们得到系统函数$H_a(s)$之后，$s$ 替换为 $z^{-1}$ 分式：$\boxed{\displaystyle H(z) = H_a(s) \big|_{s=\frac{2}{T}\frac{1-z^{-1}}{1+z^{-1}}}}$，再化简为标准形式即可。
