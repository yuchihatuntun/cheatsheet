\section{第一章\ 采样定理}

\textbf{\hlyellow{模拟信号数字化处理方法}}
\textbf{\hlgreen{采样与量化}}

信号依据\textbf{时间轴}与\textbf{幅值轴}的连续/离散属性，分为四类：
\vspace{-0.2cm}

\begin{minipage}{\columnwidth}

\end{minipage}
\begin{minipage}{\columnwidth}
\concepttable{
    \conrow{量化阶梯}{时间连续、取值离散}
    \conrow{数字信号}{时间离散、取值离散}

\end{minipage}

\vspace{-0.2cm}
\textbf{\hlgreen{信号处理流程}}

$
x_a(t)
\xrightarrow[\text{限最高频率，防频谱混叠}]{\text{预滤波}}
\xrightarrow[\text{采样量化}]{\text{A/D}}
x(n)
\xrightarrow[\text{核心}]{\text{DSP}}
y(n)
\xrightarrow[\text{恢复阶梯模拟信号}]{\text{D/A}}
\xrightarrow[\text{滤除高频}]{\text{平滑滤波}}
y_a(t)
$

\textbf{\hlgreen{时域混叠}}
\entry{定义}{频率为$F_0$ 与 $F_0+kF_s$ 的周期信号经同一 $F_s$ 采样后所得序列完全相同}

\textbf{数字角频率 $w$：}$w = \dfrac{2\pi f}{f_s} = 2\pi f_0$ \ \ \textbf{模拟角频率 $\Omega$：}$\Omega = 2\pi f = w f_s$\ \ \textbf{映射：}$\displaystyle w = \Omega T_s,\quad \Omega = \frac{w}{T_s}$

\textbf{\hlgreen{理想采样信号}}
$\hat{x}_a(t)$ 是强度为 $x_a(nT)$ 的\textbf{\textcolor{myred}{冲激串}}，\textbf{\textcolor{myred}{仍是连续时间信号}}

\concepttable{
    \conrow{采样信号}{$\boxed{\hat{x}_a(t) = x_a(t) \cdot P_\delta(t) = \sum_{n=-\infty}^{\infty} x_a(nT)\,\delta(t - nT)}$}
    \conrow{频域形式}{$\hat{X}_a(j\Omega) = \frac{1}{T} \sum_{k=-\infty}^{\infty} X_a(j(\Omega - k\Omega_s))$}
}

\textbf{\textcolor{myred}{采样后频谱是原频谱以 $\Omega_s$ 为周期的周期延拓(周期复制)，幅度缩放 $1/T$}}。$k=0$ 为基带，$k\neq 0$ 为镜像谱

\textbf{\hlgreen{奈奎斯特采样定理}}
采样频率 $\Omega_s$ 必须大于等于信号最高频率 $\Omega_c$ 的两倍。

\textbf{\textcolor{myred}{AM信号为基带信号与载波相乘，频谱为基带频谱线性搬移至$\pm\Omega_c$处}}，故最高频率为载波频率与基带最高频率之和

\concepttable{
    \conrow{过采样（正常情况）}{$\Omega_c < \Omega_s/2$，镜像间有隔离带，无混叠，低通滤波（截止 $\Omega_s/2$）可无失真重构}
    \conrow{欠采样}{$\Omega_c > \Omega_s/2$，\textbf{\textcolor{myred}{高频镜像侵入基带，谱线代数叠加}}，频谱结构不可逆破坏}
}

\textbf{\hlgreen{卷积和计算（不进位乘法）}}
判断零位置：两序列起点位置之和等于卷积和序列起点位置。

\textbf{\hlgreen{时域-频域对偶}}
\textbf{\textcolor{myred}{连续$\leftrightarrow$非周期，离散$\leftrightarrow$周期。}}

\textbf{\hlyellow{有限字长效应}}
在数字系统中，由于寄存器、运算单元、存储单元的\textbf{位数（字长）有限，无法精确表示无限精度的数值与运算}，从而引入量化误差、导致系统性能下降的现象，统称为有限字长效应。

\section{第二章\ 时域离散信号及系统频域分析}

\textbf{\hlyellow{时域离散信号的傅里叶变换}}
\textbf{\hlgreen{DTFT}}

\entry{定义式}{
    $\boxed{X(e^{j\omega}) = \text{DTFT}[x(n)] = \sum_{n=-\infty}^{\infty} x(n)e^{-j\omega n}} \quad \displaystyle\sum_{n=-\infty}^{\infty} |x(n)| < \infty$}

\textbf{\hlgreen{IDTFT}}
\textbf{定义式：}$\boxed{x(n) = \text{IDTFT}[X(e^{j\omega})] = \frac{1}{2\pi}\int_{-\pi}^{\pi} X(e^{j\omega})e^{j\omega n}d\omega}$

\textbf{\hlgreen{DTFT 的性质}}
$x(2n) \stackrel{\mathcal{F}}{\longleftrightarrow} \frac{1}{2}\left[X\left(\mathrm{e}^{\mathrm{j}\frac{\omega}{2}}\right) + X\left(\mathrm{e}^{\mathrm{j}\left(\frac{\omega}{2}-\pi\right)}\right)\right]$

\textbf{时域信号-频域信号性质对应：}\textbf{\textcolor{myred}{实偶 $\rightarrow$ 实偶，实奇 $\rightarrow$ 虚奇，虚奇 $\rightarrow$ 实奇，虚偶 $\rightarrow$ 虚偶}}

\textbf{\textcolor{myred}{三角函数变换：}}$\cos\omega = \frac{e^{j\omega} + e^{-j\omega}}{2}, \quad \sin\omega = \frac{e^{j\omega} - e^{-j\omega}}{2j}$

\textbf{\textcolor{myred}{实因果序列}} $x(n)=x_e(n)+x_o(n)$，因果性使二者不独立，可由其一恢复：
\vspace{-0.2cm}
\[
\boxed{x(n)=\begin{cases}0,&n<0\\x_e(0),&n=0\\2x_e(n),&n>0\end{cases}\quad
x(n)=\begin{cases}0,&n<0\\x(0),&n=0\\2x_o(n),&n>0\end{cases}}
\]

\textbf{\hlgreen{DFS}}
设 $x(n)$ 为离散周期序列，周期为 $N$，则傅里叶系数$\tilde{X}(k)$以及级数形式：

$\boxed{\tilde{X}(k)=\sum_{n=0}^{N-1} \tilde{x}(n) e^{-j \frac{2 \pi}{N} k n}}$
$\boxed{\tilde{x}(n)=\frac{1}{N} \sum_{k=0}^{N-1} \tilde{X}(k) e^{j \frac{2 \pi}{N} k n} \quad 0<k,n<N-1}$

\textbf{\hlgreen{广义 DTFT}}
\entry{定义式}{$X(e^{j\omega}) = \mathrm{FT}[\tilde{x}(n)] = \frac{2\pi}{N}\sum_{k=-\infty}^{\infty}\tilde{X}(k)\delta\left(\omega-\frac{2\pi}{N}k\right)$}

\textbf{\hlgreen{Z 变换}}

\textbf{双边 Z 变换}$\boxed{X(z) \stackrel{\text{def}}{=} \displaystyle\sum_{n=-\infty}^{\infty} x(n) z^{-n}}$\ \ 
\textbf{单边 Z 变换}$\boxed{X(z) \stackrel{\text{def}}{=} \displaystyle\sum_{n=0}^{\infty} x(n) z^{-n}}$

\textbf{\hlgreen{Z 变换收敛域}}
\textbf{\textcolor{myred}{收敛域必为以原点为中心的环状区域}}。

仅在 $n_1\le n\le n_2$ 内非零，$X(z)=\sum_{n=n_1}^{n_2}x(n)z^{-n}$。有限项求和，每项有界则收敛。

\textbf{\hlgreen{逆Z变换（部分分式展开法）}}
\textbf{整体思路}：把大的有理分式展开为多个简单的分式，每个分式通过查表得到对应的逆变换。

\textbf{左序列处理：}如果分离出来的分式形式为$\frac{bz}{z+a}$，且由收敛域判断其为左序列，则\textbf{\textcolor{myred}{其逆变换为$-b(-a)^{n}u(-n-1)$}}，如果遇到不规则的$u(f(n))$，则为\textbf{\textcolor{myred}{$u(-f(n)-1)$的方式处理为左序列}}。

\textbf{\hlyellow{系统函数与频率特性}}
\textbf{\textcolor{myred}{系统稳定的条件是 $H(z)$ 的所有极点都在单位圆内}}，此时DTFT存在(单位圆特例)。

\textbf{LTI 离散系统的三种表示方式}：冲激响应，差分方程，系统函数

\textbf{\hlgreen{零极点分布与频率特性}}
$H(z)= K_1 z^{N-M} \frac{\prod_{m=1}^M (z - c_m)}{\prod_{k=1}^N (z - d_k)}$，其中$c_m$为零点，$d_k$为极点。

系统稳定时，可化简为：$H(e^{j\omega}) = K e^{j(N-M)\omega} \frac{\prod_{m=1}^M (e^{j\omega} - c_m)}{\prod_{k=1}^N (e^{j\omega} - d_k)} = |H(e^{j\omega})| e^{j \arg[H(e^{j\omega})]}$

\textbf{幅频函数（$|H(e^{j\omega})|$）}：等于增益模长（K），乘以单位圆当前频点 $e^{j\omega}$ 到所有零点的向量长度之积，再除以该频点到所有极点的向量长度之积。

\section{第三章\ 离散傅里叶变换}

\entry{定义}{对于一个长度为 $M$ 的有限长序列 $x(n)$，其 $N$ 点 DFT 定义为：
\vspace{-1cm}
$$\boxed{X(k) = \text{DFT}[x(n)]_N = \sum_{n=0}^{N-1} x(n) e^{-j\frac{2\pi}{N}kn}, \quad k = 0,1,\cdots,N-1}$$}
\vspace{-1cm}
\concepttable{
    \conrow{简化}{旋转因子$\boxed{W_N = e^{-j\frac{2\pi}{N}}}$，$\boxed{X(k) = \sum_{n=0}^{N-1} x(n) W_N^{kn}}$}
}

\textbf{逆变换：}$\boxed{x(n) = \text{IDFT}[X(k)]_N = \frac{1}{N} \sum_{k=0}^{N-1} X(k) W_N^{-kn}, \quad n = 0,1,\cdots,N-1}$

\concepttable{
    \conrow{周期性\&共轭}{$W_N^{kn}=W_N^{kn+mN}$;\quad $W_N^{-kn}=(W_N^{kn})^*=W_N^{N-kn}$}
    \conrow{半周期\&1/4}{\textbf{\textcolor{myred}{$W_N^{N/2}=-1$}}\textbf{\textcolor{myred}{$W_N^{N/4}=-j,W_N^{-N/4}=j$}}}
}

在 $N \ge M$ 的前提下，序列的 $N$ 点 DFT 是其 DTFT 在\textbf{\textcolor{myred}{频域区间 $[0, 2\pi)$ 上的 $N$ 点}}等间隔采样，采样间隔（即基频/频率分辨率）为 $\Delta \omega = \frac{2\pi}{N}$，即：$X(k) = \left. X(e^{j\omega}) \right|_{\omega = \frac{2\pi}{N}k}$

FT 的处理对象 $x_N(n)$ 是 DFS 的处理对象 $\tilde{x}_N(n)$ 的主值序列。DFT 的计算结果 $X(k)$ 是 DFS 计算结果 $\tilde{X}(k)$ 的主值序列，\textbf{即DFS 是 DFT 的周期性延拓}。

\textbf{\hlgreen{时域混叠（$N < M$）}}： 当设定的变换点数 $N$ 小于信号实际长度 $M$ 时，时域以 $N$ 为周期进行延拓会导致\textbf{\textcolor{myred}{前后周期的波形在时域发生重叠}}（即时域混叠）。

\textbf{循环卷积与线性卷积的关系：}
设 $x(n)$ 长度为 $L_x$，$y(n)$ 长度为 $L_y$，DFT 点数为 $N$。线性卷积 $f_l(n)=x(n)*y(n)$ 长度为 $L=L_x+L_y-1$。$N$ 点循环卷积 $f(n)=\mathrm{IDFT}[X(k)Y(k)]_N$ 与线性卷积满足
$f(n) = \sum_{m=-\infty}^{\infty} f_l(n+Nm) R_N(n)$

当 $N \ge L$ 时无混叠，对 $0 \le n \le N-1$ 有 $f(n)=f_l(n)$；

当 $N < L$ 时，混叠区间为 $0 \le n \le L-N-1$，在无混叠区间 $L-N \le n \le N-1$ 上 $f(n)=f_l(n)$。

\textbf{\hlyellow{频域采样}}

已知$X(e^{j\omega}) = DTFT[x(n)] \quad \text{或者} \quad  X(z) = ZT[x(n)])$，对 $X(e^{j\omega}),X(z)$ 等间隔采样得到

$$\boxed{X_N(k) = X(e^{j\omega}) \big|_{\omega=\frac{2\pi}{N}k} = X(z) \big|_{z = W_N^{-k}} \quad k = 0 \cdots N-1} $$

对采样结果逆离散傅里叶变换：

$$x_N(n) = IDFT[X_N(k)] = \sum_{n=-\infty}^{\infty} x(n+iN) R_N(n)$$

如果 $x(n)$ 的长度为 $M$，\textbf{\textcolor{myred}{当频域采样点数 $N \geqslant M$ 时，$x_N(n) = x(n)$}}，否则

$\tilde{x}_N(n) = \sum_{n=-\infty}^{\infty} x(n+iN)$ 会发生时域混叠。通过频率域采样得到频域离散序列 $x_N(k)$，再对 $x_N(k)$ 进行 IDFT 得到的序列 $x_N(n)$ 应是原序列 $x(n)$ 以采样点数 $N$ 为周期进行周期化后的主值区序列。

\textbf{2. 幅度波动系数：}$\boxed{\varepsilon = \sqrt{10^{-\alpha_p/10} - 1}}$

\textbf{3. 归一化函数：}$\boxed{G(p) = \frac{1}{\varepsilon 2^{N-1} \prod_{k=1}^{N} (p - p_k)}}$
\ 
\textbf{5. 去归一化：}$\boxed{H_a(s) = \left. G(p) \right|_{p = s/\Omega_p}}$

\textbf{4. 极点：}$\boxed{p_k = -\text{sh}\,\xi \sin\frac{(2k-1)\pi}{2N} + j\,\text{ch}\,\xi \cos\frac{(2k-1)\pi}{2N},\ \xi = \frac{1}{N}\text{arsh}\frac{1}{\varepsilon}}$

\begin{multicols}{2}
\begin{minipage}{\columnwidth}
\textbf{\hlgreen{I型}}

$|H_a(j\Omega)|^2 = \frac{1}{1 + \varepsilon^2 C_N^2(\Omega/\Omega_p)}$

$\varepsilon$ 越大，通带内波纹幅度越大。$C_N(x)$在通带内为余弦函数，在阻带内为双曲余弦函数。\textbf{与巴特沃斯对比，在相同阶数 $N$ 的情况下，切比雪夫 I 型滤波器的过渡带比巴特沃斯滤波器更陡峭。}
\end{minipage}
\begin{minipage}{\columnwidth}
\textbf{\hlgreen{II型}}

$|H_a(j\Omega)|^2 = \frac{1}{1 + \varepsilon^2 \left( \frac{C_N(\Omega_s/\Omega_p)}{C_N(\Omega_s/\Omega)} \right)^2}$

切比雪夫 I 型滤波器是\textbf{全极点滤波器}，只包含极点。而切比雪夫 II 型滤波器\textbf{既包含极点，又包含零点}。
\end{minipage}
\end{multicols}

\textbf{\hlyellow{频率变换}}
将下面表格中的$p$代入原归一化系统函数$G(p)$

\concepttable{
    \conrow{低通→高通}{$p = \dfrac{\Omega_c}{s}$（高通滤波器的截止频率为 $\Omega_c$），注意$N = -\frac{lg(k_{sp})}{lg(\lambda_{sp})}$}
    \conrow{低通→带通}{$p = \dfrac{s^2 + \Omega_l \Omega_u}{s(\Omega_u - \Omega_l)}$（带通有两个截止频率（上截止 $\Omega_u$、下截止 $\Omega_l$））}
    \conrow{低通→带阻}{$p = \dfrac{s(\Omega_u - \Omega_l)}{s^2 + \Omega_l \Omega_u}$（带阻有两个截止频率（上截止 $\Omega_u$、下截止 $\Omega_l$））}
    \conrow{低通→低通}{$p = \frac{\Omega_c}{\Omega_c'} s$（低通原型 $G(p)$ 的截止频率是 $\Omega_c$，目标低通滤波器的截止频率是 $\Omega_c'$）}
}

\textbf{\hlyellow{模拟滤波器$\rightarrow$数字滤波器}}
\textbf{\hlgreen{脉冲响应不变法}}

\textbf{第一步：}我们根据 $\Omega_p, \Omega_s, \alpha_p, \alpha_s$ 设计好模拟滤波器，得到 $H_a(s)$。

\textbf{第二步：}先进行部分分式展开，即$H_a(s) = \sum_{i=0}^{N-1} \frac{A_i}{s - s_i}$，再极点映射（$s \to z$）：

\concepttable{
    \conrow{单阶极点}{$\displaystyle\frac{A_i}{s - s_i} \longrightarrow \frac{A_i T}{1 - e^{s_i T} z^{-1}}$（最常用）}
    \conrow{$2$ 重极点}{$\displaystyle\frac{A_i}{(s - s_i)^2} \longrightarrow \frac{A_i T^2 e^{s_i T} z^{-1}}{(1 - e^{s_i T} z^{-1})^2}$}
}
\vspace{-2pt}
\textbf{\hlgreen{双线性变换法}}

得到系统函数$H_a(s)$之后，$s$ 替换为 $z^{-1}$ 分式：$\boxed{\displaystyle H(z) = H_a(s) \big|_{s=\frac{2}{T}\frac{1-z^{-1}}{1+z^{-1}}}}$，再化为标准形式即可。

\section{第六章\ FIR 数字滤波器设计}

\textbf{\hlgreen{理想 FIR 滤波器特征}}

\concepttable{
    \conrow{主瓣}{尽量窄，以获得陡峭过渡带}
    \conrow{旁瓣}{尽量小，能量集中于主瓣，减少肩峰与余振，增大阻带衰减}
}

\textbf{\hlgreen{线性相位条件}}

\concepttable{
    \conrow{第一类（偶对称）}{$h(n)$ 为实序列，对 $\frac{N-1}{2}$ 偶对称：$h(n) = h(N-n-1)$}
    \conrow{第二类（奇对称）}{$h(n)$ 为实序列，对 $\frac{N-1}{2}$ 奇对称：$h(n) = -h(N-n-1)$}
}

\textbf{相位函数：}

\begin{itemize}
  \item 群延迟 $\alpha = (N-1)/2$。
  \item 若 $h(n)$ 满足第一类线性相位条件，则 $\theta(\omega) = -\omega\alpha$。
  \item 若 $h(n)$ 满足第二类线性相位条件，则 $\theta(\omega) = -\omega\alpha \pm \frac{\pi}{2}$。
\end{itemize}

\textbf{FIR 线性相位滤波器零点分布特点：必须是互为倒数的共轭对。}

\textbf{\hlgreen{幅度特性与可实现滤波器类型}}

\concepttable{
    \conrow{$h$ 偶对称，$N$ 奇数}{$H_g(\omega)$ 关于 $\omega=0,\pi,2\pi$ 偶对称，$H_g(0)=H_g(\pi)=H_g(2\pi)\neq0$，
    
    \textbf{可实现所有类型}}
    \conrow{$h$ 偶对称，$N$ 偶数}{$H_g(\omega)$ 关于 $\omega=\pi$ 奇对称，$H_g(\pi)=0$，
    
    \textbf{仅适合低通、带通}}
    \conrow{$h$ 奇对称，$N$ 奇数}{$H_g(\omega)$ 关于 $\omega=0,\pi,2\pi$ 奇对称，$H_g(0)=H_g(\pi)=H_g(2\pi)=0$，
    
    \textbf{仅适合高通、带阻}}
    \conrow{$h$ 奇对称，$N$ 偶数}{$H_g(\omega)$ 关于 $\omega=0,2\pi$ 奇对称，$H_g(\pi)=H_g(2\pi)=0$，
    
    \textbf{仅适合高通、带通}}
}

\textbf{\hlgreen{吉布斯效应（窗函数截断）}}

\concepttable{
    \conrow{频域反映}{过渡带 $\approx$ 主瓣宽 $4\pi/N$；通带波动、阻带余振，最大负峰在 $\omega_c+2\pi/N$}
    \conrow{原因}{主瓣宽 $\rightarrow$ 过渡带；旁瓣 $\rightarrow$ 肩峰/余振，阻带衰减下降}
    \conrow{改善}{选用旁瓣更小、主瓣更窄的窗函数（如 Hamming、Blackman 等）}
}

\textbf{\hlgreen{IIR 与 FIR 数字滤波器比较}}

\concepttable{
    \conrow{性能}{FIR 易实现严格线性相位；IIR 相位非线性需补偿。IIR 零极点系统，低阶高选择性；FIR 全零点，同性能阶次更高}
    \conrow{结构}{IIR 递归型，极点须在单位圆内保稳定，有限字长影响大，易寄主振荡；FIR 非递归，极点仅在原点，永远稳定，误差小}
    \conrow{运算}{IIR 脉冲响应无限长；FIR 有限长，可用 FFT 快速运算}
    \conrow{设计}{IIR 可借助模拟滤波器公式/查表，计算量小；FIR 需计算机辅助设计，要求高}
}

\textbf{常用函数}：当系统（传输函数）为$e^{-j\omega\alpha}$ 时，单位脉冲响应 $h_d(n)$ 为：

\begin{itemize}
\item \textbf{理想低通}（$|\omega| \le \omega_c$）：
\[
h_d(n) = \frac{\sin[\omega_c(n-\alpha)]}{\pi(n-\alpha)}, \quad \alpha = \frac{N-1}{2}
\]
\item \textbf{理想高通}（$|\omega| \ge \omega_c$）：
\[
h_d(n) = \frac{\sin[\pi(n-\alpha)] - \sin[\omega_c(n-\alpha)]}{\pi(n-\alpha)}
\]
\item \textbf{理想带通}（$\omega_{c1} \le |\omega| \le \omega_{c2}$）：
\[
h_d(n) = \frac{\sin[\omega_{c2}(n-\alpha)] - \sin[\omega_{c1}(n-\alpha)]}{\pi(n-\alpha)}
\]
\item \textbf{理想带阻}（$|\omega| \le \omega_{c1}$ 或 $|\omega| \ge \omega_{c2}$）：
\[
h_d(n) = \frac{\sin[\omega_{c1}(n-\alpha)] + \sin[\pi(n-\alpha)] - \sin[\omega_{c2}(n-\alpha)]}{\pi(n-\alpha)}
\]
\end{itemize}
