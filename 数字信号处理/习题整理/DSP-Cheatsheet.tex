\prob{1-8}{}{
  单位脉冲响应分别为$h_1(n) = \delta(n) - \delta(n-1)$, $h_2(n) = h_1(n)$, $h_3(n) = u(n)$的三个系统级联在一起。此级联系统的单位脉冲响应$h_s(n)$是什么？连接的次序对整个系统有影响吗？
}

\textbf{解：}

利用 Z 变换求解。三个系统的单位脉冲响应分别为：
\[
h_1(n) = \delta(n) - \delta(n-1) \quad \xrightarrow{\mathcal{Z}} \quad H_1(z) = 1 - z^{-1}
\]
\[
h_2(n) = h_1(n) \quad \xrightarrow{\mathcal{Z}} \quad H_2(z) = 1 - z^{-1}
\]
\[
h_3(n) = u(n) \quad \xrightarrow{\mathcal{Z}} \quad H_3(z) = \frac{1}{1 - z^{-1}}, \quad |z| > 1
\]

级联系统的 Z 变换为三者相乘：
\[
H_s(z) = H_1(z) H_2(z) H_3(z) = (1 - z^{-1})(1 - z^{-1})\frac{1}{1 - z^{-1}} = 1 - z^{-1}
\]

求逆 Z 变换得级联系统的单位脉冲响应：
\[
h_s(n) = \delta(n) - \delta(n-1) = h_1(n) = h_2(n)
\]

\textbf{连接的次序对整个系统没有影响}，因为 Z 域相乘满足交换律，且最终结果与次序无关。

有一模拟信号 $x_a(t) = \cos(2\pi ft + \varphi)$，其中 $f = 30\text{Hz}$，$\varphi = \pi/2$。
(1) 求信号 $x_a(t)$ 的周期。

(2) 用采样间隔 $T = 0.01\text{s}$ 对 $x_a(t)$ 进行采样，写出采样信号 $\widehat{x}_a(t)$ 的表达式。

(3) 画出对应 $\widehat{x}_a(t)$ 的时域离散信号 $x(n)$ 的波形，并求出 $x(n)$ 的周期。
}

\textbf{解题思路：} 利用连续信号周期公式计算原信号周期；通过理想采样模型将连续信号与冲激串相乘得到采样信号表达式；根据采样频率计算离散数字频率，进而由数字频率与周期的关系确定离散序列的周期。

(1) 模拟信号周期：
\[
T = \frac{2\pi}{2\pi f} = \frac{2\pi}{60\pi} = \frac{1}{30}\,\text{s}
\]

(2) 采样信号表达式：
\[
p_s(t) = \sum_{n=-\infty}^{+\infty} \delta(t-nT)
      = \sum_{n=-\infty}^{+\infty} \delta(t-0.01n)
\]
\[
\begin{aligned}
\hat{x}_a(t) &= x_a(t)p_s(t) \\
            &= \sum_{n=-\infty}^{+\infty} x_a(t)\delta(t-nT) \\
            &= \sum_{n=-\infty}^{+\infty} x_a(nT)\delta(t-nT) \\
            &= \sum_{n=-\infty}^{+\infty} \cos(0.6\pi n + \tfrac{\pi}{2})
               \delta(t-0.01n)
\end{aligned}
\]

(3) 离散序列周期：
数字频率 $\omega = 2\pi f T = 0.6\pi$，
由 $\dfrac{2\pi}{\omega} = \dfrac{10}{3}$ 得 $x(n)$ 的周期 $N = 10$。

设 $f(t)$ 的奈奎斯特频率为 $\Omega_N'$。

载波频率为 $\Omega_c$，因此 $f(t)$ 的最高频率为
\[
\Omega_c' = \Omega_c + \frac{\Omega_N}{2}.
\]
由奈奎斯特采样定理得
\[
\Omega_N' = 2\Omega_c' = 2\Omega_c + \Omega_N.
\]

\prob{4}{}{设$x(n) = \begin{cases} 1, & n = 0,1 \\ 0, & \text{其他} \end{cases}$，将$x(n)$以4为周期进行周期延拓，形成周期序列$\widetilde{x}(n)$，画出$x(n)$和$\widetilde{x}(n)$的波形，求出$\widetilde{x}(n)$的离散傅里叶级数$\widetilde{X}(k)$和傅里叶变换。}

\textbf{解：}

\vspace{4pt}
\begin{itemize}[leftmargin=*,itemsep=6pt]
\item[\textbf{1)}] \textbf{求 DFS：$\widetilde{X}(k)$}

$\widetilde{x}(n)$ 由 $x(n)=\{1,1\}$ 以 $N=4$ 周期延拓得到。
\[
\widetilde{X}(k)=\mathrm{DFS}[\widetilde{x}(n)]
                 =\sum_{n=0}^{3}\widetilde{x}(n)e^{-j\frac{2\pi}{4}kn}
                 =\sum_{n=0}^{1}e^{-j\frac{\pi}{2}kn}
                 =1+e^{-j\frac{\pi}{2}k}
\]
提公因子化为幅度-相位形式：
\[
\widetilde{X}(k)=e^{-j\frac{\pi}{4}k}
                  \bigl(e^{j\frac{\pi}{4}k}+e^{-j\frac{\pi}{4}k}\bigr)
                =2\cos\!\Bigl(\frac{\pi}{4}k\Bigr)\,
                  e^{-j\frac{\pi}{4}k},
\qquad \widetilde{X}(k)\ \text{以 $4$ 为周期}.
\]

\item[\textbf{2)}] \textbf{求 FT：$X(e^{j\omega})$}

周期序列的 FT 为频域冲激串（$N=4$）：
\[
X(e^{j\omega})=\mathrm{FT}[\widetilde{x}(n)]
              =\frac{2\pi}{N}\sum_{k=-\infty}^{\infty}
                \widetilde{X}(k)\,
                \delta\!\Bigl(\omega-\frac{2\pi}{N}k\Bigr)
              =\frac{\pi}{2}\sum_{k=-\infty}^{\infty}
                \widetilde{X}(k)\,
                \delta\!\Bigl(\omega-\frac{\pi}{2}k\Bigr).
\]
代入 $\widetilde{X}(k)$ 表达式：
\[
\boxed{%
X(e^{j\omega})=\pi\sum_{k=-\infty}^{\infty}
               \cos\!\Bigl(\frac{\pi}{4}k\Bigr)
               e^{-j\frac{\pi}{4}k}\,
               \delta\!\Bigl(\omega-\frac{\pi}{2}k\Bigr)
}.
\]
\end{itemize}

\prob{\textbf{\hlblue{5}}}{}{设题5图所示的序列$x(n)$的FT用$X(e^{j\omega})$表示，不直接求出$X(e^{j\omega})$，完成下列运算或工作：\\
(1) $X(e^{j0})$；\\
(2) $\int_{-\pi}^{\pi} X(e^{j\omega}) d\omega$；\\
(3) $X(e^{j\pi})$；\\
(4) 确定并画出傅里叶变换实部$\text{Re}[X(e^{j\omega})]$的时间序列$x_o(n)$；\\
(5) $\int_{-\pi}^{\pi} \left| X(e^{j\omega}) \right|^2 d\omega$；\\
(6) $\int_{-\pi}^{\pi} \left| \frac{dX(e^{j\omega})}{d\omega} \right|^2 d\omega$。\\}

\textbf{解：}

\begin{itemize}[leftmargin=*,itemsep=4pt]
\item[(1)] \textbf{\textcolor{myred}{$X(e^{j0})$ — 直流分量}}
  \[
  X(e^{j0}) = \sum_{n=-\infty}^{\infty} x(n) e^{-j0\cdot n}
            = \sum_{n=-\infty}^{\infty} x(n)
            = \sum_{n=-3}^{7} x(n) = 6
  \]
  \textcolor{myred}{要点：$\omega=0 \ \Rightarrow\ e^{-j\omega n}=1$，即序列各点之和。}

\item[(2)] \textbf{\textcolor{myred}{$\displaystyle\int_{-\pi}^{\pi} X(e^{j\omega})\,\mathrm{d}\omega$ — 反变换求 $x(0)$}}
  \[
  x(n) = \frac{1}{2\pi}\int_{-\pi}^{\pi} X(e^{j\omega})\,e^{j\omega n}\,\mathrm{d}\omega
  \quad\overset{n=0}{\Longrightarrow}\quad
  x(0) = \frac{1}{2\pi}\int_{-\pi}^{\pi} X(e^{j\omega})\,\mathrm{d}\omega
  \]
  \[
  \therefore\ \int_{-\pi}^{\pi} X(e^{j\omega})\,\mathrm{d}\omega = 2\pi\cdot x(0) = 4\pi
  \]

\item[(3)] \textbf{\textcolor{myred}{$X(e^{j\pi})$ — Nyquist 频率}}
  \[
  X(e^{j\pi}) = \sum_{n=-\infty}^{\infty} x(n)\,e^{-j\pi n}
              = \sum_{n=-3}^{7} (-1)^n x(n) = 2
  \]
  \textcolor{myred}{要点：$e^{-j\pi n}=(-1)^n$，序列交错求和。}

\item[(4)] \textbf{\textcolor{myred}{$\operatorname{Re}[X(e^{j\omega})]$ 对应的时间序列 $x_e(n)$}}
  \[
  \boxed{x_e(n) = \tfrac{1}{2}\bigl[x(n) + x^*(-n)\bigr]}
  \xrightarrow{\text{实序列}}\
  x_e(n) = \tfrac{1}{2}\bigl[x(n) + x(-n)\bigr]
  \]
  \textcolor{myred}{即共轭对称分量，其傅里叶变换为 $\operatorname{Re}[X(e^{j\omega})]$.}

\item[(5)] \textbf{\textcolor{myred}{$\displaystyle\int_{-\pi}^{\pi} |X(e^{j\omega})|^2\,\mathrm{d}\omega$ — 帕塞瓦尔定理}}
  \[
  \int_{-\pi}^{\pi} |X(e^{j\omega})|^2\,\mathrm{d}\omega
  = 2\pi\!\!\sum_{n=-\infty}^{\infty} |x(n)|^2
  = 2\pi\!\!\sum_{n=-3}^{7} |x(n)|^2 = 28\pi
  \]

\item[(6)] \textbf{\textcolor{myred}{$\displaystyle\int_{-\pi}^{\pi} \bigl|\tfrac{\mathrm{d}X}{\mathrm{d}\omega}\bigr|^2\,\mathrm{d}\omega$ — 频域微分 + 帕塞瓦尔}}
  \[
  \frac{\mathrm{d}X(e^{j\omega})}{\mathrm{d}\omega}
  = \operatorname{FT}\bigl[-jn\,x(n)\bigr]
  \]
  \[
  \therefore\ \int_{-\pi}^{\pi} \Bigl|\frac{\mathrm{d}X}{\mathrm{d}\omega}\Bigr|^2\,\mathrm{d}\omega
  = 2\pi\!\!\sum_{n=-3}^{7} |n\,x(n)|^2 = 316\pi
  \]
\end{itemize}

\prob{6}{}{试求如下序列的傅里叶变换：\\
(1) $x_1(n) = \delta(n-3)$；\\
(2) $x_2(n) = \frac{1}{2}\delta(n+1) + \delta(n) + \frac{1}{2}\delta(n-1)$；\\
(3) $x_3(n) = a^n u(n)\ (0 < a < 1)$；\\
(4) $x_4(n) = u(n+3) - u(n-4)$。}

\textbf{解：}

(1) 由 $\delta(n-3)$ 的筛选性，仅有 $n=3$ 项非零：
  \[
  X_1(e^{j\omega}) = \sum_{n=-\infty}^{\infty} \delta(n-3) e^{-j\omega n} = e^{-j3\omega}
  \]

(2) 序列仅在 $n=-1,0,1$ 处非零，直接代入 FT 定义，并利用欧拉公式合并共轭项：
  \[
  X_2(e^{j\omega}) = \sum_{n=-\infty}^{\infty} x_2(n) e^{-j\omega n}
  = \frac{1}{2}e^{j\omega} + 1 + \frac{1}{2}e^{-j\omega}
  = 1 + \cos\omega
  \]

(3) 由 $u(n)$ 截断负半轴，使求和下限为 $n=0$；再利用几何级数公式 $\sum_{n=0}^{\infty} r^n = \frac{1}{1-r}$（$|r|<1$），此处 $r = a e^{-j\omega}$ 且 $0<a<1$ 保证收敛：
  \[
  X_3(e^{j\omega}) = \sum_{n=-\infty}^{\infty} a^n u(n) e^{-j\omega n}
  = \sum_{n=0}^{\infty} \bigl(a e^{-j\omega}\bigr)^n
  = \frac{1}{1 - a e^{-j\omega}}
  \]

(4) $u(n+3)-u(n-4)$ 为长度 7 的矩形窗（$n\in[-3,3]$），可分正负半轴求和，然后分别用有限项等比级数公式收取：
  \[
  \begin{aligned}
  X_4(e^{j\omega})
  &= \sum_{n=-\infty}^{\infty} \bigl[u(n+3) - u(n-4)\bigr] e^{-j\omega n}
   = \sum_{n=-3}^{3} e^{-j\omega n} \\[4pt]
  &= \sum_{n=0}^{3} e^{-j\omega n} + \sum_{n=-3}^{-1} e^{-j\omega n}
   = \frac{1 - e^{-j4\omega}}{1 - e^{-j\omega}} + \frac{e^{j\omega} - e^{-j3\omega}}{1 - e^{-j\omega}} \cdot e^{j\omega}
   = \frac{1 - e^{-j4\omega}}{1 - e^{-j\omega}} + \frac{e^{j\omega}\bigl(1 - e^{-j3\omega}\bigr)}{1 - e^{-j\omega}} e^{j\omega}
  \end{aligned}
  \]

\]

\prob{3-8}{}{已知 $x(n)$ 是长度为 $N$ 的有限长度序列，$X(k)=\text{DFT}[x(n)]$，在每两点之间插入一个 $0$ 值，得到序列
$y(n) = \begin{cases} x(n/2), & n \text{为偶数} \\ 0, & n \text{为奇数} \end{cases}$，
试求 $Y(k)=\text{DFT}[y(n)]$ 与 $X(k)$ 之间的关系。}

\textbf{解：}

\textbf{思路：} 将 $y(n)$ 按奇偶位置分解为两个 $N$ 点子序列，利用 DFT 定义式分别求和，结合旋转因子性质 $W_{2N}^{2rk}=W_N^{rk}$ 化简，得到 $Y(k)$ 与 $X(k)$ 的周期延拓关系。

按 $n$ 的奇偶把 $y(n)$ 分解为两个 $N$ 点的子序列
\[
\left\{\begin{matrix}
y(2r)=x_{1}(r)=x(r) \\
y(2r+1)=x_{2}(r)=0
\end{matrix}\right.,
\quad r=0,\dots,N-1.
\]

由 DFT 定义（这里的推导酌情复习）：
\begin{align*}
Y(k) &= \text{DFT}[y(n)] = \sum_{n=0}^{2N-1} y(n) W_{2N}^{nk} \\
&= \sum_{r=0}^{N-1} y(2r) W_{2N}^{2rk}
   + \sum_{r=0}^{N-1} y(2r+1) W_{2N}^{(2r+1)k} \\
&= \sum_{r=0}^{N-1} x_{1}(r) W_{N}^{rk}
   + W_{2N}^{k} \sum_{r=0}^{N-1} x_{2}(r) W_{N}^{rk} \\
&= \sum_{r=0}^{N-1} x(r) W_{N}^{rk}
   + W_{2N}^{k} \sum_{r=0}^{N-1} 0 \cdot W_{N}^{rk} \\
&= X(k), \quad k=0,\dots,N-1.
\end{align*}

对于 $k=0,\dots,N-1$，有
\[
Y(k+N) = X_{1}(k) - W_{2N}^{k} X_{2}(k) = X(k).
\]

因此
\[
Y(k) = X((k))_{N} R_{2N}(k),
\]
即 \textbf{\textcolor{myred}{$Y(k)$ 是 $X(k)$ 以 $N$ 为周期的周期延拓，并受 $2N$ 点矩形窗截断。}}

\prob{3-9}{}{已知 $X(k)$，求其对应的离散傅里叶逆变换 $\text{IDFT}[X(k)]$：\\
(1) $X(k) = \begin{cases} \frac{N}{2} e^{j\theta}, & k = m \\ 0, & \text{其它}k \\ \frac{N}{2} e^{j\theta}, & k = N-m \end{cases}$ \\
(2) $X(k) = \begin{cases} -j\frac{N}{2} e^{j\theta}, & k = m \\ 0, & \text{其它}k \\ j\frac{N}{2} e^{-j\theta}, & k = N-m \end{cases}$}

\textbf{解：}

\textbf{思路：} 利用 IDFT 定义式直接求和，结合 $W_N^{-(N-m)n} = W_N^{mn}$ 化简，根据欧拉公式将复指数和转化为三角函数。

(1) 由 IDFT 定义：
\begin{align*}
x(n) &= \frac{1}{N} \sum_{k=0}^{N-1} X(k) W_N^{-kn} \\
&= \frac{1}{N} \left[ \frac{N}{2} e^{j\theta} e^{j\frac{2\pi}{N}mn}
   + \frac{N}{2} e^{-j\theta} e^{j\frac{2\pi}{N}(N-m)n} \right] \\
&= \frac{1}{2} \left[ e^{j\left(\frac{2\pi}{N}mn + \theta\right)}
   + e^{-j\left(\frac{2\pi}{N}mn + \theta\right)} \right] \\
&= \cos\left(\frac{2\pi}{N}mn + \theta\right),
   \quad n = 0, 1, \dots, N-1
\end{align*}

(2) 同理：
\begin{align*}
x(n) &= \frac{1}{N} \sum_{k=0}^{N-1} X(k) W_N^{-kn} \\
&= \frac{1}{N} \left[ -\frac{N}{2} j e^{j\theta} e^{j\frac{2\pi}{N}mn}
   + \frac{N}{2} j e^{-j\theta} e^{j\frac{2\pi}{N}(N-m)n} \right] \\
&= \frac{1}{2j} \left[ e^{j\left(\frac{2\pi}{N}mn + \theta\right)}
   - e^{-j\left(\frac{2\pi}{N}mn + \theta\right)} \right] \\
&= \sin\left(\frac{2\pi}{N}mn + \theta\right),
   \quad n = 0, 1, \dots, N-1
\end{align*}

\textbf{\hlgreen{频域采样}}

\prob{3-22}{}{设有一频谱分析用的信号处理器，采样点数必须为 $2$ 的整数幂，要求达到的频谱分辨率为 $\leq 10Hz$，若采用的采样间隔为 $0.1ms$，试确定：\\

(2) 所允许处理信号的最高频率；\\
(3) 在一个记录中的最小点数。}

\textbf{系统信号处理流程：}

$$\begin{aligned}
\text{模拟信号}(f_H) \ \ &\xrightarrow[\begin{gathered} T_s, \; f_s \\ (f_s \ge 2f_H) \end{gathered}]{\text{时域采样 (A/D)}} \ \ \text{离散非周期信号} \ \ \xrightarrow[\begin{gathered} T_p, \; N \\ (T_p = N \cdot T_s) \end{gathered}]{\text{信号截断 (加窗)}} \ \ \text{有限长序列}(N) \\
\\
&\xrightarrow[\begin{gathered} \Delta F \\ (\Delta F = \frac{f_s}{N} = \frac{1}{T_p}), \; N=2^m \end{gathered}]{\text{数字处理 (DFT/FFT 计算)}} \ \ \text{离散频域序列}(\Delta F)
\end{aligned}$$

\textbf{解：}

(1) 频域分辨率 $\Delta F$ 严格取决于时域的观察总时间 $T_p$。\textbf{\textcolor{myred}{只要观察时间足够长，频域就足够精细}}。

根据 $\Delta F = \frac{1}{T_p}$，得：

$$T_p = \frac{1}{\Delta F} \ge \frac{1}{10} = 0.1 \text{ s}$$

(2) 由时域采样间隔 $T_s$ 先求出采样率 $f_s$，再利用奈奎斯特条件限制最高频率。

计算采样频率：

$$f_s = \frac{1}{T_s} = \frac{1}{10^{-4} \text{ s}} = 10000 \text{ Hz}$$

根据采样定理 $f_s \ge 2f_H$，得：

$$f_H \le \frac{1}{2} f_s = \frac{10000}{2} = 5000 \text{ Hz} = 5 \text{ kHz}$$

(3) 理论计算点数(DFT)：

$$N_{\text{理论}} = \frac{f_s}{\Delta F} = \frac{10000 \text{ Hz}}{10 \text{ Hz}} = 1000 \text{ 点}$$

题目要求 $N$ 必须为 $2$ 的整数幂，由 $2^9 = 512$且$2^{10} = 1024$（大于 1000），所以 $N = 1024$。

\prob{5-2}{}{
给出下列系统的直接 I 型、直接 II 型、直接 II 型转置结构、级联型和并联型结构。

(1) $y(n) = \frac{3}{4} y(n-1) - \frac{1}{8} y(n-2) + x(n) + \frac{1}{3} x(n-1)$

(2) $y(n) = -0.1 y(n-1) + 0.2 y(n-2) + 3x(n) + 3.6x(n-1) + 0.6x(n-2)$

(3) $y(n) = y(n-1) - \frac{1}{2} y(n-2) + x(n) - x(n-1) + x(n-2)$

(4) $y(n) = \frac{1}{2} y(n-1) - \frac{1}{4} y(n-2) + x(n) + x(n-1)$
}

\textbf{解：}

由差分方程以及题干要求来看，系统是 \textbf{IIR 系统}，求解系统函数 $H(z)$：

(1) $H(z) = \dfrac{1 + \frac{1}{3}z^{-1}}{1 - \frac{3}{4}z^{-1} + \frac{1}{8}z^{-2}}$

(2) $H(z) = \dfrac{3 + 3.6z^{-1} + 0.6z^{-2}}{1 + 0.1z^{-1} - 0.2z^{-2}}$

(3) $H(z) = \dfrac{1 - z^{-1} + z^{-2}}{1 - z^{-1} + \frac{1}{2}z^{-2}}$

(4) $H(z) = \dfrac{1 + z^{-1}}{1 - \frac{1}{2}z^{-1} + \frac{1}{4}z^{-2}}$



\textbf{级联型结构：}

(1) 分母因式分解：$1 - \frac{3}{4}z^{-1} + \frac{1}{8}z^{-2} = (1 - \frac{1}{2}z^{-1})(1 - \frac{1}{4}z^{-1})$，故
\[
H(z) = \frac{1 + \frac{1}{3}z^{-1}}{(1 - \frac{1}{2}z^{-1})(1 - \frac{1}{4}z^{-1})}.
\]

\textbf{并联型结构：}

(1) 部分分式展开：
\[
H(z) = \frac{1 + \frac{1}{3}z^{-1}}{(1 - \frac{1}{2}z^{-1})(1 - \frac{1}{4}z^{-1})}
     = \frac{A}{1 - \frac{1}{2}z^{-1}} + \frac{B}{1 - \frac{1}{4}z^{-1}}.
\]
\textbf{系数求解：}

$A = \left.(1 - \frac{1}{4}z^{-1})H(z)\right|_{z^{-1}=2} = \frac{1 + \frac{1}{3}\cdot 2}{1 - \frac{1}{4}\cdot 2} = \frac{1 + \frac{2}{3}}{1 - \frac{1}{2}} = \frac{5/3}{1/2} = \frac{10}{3}$，

$B = \left.(1 - \frac{1}{2}z^{-1})H(z)\right|_{z^{-1}=4} = \frac{1 + \frac{1}{3}\cdot 4}{1 - \frac{1}{2}\cdot 4} = \frac{1 + \frac{4}{3}}{1 - 2} = \frac{7/3}{-1} = -\frac{7}{3}$。

故
\[
H(z) = \frac{10/3}{1 - \frac{1}{2}z^{-1}} - \frac{7/3}{1 - \frac{1}{4}z^{-1}}.
\]

\section{第五章\ IIR数字滤波器的设计}

\prob{1}{}{设计一个巴特沃斯低通滤波器，要求通带截止频率 $f_p=6\,\text{kHz}$，通带最大衰减 $\alpha_p=3\,\text{dB}$，阻带截止频率 $f_s=12\,\text{kHz}$，阻带最小衰减 $\alpha_s=25\,\text{dB}$。求出滤波器归一化系统函数 $G(p)$ 以及实际的 $H_a(s)$。}

\textbf{解题思路：} 根据巴特沃斯低通滤波器的技术指标，先由通带和阻带衰减计算滤波器阶数 $N$，再查表或计算极点得到归一化系统函数 $G(p)$，最后通过频率变换（去归一化）得到实际系统函数 $H_a(s)$。

\textbf{解：}

（1）求阶数 $N$。

由阶数公式 $N = \frac{\lg k_{sp}}{\lg \lambda_{sp}}$，其中
\[
k_{sp} = \sqrt{\frac{10^{0.1\alpha_s} - 1}{10^{0.1\alpha_p} - 1}}
      = \sqrt{\frac{10^{2.5} - 1}{10^{0.3} - 1}} \approx 17.794,
\]
\[
\lambda_{sp} = \frac{\Omega_s}{\Omega_p}
            = \frac{2\pi \times 12 \times 10^3}{2\pi \times 6 \times 10^3} = 2.
\]
代入得
\[
N = \frac{\lg 17.794}{\lg 2} = 4.15,
\]
故取 $N = 5$（实际中也可取 $N = 4$，以简化实现）。

（2）求归一化系统函数 $G(p)$。

由 $N = 5$ 查表得五阶巴特沃斯归一化低通滤波器系统函数：
\[
G(p) = \frac{1}{p^5 + 3.2361p^4 + 5.2361p^3 + 5.2361p^2 + 3.2361p + 1},
\]
或等价地
\[
G(p) = \frac{1}{(p^2 + 0.618p + 1)(p^2 + 1.618p + 1)(p + 1)}.
\]

也可由极点公式
\[
p_k = e^{j\pi\left(\frac{1}{2} + \frac{2k+1}{2N}\right)}, \quad k = 0,1,2,3,4
\]
计算极点，再按
\[
G(p) = \frac{1}{\prod_{k=0}^{4}(p - p_k)}
\]
展开得到相同结果。

（3）去归一化（LP-LP 频率变换）。

由 $\alpha_p = 3\,\text{dB}$ 知 $\Omega_c = \Omega_p = 2\pi \times 6 \times 10^3\,\text{rad/s}$，故
\[
H_a(s) = G(p)\big|_{p = s / \Omega_c}
       = \frac{\Omega_c^5}{s^5 + 3.2361\Omega_c s^4 + 5.2361\Omega_c^2 s^3
         + 5.2361\Omega_c^3 s^2 + 3.2361\Omega_c^4 s + \Omega_c^5}.
\]

分母因式形式为
\[
H_a(s) = \frac{\Omega_c^5}{(s^2 + 0.6180\Omega_c s + \Omega_c^2)
                        (s^2 + 1.6180\Omega_c s + \Omega_c^2)
                        (s + \Omega_c)}.
\]

上式中 $\Omega_c$ 未代入数值，以便清晰展示去归一化后截止频率对系统函数的影响。

\prob{2}{}{设计一个切比雪夫低通滤波器，要求通带截止频率 $f_p=3\,\text{kHz}$，通带最大衰减 $\alpha_p=0.2\,\text{dB}$，阻带截止频率 $f_s=12\,\text{kHz}$，阻带最小衰减 $\alpha_s=50\,\text{dB}$。求出滤波器归一化系统函数 $G(p)$ 和实际的 $H_a(s)$。}

\textbf{解题思路}：根据切比雪夫低通滤波器的技术指标，先确定通带、阻带参数与频率比，计算波纹参数 $\varepsilon$ 与阶数 $N$，再求解归一化原型滤波器极点并得到 $G(p)$，最后通过频率去归一化得到实际系统函数 $H_a(s)$。

\begin{enumerate}
\item[(1)] \textbf{确定技术指标}。
\[
\alpha_{p} = 0.2\ \mathrm{dB},\quad
\Omega_{p} = 2\pi f_{p} = 6\pi \times 10^{3}\ \mathrm{rad/s}
\]
\[
\alpha_{s} = 50\ \mathrm{dB},\quad
\Omega_{s} = 2\pi f_{s} = 24\pi \times 10^{3}\ \mathrm{rad/s}
\]
\[
\lambda_{p} = 1,\quad
\lambda_{s} = \frac{\Omega_{s}}{\Omega_{p}} = 4
\]

\item[(2)] \textbf{求阶数 $N$ 与波纹参数 $\varepsilon$}。
\[
k^{-1} = \sqrt{\frac{10^{0.1\alpha_{s}} - 1}{10^{0.1\alpha_{p}} - 1}}
      \approx 1456.65
\]
\[
N = \frac{\operatorname{arch} k^{-1}}{\operatorname{arch} \lambda_{s}}
  = \frac{\operatorname{arch} 1456.65}{\operatorname{arch} 4}
  \approx 3.8659
\]
取 $N = 4$ 以满足指标要求。
\[
\varepsilon = \sqrt{10^{0.1\alpha_{p}} - 1} = 0.2171
\]

\item[(3)] \textbf{求归一化系统函数 $G(p)$}。
\[
G(p) = \frac{1}{\varepsilon \cdot 2^{N-1} \prod_{k=1}^{N}(p-p_{k})}
     = \frac{1}{1.7368 \prod_{k=1}^{4}(p-p_{k})}
\]
极点由下式给出：
\[
p_{k} = -\cosh \xi \sin \frac{(2k-1)\pi}{2N}
        + j \cosh \xi \cos \frac{(2k-1)\pi}{2N},
\quad k=1,2,3,4
\]
其中
\[
\xi = \frac{1}{N} \operatorname{arsh}\frac{1}{\varepsilon}
    = \frac{1}{4} \operatorname{arsh}\frac{1}{0.2171}
    \approx 0.5580
\]
计算得
\begin{align*}
p_{1} &= -0.4438 + j 1.0715, &
p_{2} &= -1.0715 + j 0.4438, \\
p_{3} &= -1.0715 - j 0.4438, &
p_{4} &= -0.4438 - j 1.0715.
\end{align*}

\item[(4)] \textbf{去归一化得实际系统函数 $H_a(s)$}。
令 $s_k = \Omega_p p_k = 6\pi \times 10^3 p_k$，则
\[
H_a(s) = G(p)\big|_{p = s/\Omega_p}
       = \frac{\Omega_p^4}{1.7368 \prod_{k=1}^{4}(s - s_k)}.
\]
利用共轭对称性 $p_4 = p_1^*,\ p_3 = p_2^*$，将分母化为二阶实系数因子：
\[
H_a(s) = \frac{7.2687 \times 10^{16}}
               {(s^2 + 1.6731 \times 10^4 s + 4.7791 \times 10^8)}
               {(s^2 + 4.0394 \times 10^4 s + 4.7790 \times 10^8)}.
\]
分母多项式形式可由上式展开得到。
\end{enumerate}

\prob{3}{}{设计一个巴特沃斯高通滤波器，要求其通带截止频率 $f_p=20\,\text{kHz}$，阻带截止频率 $f_s=10\,\text{kHz}$，$f_p$ 处最大衰减为 $3\,\text{dB}$，阻带最小衰减 $\alpha_s=15\,\text{dB}$。求出该高通滤波器的系统函数 $H_a(s)$。}

解：\textbf{解题思路}：通过频率变换将高通指标转换为归一化低通指标，按巴特沃斯逼近确定滤波器阶数及归一化系统函数，再通过 $p = \Omega_c / s$ 变换得到实际高通系统函数。

\begin{enumerate}
\item[(1)] \textbf{确定高通滤波器技术指标}。  
$f_{p}=20\,\text{kHz}$，$\alpha_{p}=3\,\text{dB}$；  
$f_{s}=10\,\text{kHz}$，$\alpha_{s}=15\,\text{dB}$。

\item[(2)] \textbf{转换为归一化低通指标}。  
由高通到低通频率变换公式 $\lambda_{p}=1$，$\lambda_{s}=\Omega_{p}/\Omega_{s}$，得  
\[
\lambda_{p}=1,\quad a_{p}=3\,\text{dB},\quad
\lambda_{s}=2,\quad a_{s}=15\,\text{dB}.
\]

\item[(3)] \textbf{设计归一化低通 $G(p)$}。  
巴特沃斯滤波器参数计算：
\[
k_{sp}=\sqrt{\frac{10^{0.1\alpha_{p}}-1}{10^{0.1\alpha_{s}}-1}}=0.18,\quad
\lambda_{sp}=\frac{\lambda_{s}}{\lambda_{p}}=2,
\]
\[
N=-\frac{\lg k_{sp}}{\lg\lambda_{sp}}=-\frac{\lg0.18}{\lg2}=2.47.
\]
取 $N=3$，查表得三阶巴特沃斯归一化低通系统函数：
\[
G(p)=\frac{1}{p^{3}+2p^{2}+2p+1}.
\]

\item[(4)] \textbf{频率变换得高通系统函数 $H(s)$}。  
令 $p=\Omega_c/s$，代入 $G(p)$ 得
\[
H(s)=G(p)\big|_{p=\Omega_c/s}
     =\frac{s^{3}}{s^{3}+2\Omega_c s^{2}+2\Omega_c^{2}s+\Omega_c^{3}}.
\]
\end{enumerate}

\prob{4}{}{已知模拟滤波器的系统函数 $H_a(s)$ 如下：\newline (1) $H_a(s)=\dfrac{s+a}{(s+a)^2+b^2}$\newline (2) $H_a(s)=\dfrac{b}{(s+a)^2+b^2}$\newline 式中 $a$、$b$ 为常数，设 $H_a(s)$ 因果稳定，试采用脉冲响应不变法将其转换成数字滤波器 $H(z)$。}

解：本题针对两种二阶模拟滤波器基本节，采用脉冲响应不变法推导其数字滤波器系统函数。基本步骤为：确定模拟滤波器极点，进行部分分式展开，对每一项极点套用脉冲响应不变法变换公式 $H(z)=\sum_{k}\frac{A_k}{1-e^{s_k T}z^{-1}}$，最后通分合并为实系数有理分式形式。设采样周期为 $T$。

\begin{enumerate}
\item[(1)] $ H_{a}(s)=\dfrac{s+a}{(s+a)^{2}+b^{2}} $

\textbf{极点：}
$$ s_{1}=-a+\mathrm{j}b, \quad s_{2}=-a-\mathrm{j}b $$

\textbf{部分分式展开：}
设 $H_{a}(s)=\dfrac{A_{1}}{s-s_{1}}+\dfrac{A_{2}}{s-s_{2}}$，比较分子系数得
$$
\begin{cases}
A_{1}+A_{2}=1 \\
-A_{1}s_{2}-A_{2}s_{1}=a
\end{cases}
$$
解得 $A_{1}=A_{2}=1/2$，即
$$ H_{a}(s)=\frac{1/2}{s-(-a+\mathrm{j}b)}+\frac{1/2}{s-(-a-\mathrm{j}b)} $$

\textbf{脉冲响应不变法变换：}
$$
H(z)=\frac{1/2}{1-e^{(-a+\mathrm{j}b)T}z^{-1}}+\frac{1/2}{1-e^{(-a-\mathrm{j}b)T}z^{-1}}
$$

\textbf{合并化简：}
$$
H(z)=\frac{1-z^{-1}e^{-aT}\cos(bT)}
          {1-2e^{-aT}\cos(bT)z^{-1}+e^{-2aT}z^{-2}}
$$

\item[(2)] $ H_{a}(s)=\dfrac{b}{(s+a)^{2}+b^{2}} $

\textbf{极点：}
$$ s_{1}=-a+\mathrm{j}b, \quad s_{2}=-a-\mathrm{j}b $$

\textbf{部分分式展开：}
$$ H_{a}(s)=\frac{\mathrm{j}/2}{s-(-a-\mathrm{j}b)}+\frac{-\mathrm{j}/2}{s-(-a+\mathrm{j}b)} $$

\textbf{脉冲响应不变法变换：}
$$
H(z)=\frac{\mathrm{j}/2}{1-e^{(-a-\mathrm{j}b)T}z^{-1}}
     +\frac{-\mathrm{j}/2}{1-e^{(-a+\mathrm{j}b)T}z^{-1}}
$$

\textbf{合并化简：}
$$
H(z)=\frac{z^{-1}e^{-aT}\sin(bT)}
          {1-2e^{-aT}\cos(bT)z^{-1}+e^{-2aT}z^{-2}}
$$
\end{enumerate}

\prob{5}{}{已知模拟滤波器的系统函数如下：\newline (1) $H_a(s)=\dfrac{1}{s^2+s+1}$\newline (2) $H_a(s)=\dfrac{b}{2s^2+3s+1}$\newline 试采用脉冲响应不变法和双线性变换法将其转换为数字滤波器。设 $T=2\,\text{s}$。}

解：
\textbf{解题思路：}
脉冲响应不变法通过部分分式展开将模拟滤波器极点映射为数字滤波器极点，利用 $H(z)=\sum_{k}\frac{A_k}{1-e^{s_k T}z^{-1}}$ 进行变换；双线性变换法通过 $s=\frac{2}{T}\frac{1-z^{-1}}{1+z^{-1}}$ 代入模拟传递函数，经代数化简得到数字滤波器传递函数。

Ⅰ. 脉冲响应不变法（$T=2$）

(1) $H_a(s)=\dfrac{1}{s^2+s+1}$

极点：
$$s_{1,2}=-\frac{1}{2}\pm j\frac{\sqrt{3}}{2}$$

部分分式展开：
$$H_a(s)=\frac{-j\frac{\sqrt{3}}{3}}{s-s_1}+\frac{j\frac{\sqrt{3}}{3}}{s-s_2}$$

数字滤波器传递函数：
$$H(z)=\frac{-j\frac{\sqrt{3}}{3}}{1-e^{s_1 T}z^{-1}}
      +\frac{j\frac{\sqrt{3}}{3}}{1-e^{s_2 T}z^{-1}}$$

代入 $T=2$，合并化简：
$$H(z)=\frac{2\sqrt{3}}{3}\cdot
      \frac{z^{-1}e^{-1}\sin\sqrt{3}}
           {1-2z^{-1}e^{-1}\cos\sqrt{3}+e^{-2}z^{-2}}$$

(2) $H_a(s)=\dfrac{b}{2s^2+3s+1}=\dfrac{b}{(2s+1)(s+1)}$

极点：$s_1=-\dfrac{1}{2},\ s_2=-1$

部分分式展开：
$$H_a(s)=\frac{-b}{s+1}+\frac{b}{s+\frac{1}{2}}$$

数字滤波器传递函数：
$$H(z)=\frac{-b}{1-e^{-T}z^{-1}}+\frac{b}{1-e^{-\frac{1}{2}T}z^{-1}}$$

代入 $T=2$，合并化简：
$$H(z)=\frac{bz^{-1}(e^{-1}-e^{-2})}
           {1-(e^{-1}+e^{-2})z^{-1}+e^{-3}z^{-2}}$$

Ⅱ. 双线性变换法（$T=2$）

(1) $H_a(s)=\dfrac{1}{s^2+s+1}$

代入 $s=\dfrac{2}{T}\dfrac{1-z^{-1}}{1+z^{-1}}=\dfrac{1-z^{-1}}{1+z^{-1}}$：
$$H(z)=\frac{1}{\left(\frac{1-z^{-1}}{1+z^{-1}}\right)^2
            +\frac{1-z^{-1}}{1+z^{-1}}+1}$$

化简：
$$H(z)=\frac{(1+z^{-1})^2}{(1-z^{-1})^2+(1-z^{-1})(1+z^{-1})+(1+z^{-1})^2}
      =\frac{1+2z^{-1}+z^{-2}}{3+z^{-2}}$$

(2) $H_a(s)=\dfrac{b}{2s^2+3s+1}$

代入 $s=\dfrac{1-z^{-1}}{1+z^{-1}}$：
$$H(z)=\frac{b}{2\left(\frac{1-z^{-1}}{1+z^{-1}}\right)^2
            +3\frac{1-z^{-1}}{1+z^{-1}}+1}$$

化简：
$$H(z)=\frac{b(1+z^{-1})^2}
           {2(1-z^{-1})^2+3(1-z^{-2})+(1+z^{-1})^2}
      =\frac{b(1+2z^{-1}+z^{-2})}{6-2z^{-1}}$$

     \item \textbf{现代滤波器}：适用于有用信号与干扰信号占据相同频带的情况，利用随机信号的统计规律从噪声干扰中最佳地提取信号。

     \item \textbf{IIR 滤波器}：无限长单位脉冲响应，系统网络结构中存在反馈回路。

   \end{itemize}

\prob{6-4}{}{设计一个巴特沃斯低通滤波器，要求通带截止频率 $f_p = 6\text{ kHz}$，通带最大衰减 $\alpha_p = 3\text{dB}$，阻带截止频率 $f_s = 12\text{kHz}$，阻带最小衰减 $\alpha_s = 25\text{dB}$。求出滤波器规一化系统函数 $H_a(p)$ 以及实际的 $H_a(s)$。}

\textbf{解：}

\textbf{解题思路：}根据巴特沃斯低通滤波器的设计步骤，首先由给定的通带与阻带指标确定阶数 $N$，随后查表获得归一化系统函数 $H_a(p)$，最后通过截止频率 $\Omega_c$ 去归一化得到实际系统函数 $H_a(s)$。

（1）\textbf{确定技术指标}
\[
\alpha_p = 3\ \text{dB},\quad \Omega_p = 2\pi f_p = 12\pi\ \text{k rad/s}
\]
\[
\alpha_s = 25\ \text{dB},\quad \Omega_s = 2\pi f_s = 24\pi\ \text{k rad/s}
\]

（2）\textbf{求阶数 $N$}
\[
k_{sp} = \sqrt{\frac{10^{-\alpha_p/10} - 1}{10^{-\alpha_s/10} - 1}} = 0.56,\quad
\lambda_{sp} = \frac{\Omega_s}{\Omega_p} = 2
\]
\[
N = -\frac{\lg k_{sp}}{\lg \lambda_{sp}} = 4.16 \quad \Rightarrow \quad N = 5
\]

（3）\textbf{归一化系统函数 $H_a(p)$}
\[
H_a(p) = \frac{1}{p^5 + b_4 p^4 + b_3 p^3 + b_2 p^2 + b_1 p + b_0}
\]
其中 $b_4 = 3.2361,\ b_3 = 5.2361,\ b_2 = 5.2361,\ b_1 = 3.2361,\ b_0 = 1.0000$。

（4）\textbf{3 dB 截止频率 $\Omega_c$}
\[
\Omega_c = \Omega_p (10^{0.1\alpha_p} - 1)^{1/2N}
= 11.9943\pi\ \text{k rad/s} \approx 37681\ \text{rad/s}
\]

（5）\textbf{去归一化得实际系统函数 $H_a(s)$}
令 $p = s / \Omega_c$，代入可得$H_a(s)$，这里不再展开。

\prob{7-2}{}{
  试述线性相位 FIR 数字滤波器的条件和幅度特性。
}

\noindent\textbf{一、线性相位条件}

\begin{itemize}
  \item 第一类（偶对称）：$h(n)$ 为实序列，且对 $\frac{N-1}{2}$ 偶对称，即
        $h(n) = h(N-n-1)$；
  \item 第二类（奇对称）：$h(n)$ 为实序列，且对 $\frac{N-1}{2}$ 奇对称，即
        $h(n) = -h(N-n-1)$。
\end{itemize}

\noindent\textbf{二、幅度特性与可实现滤波器类型}

\begin{enumerate}
  \item $h(n)$ 偶对称，$N$ 为奇数：\\
        $H_g(\omega)$ 关于 $\omega=0,\pi,2\pi$ 偶对称，且
        $H_g(0)=H_g(\pi)=H_g(2\pi)\neq0$，
        可实现低通、高通、带通、带阻等各类滤波器。

  \item $h(n)$ 偶对称，$N$ 为偶数：\\
        $H_g(\omega)$ 关于 $\omega=\pi$ 奇对称，关于 $\omega=0,2\pi$ 偶对称，
        且 $H_g(\pi)=0$，
        只适合实现低通、带通，不适合高通和带阻。

  \item $h(n)$ 奇对称，$N$ 为奇数：\\
        $H_g(\omega)$ 关于 $\omega=0,\pi,2\pi$ 奇对称，且
        $H_g(0)=H_g(\pi)=H_g(2\pi)=0$，
        只适合实现高通和带阻滤波器。

  \item $h(n)$ 奇对称，$N$ 为偶数：\\
        $H_g(\omega)$ 关于 $\omega=0,2\pi$ 奇对称，关于 $\omega=\pi$ 偶对称，
        且 $H_g(\pi)=H_g(2\pi)=0$，
        只适合实现高通或带通，不适合低通和带阻。
\end{enumerate}

\prob{7-4}{}{
    试述较理想的窗函数应该符合什么标准？
}

\textbf{解：}

\begin{itemize}
  \item 窗函数的频谱\textbf{\textcolor{myred}{主瓣宽度尽量窄}}，以期获得较陡峭的过渡带特性；
  \item \textbf{\textcolor{myred}{旁瓣应尽量小，且其幅度与主瓣相比应尽可能少}}，即窗口频谱的能量应尽量集中于主瓣，借以减少肩峰与余振，从而增大帯阻衰减。
\end{itemize}

\prob{7-5}{}{

\textbf{解：}

1、在理想特性的不连续点\textbf{\textcolor{myred}{$\omega = \omega_c$附近形成过渡带}}，过渡带的带宽宽度近似等于窗函数频谱的主瓣宽度$\frac{4\pi}{N}$；  

2、通带内产生了波动，阻带内产生了余振，\textbf{\textcolor{myred}{最大负峰出现在$\omega = \omega_c + \frac{2\pi}{N}$处}}，通带与阻带中波动的情况与窗函数的幅度特性有关。$N$越大，窗函数幅频特性的波动越快，通带、阻带内的波动也就越快。滤波器波动的大小取决于窗函数旁瓣的大小。

\[

\[

\]
\[

  -\sin[\omega_c(n-\alpha)]\bigr\}
\]

即：高通 = 全通 $-$ 低通。

\[

\concepttable{
    \conrow{FT}{$X(j\Omega) = \int_{-\infty}^{\infty} x(t)e^{-j\Omega t}dt$，时域连续非周期 $\mapsto$ 频域连续复函数}
    \conrow{IFT}{$x(t) = \frac{1}{2\pi} \int_{-\infty}^{\infty} X(j\Omega)e^{j\Omega t}d\Omega$，归一化系数 $1/2\pi$}
    \conrow{频率变量}{\textbf{\textcolor{myred}{$\Omega$ 为模拟角频率（rad/s）}}，区别于数字角频率 $\omega$}

\concepttable{

    \conrow{FS 系数}{$C_m = \frac{1}{T_0} \int_{T_0} x(t)e^{-jm\omega_0 t}dt$}
    \conrow{FS vs.\ FT}{\textbf{\textcolor{myred}{时域周期 $\mapsto$ 频域离散（谱线）；非周期 $\mapsto$ 连续谱}}}
}

\concepttable{
    \conrow{时域卷积}{$x(t) * y(t) \overset{\text{FT}}{\leftrightarrow} X(j\Omega) \cdot Y(j\Omega)$}
    \conrow{系统分析}{$y(t)=x(t)*h(t) \;\Rightarrow\; Y(j\Omega)=X(j\Omega)H(j\Omega)$}
    \conrow{频域卷积}{$x(t) \cdot y(t) \overset{\text{FT}}{\leftrightarrow} \frac{1}{2\pi} X(j\Omega) * Y(j\Omega)$}

    \conrow{3. 弱条件（功率有限）}{$\lim_{T\to\infty}\frac{1}{2T}\int_{-T}^{T}|x(t)|^2dt<\infty$}
    \conrow{4. 周期信号}{周期信号为功率有限信号的子类；可用FS求$C_m$，也可用广义FT}

    \comprow{$\sum_{n=-\infty}^{\infty}\delta(t-nT)$ & \multicolumn{3}{L}{$\frac{2\pi}{T}\sum_{k=-\infty}^{\infty}\delta\!\left(\Omega-\frac{2\pi k}{T}\right)$}}

\textbf{数学证明}

设 $x_a(t)=A\cos(2\pi F_0 t+\theta)$，另取 $F_k = F_0 + kF_s\ (k \in \mathbb{N})$，则 $x_a'(t)=A\cos(2\pi F_k t+\theta)$。

以 $F_s$ 采样，$t=n/F_s$：
$$x'(n)=A\cos\!\left(2\pi\frac{F_0+kF_s}{F_s}n+\theta\right)=A\cos\!\left(\frac{2\pi nF_0}{F_s}+\theta+2\pi kn\right)$$

由 $\cos(\phi+2\pi kn)=\cos(\phi)$（$k,n\in\mathbb{Z}$），得
$$x'(n)=A\cos\!\left(\frac{2\pi nF_0}{F_s}+\theta\right)=x(n)$$
